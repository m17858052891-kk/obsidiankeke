>       and   "${end}"
>       and   is_td_finish = 1
>       and   source_type <> 2
> )
> select  dis_group, avg(est_price_amt) from order_info group by dis_group;
> ```


## 累计应答率阈值测算

**逻辑**：选取所有有补勾行为的订单中，最终没有完单的订单，统计其累计应答率分布。然后分别选择70%分位点和85%分位点，分别作为加价快走和默勾的累计应答率的阈值。

选取了所有有补勾行为的订单，按照最终是否被应答分成两组，统计出各组累计应答率（取max_ar）的分布

> [!example]- SQL 实现细节（101 行）
> ```SQL
> with
> ar_log as 	-- 获取所有经过ar模型的cp的预测值
> (
>     select  param ['traceid'] as trace_id
>             ,param ['estimate_id'] as estimate_id
>             ,param['predict_value'] as predict_value
>     from    kflower_strategy.ods_log_kflower_ether_price_anchor_cr_model
>     where   concat_ws('-', year, month, day) between "${begin}"
>       and   "${end}"
>       and   param['version'] = 'version-77557774'
>       and   param['city'] in (8, 26, 32, 6, 13, 58, 85, 25)
> )
> ,anycar_order_info as 	-- 所有呼叫订单cp打平数据
> (
>     select  order_id
>             ,bubble_trace_id
>             ,product_id
>             ,estimate_id
>             ,product_2level
>             ,start_city_id as city_id
>             ,is_wait_call
>     from    kflower_dw.dwd_kf_trd_order_anycar_di
>     where   dt between "${begin}"
>       and   "${end}"
>       and   is_td_call = 1	-- 当天呼叫
>       and   source_type <> 2	-- 去掉改派
>       and   product_1level <> 4	-- 去掉pbd
>       and   level_type not in (105, 200)	-- 盒子呼叫去掉：
> )
> ,wait_call_order_info as 	-- 过滤出有补勾的订单
> (
>     select  order_id
>             ,bubble_trace_id
>             ,sum(is_wait_call) as wait_call_flag
>     from    anycar_order_info
>     group by order_id, bubble_trace_id
> )
> ,order_info as 	-- 获取所有未被应答的订单
> (
>     select  bubble_trace_id
>             ,order_id
>             ,is_td_answer
>     from    kflower_dw.dwd_kf_trd_order_base_di
>     where   dt between "${begin}"
>       and   "${end}"
>       and   is_td_call = 1	-- 当天呼叫
>       --       and   is_td_answer = 0   -- 未应答
>       --       and   cancel_type = 1    -- 应答前乘客取消
>       and   product_1level <> 4	-- 去掉PBD订单
>       and   source_type <> 2	-- 去掉改派
> )
> ,target_trace_order as -- 目标订单的traceid，即所有有补勾行为的订单
> (
>     select  a.bubble_trace_id
>             ,a.order_id
>             ,is_td_answer
>     from    order_info a
>     join    wait_call_order_info b
>     on      a.bubble_trace_id = b.bubble_trace_id
>       and   a.order_id = b.order_id
> )
> ,bub_ar_info as         -- cp维度的ar数据
> (
>     select  a.*
>             ,predict_value
>             ,is_td_answer
>     from    anycar_order_info a
>     join    target_trace_order b
>     on      a.bubble_trace_id = b.bubble_trace_id
>       and   a.order_id = b.order_id
>     join    ar_log c
>     on      a.bubble_trace_id = c.trace_id
>       and   a.estimate_id = c.estimate_id
> )
> ,trace_ar_info as 
> (
>     select  city_id
>             ,bubble_trace_id
>             ,order_id
>             ,is_td_answer
>             ,max(predict_value) as trace_acc_ar
>     from    bub_ar_info
>     group by city_id, bubble_trace_id, order_id, is_td_answer
> )
> select  city_id
>         ,is_td_answer
>         ,percentile_approx(trace_acc_ar, 0.95) as threshold_95
>         ,percentile_approx(trace_acc_ar, 0.90) as threshold_90
>         ,percentile_approx(trace_acc_ar, 0.85) as threshold_85
>         ,percentile_approx(trace_acc_ar, 0.80) as threshold_80
>         ,percentile_approx(trace_acc_ar, 0.75) as threshold_75
>         ,percentile_approx(trace_acc_ar, 0.70) as threshold_70
>         ,percentile_approx(trace_acc_ar, 0.65) as threshold_65
>         ,percentile_approx(trace_acc_ar, 0.60) as threshold_60
>         ,percentile_approx(trace_acc_ar, 0.55) as threshold_55
>         ,percentile_approx(trace_acc_ar, 0.50) as threshold_50
>         ,percentile_approx(trace_acc_ar, 0.45) as threshold_45
>         ,percentile_approx(trace_acc_ar, 0.40) as threshold_40
> from    trace_ar_info
> group by city_id, is_td_answer
> ;
> ```
