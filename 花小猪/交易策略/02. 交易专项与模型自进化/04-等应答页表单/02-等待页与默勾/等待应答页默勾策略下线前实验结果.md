### 结论

直接看 AB 期，`60/70/80` 相比 `control` 的等待页链路差异仍然很明显：追加订单率高约 0.23-0.27pp，平均默勾盒子数高约 1.61-1.65，默勾呼叫率高约 5.21-5.27pp，默勾订单率低约 14pp。

但这些组间差异在 AA 期已经基本存在。做 `AB-AA` 校正后，净效果很小：追加订单率在 -0.034pp 到 +0.009pp 之间，平均默勾盒子数在 -0.033 到 +0.014 之间，默勾呼叫率在 -0.001pp 到 +0.147pp 之间。

因此，这次更合理的实验观察结论是：`60/70/80` 与 `control` 的 raw gap 主要是分组基线差异，AB 期没有看到明显新增的策略效果；三个实验组之间也没有稳定的阈值梯度收益。

> [!example]- SQL 实现细节（258 行）
> ```SQL
> -- AA/AB DID readout for waiting-answer-page default selection experiment.
> -- AA: 2026-06-01 to 2026-06-07
> -- AB: 2026-06-08 to 2026-06-14
> --
> -- result_type:
> --   raw             : period-level raw metrics for control/60/70/80.
> --   delta_vs_control: treatment minus control within the same period.
> --   did_ab_minus_aa : (AB treatment-control) - (AA treatment-control).
> --
> -- For delta_vs_control and did_ab_minus_aa rows, rate columns are decimal
> -- differences. Multiply by 100 to read them as percentage points.
>
> with order_info as (
>     select
>         passenger_id,
>         case
>             when pmod(cast(passenger_id as bigint), 10) in (0, 1) then '80'
>             when pmod(cast(passenger_id as bigint), 10) in (2, 3) then '70'
>             when pmod(cast(passenger_id as bigint), 10) in (4, 5) then '60'
>             else 'control'
>         end as exp_group,
>         case
>             when dt between '2026-06-01' and '2026-06-07' then 'AA'
>             when dt between '2026-06-08' and '2026-06-14' then 'AB'
>         end as period,
>         order_id,
>         dt,
>         trace_id
>     from kflower_dw.dwd_kf_trd_order_base_di
>     where dt between '2026-06-01' and '2026-06-14'
>       and product_1level <> 4
>       and is_td_call = 1
>       and passenger_id is not null
> ),
> wait_info as (
>     select
>         order_id,
>         trace_id,
>         city_id,
>         is_append,
>         is_skip,
>         is_first_hit,
>         estimate_id,
>         recmd_trace_id,
>         product_id,
>         selected_status,
>         car_tag,
>         car_rank,
>         level_type,
>         estimate_trace_id,
>         append_trace_id,
>         append_time,
>         dt
>     from kflower_dw.dwd_kf_trd_wait_scene_di
>     where dt between '2026-06-01' and '2026-06-14'
> ),
> append_log as (
>     select
>         estimate_id,
>         order_id,
>         max(log_type) as log_type
>     from kflower_dw.dwd_kf_trd_order_append_di
>     where dt between '2026-06-01' and '2026-06-14'
>     group by order_id, estimate_id
> ),
> joint_t1 as (
>     select
>         a.order_id,
>         max(b.log_type) as log_type,
>         count(distinct if(a.selected_status = 1, a.product_id, null)) as num_MoGou1,
>         count(distinct if(a.selected_status = 1 and a.level_type in (0, 101, 102, 103), a.product_id, null))
>           + count(distinct if(a.selected_status = 1 and a.level_type in (200, 105), a.level_type, null)) as num_MoGou2,
>         count(if(b.log_type = 2, b.log_type, null)) as num_send_ZhuiJia,
>         count(distinct if(b.log_type = 2, a.product_id, null)) as num_send_ZhuiJia_product,
>         count(distinct if(b.log_type = 2 and a.level_type in (0, 101, 102, 103), a.product_id, null))
>           + count(distinct if(b.log_type = 2 and a.level_type in (200, 105), a.level_type, null)) as num_send_ZhuiJia_hezi,
>         count(distinct if(b.log_type = 2, a.trace_id, null)) as num_ZhuiJia,
>         if(
>             sort_array(collect_list(if(b.log_type = 2, a.product_id, null)))
>               = sort_array(collect_list(if(a.selected_status = 1, a.product_id, null))),
>             1,
>             0
>         ) as is_MoGou
>     from wait_info a
>     left join append_log b
>       on a.order_id = b.order_id
>      and a.estimate_id = b.estimate_id
>     group by a.order_id
> ),
> joint_call as (
>     select
>         a.order_id,
>         a.trace_id,
>         max(b.log_type) as log_type,
>         count(*) as num_trace,
>         if(
>             sort_array(collect_list(if(b.log_type = 2, a.product_id, null)))
>               = sort_array(collect_list(if(a.selected_status = 1, a.product_id, null))),
>             1,
>             0
>         ) as is_MoGou_call
>     from wait_info a
>     left join append_log b
>       on a.order_id = b.order_id
>      and a.estimate_id = b.estimate_id
>     group by a.order_id, a.trace_id
> ),
> base1 as (
>     select
>         a.period,
>         a.passenger_id,
>         a.exp_group,
>         a.trace_id,
>         b.*,
>         c.num_moGou_call
>     from order_info a
>     left join joint_t1 b
>       on a.order_id = b.order_id
>     left join (
>         select
>             order_id,
>             count(distinct if(is_MoGou_call = 1, if(log_type = 2, trace_id, null), null)) as num_moGou_call
>         from joint_call
>         group by order_id
>     ) c
>       on a.order_id = c.order_id
> ),
> period_metrics as (
>     select
>         period,
>         exp_group,
>         cast(count(*) as double) as count_,
>         cast(sum(num_ZhuiJia) as double) as Num_zhuijia,
>         sum(if(log_type = 2, num_MoGou1, null)) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as avg_num_MoGou_product,
>         sum(if(log_type = 2, num_MoGou2, null)) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as avg_num_MoGou_hezi,
>         sum(num_send_ZhuiJia_product) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as avg_send_zhuijia_product,
>         sum(num_send_ZhuiJia_hezi) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as avg_send_zhuijia_hezi,
>         sum(num_ZhuiJia) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as avg_num_zhuijia,
>         count(distinct if(log_type = 2, order_id, null)) / nullif(count(distinct order_id), 0) as rate_ZhuiJia,
>         sum(if(log_type = 2, is_MoGou, 0)) / nullif(count(distinct if(log_type = 2, order_id, null)), 0) as rate_MoGou_order,
>         sum(if(log_type = 2, num_moGou_call, 0)) / nullif(sum(num_ZhuiJia), 0) as rate_MoGou_call
>     from base1
>     group by period, exp_group
> ),
> delta_vs_control as (
>     select
>         m.period,
>         m.exp_group,
>         cast(null as double) as count_,
>         cast(null as double) as Num_zhuijia,
>         m.avg_num_MoGou_product - c.avg_num_MoGou_product as avg_num_MoGou_product,
>         m.avg_num_MoGou_hezi - c.avg_num_MoGou_hezi as avg_num_MoGou_hezi,
>         m.avg_send_zhuijia_product - c.avg_send_zhuijia_product as avg_send_zhuijia_product,
>         m.avg_send_zhuijia_hezi - c.avg_send_zhuijia_hezi as avg_send_zhuijia_hezi,
>         m.avg_num_zhuijia - c.avg_num_zhuijia as avg_num_zhuijia,
>         m.rate_ZhuiJia - c.rate_ZhuiJia as rate_ZhuiJia,
>         m.rate_MoGou_order - c.rate_MoGou_order as rate_MoGou_order,
>         m.rate_MoGou_call - c.rate_MoGou_call as rate_MoGou_call
>     from period_metrics m
>     inner join period_metrics c
>       on m.period = c.period
>      and c.exp_group = 'control'
>     where m.exp_group <> 'control'
> ),
> did_result as (
>     select
>         'AB-AA' as period,
>         ab.exp_group,
>         cast(null as double) as count_,
>         cast(null as double) as Num_zhuijia,
>         ab.avg_num_MoGou_product - aa.avg_num_MoGou_product as avg_num_MoGou_product,
>         ab.avg_num_MoGou_hezi - aa.avg_num_MoGou_hezi as avg_num_MoGou_hezi,
>         ab.avg_send_zhuijia_product - aa.avg_send_zhuijia_product as avg_send_zhuijia_product,
>         ab.avg_send_zhuijia_hezi - aa.avg_send_zhuijia_hezi as avg_send_zhuijia_hezi,
>         ab.avg_num_zhuijia - aa.avg_num_zhuijia as avg_num_zhuijia,
>         ab.rate_ZhuiJia - aa.rate_ZhuiJia as rate_ZhuiJia,
>         ab.rate_MoGou_order - aa.rate_MoGou_order as rate_MoGou_order,
>         ab.rate_MoGou_call - aa.rate_MoGou_call as rate_MoGou_call
>     from delta_vs_control ab
>     inner join delta_vs_control aa
>       on ab.exp_group = aa.exp_group
>      and ab.period = 'AB'
>      and aa.period = 'AA'
> )
> select
>     'raw' as result_type,
>     period,
>     exp_group,
>     cast(null as string) as baseline_group,
>     count_,
>     Num_zhuijia,
>     avg_num_MoGou_product,
>     avg_num_MoGou_hezi,
>     avg_send_zhuijia_product,
>     avg_send_zhuijia_hezi,
>     avg_num_zhuijia,
>     rate_ZhuiJia,
>     rate_MoGou_order,
>     rate_MoGou_call
> from period_metrics
>
> union all
>
> select
>     'delta_vs_control' as result_type,
>     period,
>     exp_group,
>     'control' as baseline_group,
>     count_,
>     Num_zhuijia,
>     avg_num_MoGou_product,
>     avg_num_MoGou_hezi,
>     avg_send_zhuijia_product,
>     avg_send_zhuijia_hezi,
>     avg_num_zhuijia,
>     rate_ZhuiJia,
>     rate_MoGou_order,
>     rate_MoGou_call
> from delta_vs_control
>
> union all
>
> select
>     'did_ab_minus_aa' as result_type,
>     period,
>     exp_group,
>     'control' as baseline_group,
>     count_,
>     Num_zhuijia,
>     avg_num_MoGou_product,
>     avg_num_MoGou_hezi,
>     avg_send_zhuijia_product,
>     avg_send_zhuijia_hezi,
>     avg_num_zhuijia,
>     rate_ZhuiJia,
>     rate_MoGou_order,
>     rate_MoGou_call
> from did_result
> order by
>     case result_type
>         when 'raw' then 0
>         when 'delta_vs_control' then 1
>         when 'did_ab_minus_aa' then 2
>         else 9
>     end,

>         when 'AA' then 0
>         when 'AB' then 1
>         when 'AB-AA' then 2
>         else 9
>     end,
>     case exp_group
>         when 'control' then 0
>         when '60' then 1
>         when '70' then 2

- 相关口径说明

	- **追加发单**：用户在等待应答页触发了追加发单，把更多车型/盒子发出去。

	- **默勾**：在追加发单页面里，系统预先默认勾选了一些车型/盒子，用户是否沿用了这些默认勾选。

	- **追加订单率** = 发生追加发单的订单数 / 进入等待页分析链路的订单数

	- 平均默勾盒子数 = 发生追加发单订单的默勾盒子数总和 / 发生追加发单的订单数

	- 默勾订单率 = 实际追加发单车型集合与默认勾选车型集合完全一致的订单数 / 发生追加发单的订单数

	- 默勾呼叫率 = 实际发单车型集合与默认勾选车型集合完全一致的追加 trace 数 / 追加 trace 总数
