## 附录（代码和原始数据）

> [!example]- SQL 实现细节（79 行）
> ```SQL
> with bub_info as (
>     select
>         trace_id,
>         max(estimate_fee_amt) as max_selected_fee
>     from
>         kflower_dw.dwd_kf_trd_bubble_base_all_di
>     where
>         dt between "${begin}"
>         and "${end}"
>         and product_1level <> 4 -- 不是PBD
>         and is_from_waiting_response != 1 -- 不是等待应答页
>         and selected_status = 1 -- 默勾品牌
>     group by
>         trace_id
> ),
> order_bub_info as (
>     select
>         *
>     from
>         kflower_dw.dwm_kf_trd_order_bubble_di
>     where
>         dt between "${begin}"
>         and "${end}"
>         and selected_cp_cnt > 0
>         and form_type = 3 -- 聚合表单
>         and order_type = 0 -- 实时单
>         and source_type <> 2 -- 去掉改派单
> ),
> athena_info as (
>     select
>         trace_id,
>         select_anchor_group
>     from
>         kflower_dw.dwd_kf_trd_bubble_athena_stg_di
>     where
>         dt between "${begin}"
>         and "${end}"
> ),
> -- add_info as (
> --     select
> --         *
> --     from
> --         kflower_strategy_dev.dwm_kf_trd_order_bubble_di_cgh
> --     where
> --         dt between "${begin}"
> --         and "${end}"
> -- ),
> merged_info as (
>     select
>         a.*,
>         b.max_selected_fee,
>         c.select_anchor_group
>     from
>         order_bub_info a
>         left join bub_info b on a.bubble_trace_id = b.trace_id
>         left join athena_info c on a.bubble_trace_id = c.trace_id
> )
> select
>     select_anchor_group,
>     count(
>         distinct if(
>             max_selected_fee = int(min_estimate_fee * 100),
>             bubble_trace_id,
>             null
>         )
>     ) as select_min_cnt,
>     count(distinct bubble_trace_id) as all_cnt,
>     count(
>         distinct if(
>             max_selected_fee = int(min_estimate_fee * 100)
>             and only_selected_call_cnt = 1,
>             bubble_trace_id,
>             null
>         )
>     ) as single_selection_accept_cnt
> from
>     merged_info
> group by
>     select_anchor_group;
> ```
