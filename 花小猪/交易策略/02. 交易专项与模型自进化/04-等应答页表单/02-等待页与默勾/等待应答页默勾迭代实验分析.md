> [!example]- SQL 实现细节（144 行）
> ```SQL
>
> with
> pas_tag as 
> (
>     select  distinct apollo_key as passenger_id
>             ,exp_group
>             ,cast(dims['city'] as int) as city_id
>     from    apollo.exp_group_data
>     where   ymd between '${begin}' and '${end}'-- between '${begin}' and '${end}'    -- 4weeks
>       and   experiment = 'athena_sup_select_exp'	-- 实验名称  
> )
> ,order_info as 
> (   select  passenger_id 
>             ,order_id
>             ,dt
>             ,trace_id
>     from    kflower_dw.dwd_kf_trd_order_base_di
>     where   dt between '${begin}' and '${end}' --= '2024-04-11' -- between '${begin}' and '${end}'--between '${begin}' and '${end}'
>     and     product_1level <> 4	-- 去掉pbd的流量
>     and     is_td_call = 1    
> )
> ,order_pas as 
> (
>     select  a.passenger_id
>             ,b.exp_group
>             ,a.order_id
>             ,a.trace_id
>     from    order_info a
>     left join pas_tag b on a.passenger_id = b.passenger_id
>
> )
> , wait_info as (
>     select
>         order_id,
>         trace_id,
>         city_id,
>         is_append,
>         is_skip,
>         is_first_hit,
>         estimate_id,
>         recmd_trace_id,
>         product_id,-- CAST(product_id AS CHAR) as product_id,
>         selected_status,
>         car_tag,
>         car_rank,
>         level_type,
>         estimate_trace_id,
>         append_trace_id,
>         append_time,
>         dt
>     from
>         kflower_dw.dwd_kf_trd_wait_scene_di
>     where
>         dt between '${begin}' and '${end}'--= '2024-04-11' -- between '${begin}' and '${end}'
>     -- order by product_id
> ),
> append_log as (
>     select
>         estimate_id,
>         order_id,
>         max(log_type) as log_type       -- 判断是否为追加订单（有一个为2即为追加）
>     from
>         kflower_dw.dwd_kf_trd_order_append_di
>     where
>         dt between '${begin}' and '${end}'--'2024-04-11' -- between '${begin}' and '${end}'
>     group by
>         order_id,
>         estimate_id
> )
> ,joint_t1 as 
> (
>     select  a.order_id
>             ,max(b.log_type) as log_type
>             ,count(distinct if(selected_status=1,product_id,null)) as num_MoGou1
>             ,count(distinct if(selected_status=1 and level_type in (0,101,102,103),product_id,null)) + count(distinct if(selected_status=1 and level_type in (200,105),level_type,null)) as num_MoGou2
>             ,count(if(log_type=2,log_type,null)) as num_send_ZhuiJia        -- 发单量
>             ,count(distinct if(log_type=2,product_id,null)) as num_send_ZhuiJia_product     --基于product发单量
>             ,count(distinct if(log_type=2 and level_type in (0,101,102,103),product_id,null)) + count(distinct if(log_type=2 and level_type in (200,105),level_type,null)) as num_send_ZhuiJia_hezi
>             ,count(distinct if(log_type=2,trace_id,null)) as num_ZhuiJia    -- 追加次数（trace_id种类）
>             ,if(sort_array(collect_list(if(log_type=2,product_id,null))) = sort_array(collect_list(if(selected_status=1,product_id,null))),1,0) as is_MoGou
>     from    wait_info a
>     left join append_log b
>     on      a.order_id = b.order_id
>     and   a.estimate_id = b.estimate_id
>     group by a.order_id
> ) 
> ,joint_call as
> (
>     select  a.order_id
>             ,a.trace_id
>             ,max(b.log_type) as log_type
>             ,count(*) as num_trace
>             ,if(sort_array(collect_list(if(log_type=2,product_id,null))) = sort_array(collect_list(if(selected_status=1,product_id,null))),1,0) as is_MoGou_call
>     from    wait_info a
>     left join append_log b
>     on      a.order_id = b.order_id
>     and   a.estimate_id = b.estimate_id
>     group by a.order_id,a.trace_id
> )
> ,base1 as 
> (
>     select  a.passenger_id
>             ,a.exp_group
>             ,a.trace_id
>             ,b.*
>             ,c.num_moGou_call
>     from    order_pas a
>     left join joint_t1 b on a.order_id = b.order_id
>     left join (    
>                 select  order_id
>                 ,count(distinct if(is_MoGou_call=1,if(log_type=2,trace_id,null),null)) as num_moGou_call
>                 -- ,count(distinct is_MoGou_call*trace_id) as num_moGou_call
>                 from    joint_call
>                 group by order_id
>              ) c on a.order_id = c.order_id
> )
> ,group_res1 as 
> (
>     select  exp_group
>             ,count(*) as count_
>             ,sum(num_ZhuiJia) as Num_zhuijia
>             ,sum(if(log_type=2,num_MoGou1,null))/count(distinct if(log_type=2, order_id, null)) as avg_num_MoGou_product
>             ,sum(if(log_type=2,num_MoGou2,null))/count(distinct if(log_type=2, order_id, null)) as avg_num_MoGou_hezi
>             -- ,sum(num_send_ZhuiJia) / count(distinct if(log_type=2, order_id, null)) as avg_send_zhuijia
>             ,sum(num_send_ZhuiJia_product) / count(distinct if(log_type=2, order_id, null)) as avg_send_zhuijia_product
>             ,sum(num_send_ZhuiJia_hezi) / count(distinct if(log_type=2, order_id, null)) as avg_send_zhuijia_hezi
>             ,sum(num_ZhuiJia) / count(distinct if(log_type=2, order_id, null)) as avg_num_zhuijia
>             ,count(distinct if(log_type=2, order_id, null))/count(distinct order_id) as rate_ZhuiJia
>             ,sum(if(log_type=2,is_MoGou,0))/count(distinct if(log_type=2, order_id, null)) as rate_MoGou_order
>             ,sum(if(log_type=2,num_moGou_call,0)) / sum(num_ZhuiJia) as rate_MoGou_call       
>     from    base1
>     group by exp_group
> )
> select * from group_res1;
> -- ,base2 as 
> -- (
> --     select  a.passenger_id
> --             ,a.exp_group
> --             ,a.trace_id
> --             ,b.*
> --             -- ,c.is_MoGou_call
> --     from    order_pas a
> --     left join joint_t1 b on a.trace_id = b.trace_id   
> -- )
> ```
