## 四、随机默勾组口径

口径中select_ratio即为随机默勾组的默勾比例

> [!example]- PlainText 实现细节（96 行）
> ```PlainText
> WITH
>
> exp_groups AS (
>     SELECT DISTINCT
>         param['trace_id']           AS trace_id,
>         param['random_select_rule'] AS random_select_rule
>     FROM kflower_ods.ods_log_kflower_g_order_prefee_show
>     WHERE concat_ws('-', year, month, day) = '2026-04-27'
>         AND param['random_select_rule'] IN (
>             'selected_20',
>             'selected_40',
>             'selected_60',
>             'selected_80',
>             'random',
>             'single_lowest'
>         )
> ),
>
> base_bubble AS (
>     SELECT
>         b.trace_id,
>         b.level_type,
>         b.price_type,
>         b.estimate_fee_amt,
>         b.show_estimate_fee,
>         b.selected_status,
>         b.car_merge_type,
>         b.car_merger_rank,
>         b.car_rank
>     FROM kflower_dw.dwd_kf_trd_bubble_base_all_di b
>     WHERE b.dt = '2026-04-27'
>         AND b.is_from_waiting_response != 1
>         AND b.is_anycar_form = 1
>         AND b.is_fold_cp <> 1
>         AND b.product_1level <> 4
>         AND b.city_id = 62
> ),
>
> -- 只取 random 分组的数据
> bubble_random AS (
>     SELECT b.*, g.random_select_rule
>     FROM base_bubble b
>     INNER JOIN exp_groups g ON b.trace_id = g.trace_id
>     WHERE g.random_select_rule = 'random'
> ),
>
> -- 为每行打上合并组 key（规则同输出1）
> bubble_random_with_key AS (
>     SELECT
>         *,
>         CASE
>             WHEN car_merge_type = 'sameprice' AND car_merger_rank > 0
>                 THEN concat('sameprice_', car_merger_rank)
>             ELSE concat('car_rank_', car_rank)
>         END AS merge_group_key
>     FROM bubble_random
> ),
>
> -- 同价车型合并
> merged_random AS (
>     SELECT
>         trace_id,
>         merge_group_key,
>         MAX(selected_status) AS selected_status
>     FROM bubble_random_with_key
>     GROUP BY trace_id, merge_group_key
> ),
>
> -- 按 trace_id 计算默勾比例
> trace_stats_random AS (
>     SELECT
>         trace_id,
>         COUNT(*)                             AS total_merged_car_count,
>         SUM(selected_status)                   AS total_selected_count,
>         SUM(selected_status) * 1.0 / COUNT(*) AS select_ratio
>     FROM merged_random
>     GROUP BY trace_id
> )
>
> -- 步骤 4【输出2】：默勾比例分位点
> SELECT
>     COUNT(*)                              AS total_trace_count,
>     ROUND(AVG(select_ratio), 4)           AS avg_select_ratio,
>     approx_percentile(select_ratio, 0.00) AS p0,
>     approx_percentile(select_ratio, 0.10) AS p10,
>     approx_percentile(select_ratio, 0.20) AS p20,
>     approx_percentile(select_ratio, 0.30) AS p30,
>     approx_percentile(select_ratio, 0.40) AS p40,
>     approx_percentile(select_ratio, 0.50) AS p50,
>     approx_percentile(select_ratio, 0.60) AS p60,
>     approx_percentile(select_ratio, 0.70) AS p70,
>     approx_percentile(select_ratio, 0.80) AS p80,
>     approx_percentile(select_ratio, 0.90) AS p90,
>     approx_percentile(select_ratio, 1.00) AS p100
> FROM trace_stats_random
> ;
> ```
