select
    metric,
    round(metric_t, 4) as metric_t,
    round(metric_c, 4) as metric_c,
    N_t,
    N_c,
    round(diff, 4) as diff,
    concat(round(lift * 100, 2), '%') as lift,
    round(upper_bound, 4) as upper_bound,
    round(lower_bound, 4) as lower_bound,
    case when
        upper_bound * lower_bound > 0 then '显著'
        else '不显著'
    end as siginificance
from
    (
        select
            metric,
            metric_t,
            metric_c,
            N_t,
            N_c,
            sum_ss_t,
            sum_ss_c,
            metric_t - metric_c as diff,
            case
                when metric_c <> 0 then (metric_t - metric_c) / metric_c
            end as lift,
            case
                when type = 'binomial' then metric_t - metric_c + 1.96 * sqrt(metric_t * (1 - metric_t) / N_t + metric_c * (1 - metric_c) / N_c)
                when type = 'gaussian' then metric_t - metric_c + 1.96 * sqrt(1 / N_t / (N_t -1) * (sum_ss_t - N_t * metric_t * metric_t) + 1 / N_c /(N_c -1) * (sum_ss_c - N_c * metric_c * metric_c))
                when type = 'ratio' then metric_t - metric_c + 1.96 * sqrt(metric_a * (1 - metric_a) * (1 / N_t + 1 / N_c))
            end as upper_bound,
            case
                when type = 'binomial' then metric_t - metric_c - 1.96 * sqrt(metric_t * (1 - metric_t) / N_t + metric_c * (1 - metric_c) / N_c)
                when type = 'gaussian' then metric_t - metric_c - 1.96 * sqrt(1 / N_t / (N_t -1) * (sum_ss_t - N_t * metric_t * metric_t) + 1 / N_c / (N_c -1) *(sum_ss_c - N_c * metric_c * metric_c))
                when type = 'ratio' then metric_t - metric_c - 1.96 * sqrt(metric_a * (1 - metric_a) * (1 / N_t + 1 / N_c))

                    sum(if(exp_group = 'control_group', dd_fin_cnt * dd_fin_cnt, 0)) as sum_ss_c

-- 使用方式：
-- 1. AA、AB 各运行一次，period_name 分别传入 AA、AB；
-- 2. 同时将本文件中的日期改为对应周期；

,user_stats as
(

        a.pid,
        a.exp_group,
        count(distinct b.origin_order_id) as origin_order_cnt,
        count(distinct case when b.is_last_finish = 1 then b.origin_order_id end) as finish_order_cnt,

        a.pid,

)

        count(1) as user_cnt,

        sum(origin_order_cnt) as sum_origin_order,
        sum(origin_order_cnt * origin_order_cnt) as sum_sq_origin_order,

        -- 人均归一完单数
        sum(finish_order_cnt) as sum_finish_order,
        sum(finish_order_cnt * finish_order_cnt) as sum_sq_finish_order,

        -- 每个归一完单的呼叫-计费时长：分子 / 分母
        sum(call_to_firstcharge_dur) as sum_call_to_firstcharge_dur,
        sum(call_to_firstcharge_dur * call_to_firstcharge_dur) as sum_sq_call_to_firstcharge_dur,
        sum(call_to_firstcharge_dur * finish_order_cnt) as sum_cross_charge_dur_finish
    from
        user_stats
    where
        exp_group is not null
    group by
        exp_group
)

select *
from group_stats
order by period_name, exp_group
;

                when order_status in ('完单') then unix_timestamp(begin_charge_time) - unix_timestamp(answer_time) - driver_to_last_dest_eta
                when order_status not in ('完单') and (unix_timestamp(cancel_time) - unix_timestamp(answer_time)) > driver_to_last_dest_eta then (unix_timestamp(cancel_time) - unix_timestamp(answer_time) - driver_to_last_dest_eta)
                when order_status not in ('完单') and (unix_timestamp(cancel_time) - unix_timestamp(answer_time)) <= driver_to_last_dest_eta then 0
            end) as pickup_dur,
        sum(case when order_status in ('完单') then unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time) end) as charge_dur
    from 
        order_rank_max
    where   
        platform_mode_name not in ('花小猪自营','宏鹄')
        and is_answer_serial_assign = 1
    group by 
        dt,
        passenger_id
    
    union all 

    select 
        dt,
        passenger_id,
        '非可控运力' as type,
        '非连环派' as is_serial,
        sum(case
                when order_status in ('完单') then unix_timestamp(begin_charge_time) - unix_timestamp(answer_time)
                when order_status not in ('完单') then unix_timestamp(cancel_time) - unix_timestamp(answer_time)
            end) as pickup_dur,
        sum(case when order_status in ('完单') then unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time) end) as charge_dur
    from 
        order_rank_max
    where   
        platform_mode_name not in ('花小猪自营','宏鹄')
        and is_td_answer = 1
        and is_answer_serial_assign = 0
    group by 
        dt,
        passenger_id

-- ============================================================
-- Excel 显著性计算基础统计量（乘客粒度）
--
-- 使用方式：
-- 1. AA、AB 各运行一次，period_name 分别传入 AA、AB；
-- 2. 同时将本文件中的日期改为对应周期；
-- 3. 将两次结果追加到同一张 Excel 数据表。
--
-- 四个指标都是比率指标。输出用户粒度分子、分母的和、平方和及
-- 交叉乘积和，用于 delta method 计算标准误。
-- ============================================================
,user_stats as
(
    select
        a.pid,
        a.exp_group,
        sum(b.`接驾时间`) as pickup_dur,
        sum(b.`送驾时间`) as charge_dur,
        sum(c.`归一订单数`) as origin_order_cnt,
        sum(c.`归一应答订单数`) as answer_order_cnt,
        sum(c.`归一完单订单数`) as finish_order_cnt
    from
        exp_user a
    join
        raw1 b
        on a.pid = b.passenger_id
        and a.earliest_group_dt <= b.dt -- 只看最早进组后的数据
    join
        raw3 c
        on b.passenger_id = c.passenger_id
        and b.dt = c.dt -- 按天对齐，避免同一乘客的多天数据交叉放大
    group by
        a.pid,
        a.exp_group
)

,group_stats as
(
    select
        'AB' as period_name,
        exp_group,
        count(1) as user_cnt,

        -- 公共分子：接驾时间
        sum(pickup_dur) as sum_pickup_dur,
        sum(pickup_dur * pickup_dur) as sum_sq_pickup_dur,

        -- 归一订单接驾时长：接驾时间 / 归一订单数
        sum(origin_order_cnt) as sum_origin_order,
        sum(origin_order_cnt * origin_order_cnt) as sum_sq_origin_order,
        sum(pickup_dur * origin_order_cnt) as sum_cross_pickup_origin_order,

        -- 归一应答订单接驾时长：接驾时间 / 归一应答订单数
        sum(answer_order_cnt) as sum_answer_order,
        sum(answer_order_cnt * answer_order_cnt) as sum_sq_answer_order,
        sum(pickup_dur * answer_order_cnt) as sum_cross_pickup_answer_order,

        -- 归一完单订单接驾时长：接驾时间 / 归一完单订单数
        sum(finish_order_cnt) as sum_finish_order,
        sum(finish_order_cnt * finish_order_cnt) as sum_sq_finish_order,
        sum(pickup_dur * finish_order_cnt) as sum_cross_pickup_finish_order,

        -- 接驾/送驾：接驾时间 / 送驾时间
        sum(charge_dur) as sum_charge_dur,
        sum(charge_dur * charge_dur) as sum_sq_charge_dur,
        sum(pickup_dur * charge_dur) as sum_cross_pickup_charge_dur
    from
        user_stats
    where
        exp_group is not null
    group by
        exp_group
)

select *
from group_stats
order by period_name, exp_group
;

(

            when is_td_answer = 1 and is_td_finish = 1 then '完单' --改派单当日无应答算司机取消，跨天应答完单的单可能会有司机取消记录。完单优先级最高
            when is_td_answer = 1 and is_grab_after_dri_cancel = 1 and is_grab_after_dri_cancel_without_reassign = 1 then '司机取消无改派'
            when is_td_answer = 1 and is_grab_after_dri_cancel = 1 and is_grab_after_dri_cancel_without_reassign <> 1 then '司机取消改派'
            when is_td_answer = 1 and is_grab_after_pas_cancel = 1 and is_grab_after_pas_cancel_without_reassign = 1 then '乘客取消无改派'
            when is_td_answer = 1 and is_grab_after_pas_cancel = 1 and is_grab_after_pas_cancel_without_reassign <> 1 then '乘客取消改派'
            when is_td_answer <> 1 and is_grab_after_pas_cancel_without_reassign = 1 then '有过应答且最后一单应答前乘客取消'
            when is_td_answer = 1 and is_grab_after_srvc_cancel = 1 then '客服取消无改派'

origin_order_detail as --原始订单id粒度统计
(
    select 
        *
        ,to_date(call_time_min) as dt
        ,hour(call_time_min) as tid
        ,unix_timestamp(cancel_time_all_min) - unix_timestamp(call_time_min) as call_to_firstcancel_dur
        ,unix_timestamp(answer_time_min) - unix_timestamp(call_time_min) as call_to_firstans_dur -- 发单到首次应答时间（秒），除以不含改应答量就是单均应答时长
        ,unix_timestamp(begin_charge_time_min) - unix_timestamp(call_time_min) as call_to_firstcharge_dur -- 发单到首次开始计费时间，除以完单数就是单均呼叫-计费时长
        ,unix_timestamp(arrive_time_min) - unix_timestamp(call_time_min) as call_to_firstarrive_dur -- 发单到首次到达时间，除以完单数就是单均呼叫-计费时长
    from   
    (
        select 
            origin_order_id
            ,max(city_id) city_id
            ,max(passenger_id) passenger_id 
            --本次发单首次发单时间
            ,min(case when is_td_call=1 then call_time end) call_time_min
            --本次发单首次和末次被应答时间
            ,min(case when is_td_answer=1 then answer_time end) answer_time_min
            ,max(case when is_td_answer=1 then answer_time end) answer_time_max
            --本次发单首次和末次被取消时间
            ,min(case when is_td_answer=1 and is_td_finish<>1 then cancel_time end) cancel_time_min
            ,max(case when is_td_answer=1 and is_td_finish<>1 then cancel_time end) cancel_time_max         
            --上车时间
            ,min(case when is_td_answer=1 and is_td_finish=1 then begin_charge_time end) begin_charge_time_min
            ,max(case when is_td_answer=1 and is_td_finish=1 then begin_charge_time end) begin_charge_time_max  

            ,min(case when is_td_answer=1 and is_td_finish=1 then arrive_time end) arrive_time_min
            ,max(case when is_td_answer=1 and is_td_finish=1 then arrive_time end) arrive_time_max

            ,min(case when is_td_finish<>1 and cancel_time not in ('0000-00-00 00:00:00') then cancel_time end) cancel_time_all_min
