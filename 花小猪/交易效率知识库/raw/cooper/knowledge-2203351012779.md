with base as (
select distinct
    b.dt as dt,
    city_id,
    a.trace_id,
    is_bad_supply_demand,--都是true，限定命中差供需的trace
    if(c.bubble_trace_id is not null,1,0) as is_call, --ECR
    is_td_answer, --AR
    is_td_finish, --CR 
    is_call_box,
    call_prod_cnt
from(
        select
            trace_id,
            city_id,
            passenger_id,
            combine_trace_id_by_pid
        from
            (
                select
                    trace_id,
                    city_id,
                    passenger_id,
                    combine_trace_id_by_pid,
                    row_number() over(
                        partition by combine_trace_id_by_pid
                        order by
                            bubble_time desc
                    ) as trace_seq_time_1 --最后一次冒泡
                from
                    kflower_dw.dwd_kf_trd_bubble_base_di
                where
                    dt BETWEEN '2024-11-06' and '2024-11-12'
                    and city_id = 19
                    and product_1level in (1, 2)
                    and is_from_waiting_response = 0 -- 只筛选冒泡页的trace_id
            ) z
        where
            z.trace_seq_time_1 = 1
    ) a

    --限定命中差供需的trace
    join (
        select
            distinct concat_ws('-', year, month, day) as dt,
            trace_id,
            is_bad_supply_demand
        from
            kflower_ods.ods_log_kflower_athena_stg_v3_log
        where
            concat_ws('-', year, month, day) BETWEEN '2024-11-06' and '2024-11-12'
            and city_id = 19
          --  and is_bad_supply_demand='true'  --限定命中差供需的trace
    ) b on a.trace_id = b.trace_id

    --在发单表中拿 是否应答（AR），是否完单（CR），通过left join后是否为空作为 是否发单（ECR）
    --在anycar表中拿 是否勾了盒子，是否单勾
    left join(
        select  
            bubble_trace_id
            ,max(is_td_finish) as is_td_finish
            ,max(is_td_answer) as is_td_answer
            ,max(if(product_level_id=105,1,0)) as is_call_box --是否勾盒子
            ,count(distinct product_level_id) as call_prod_cnt --发单品牌数
        from(
            select  
                bubble_trace_id
                ,max(is_td_finish) as is_td_finish
                ,max(is_td_answer) as is_td_answer
                ,CASE WHEN level_type = 105 THEN 105 -- tj
                    WHEN level_type = 200 THEN 200 -- tk
                    ELSE product_2level 
                END AS product_level_id
            from    kflower_dw.dwm_kf_trd_hxz_order_anycar_view_di
            where   dt between '2024-11-06' and '2024-11-12'
                and start_city_id = 19
                and   is_td_call = 1
                and   source_type <> 2
                and   call_time IS NOT NULL
                and   call_time <> '0000-00-00 00:00:00'
                and   order_id > 0
                and   product_1level in (1, 2, 999)
            group by bubble_trace_id
                    ,CASE WHEN level_type = 105 THEN 105 -- tj
                        WHEN level_type = 200 THEN 200 -- tk
                        ELSE product_2level 
                    END 
        )x
        group by bubble_trace_id
    )c on a.trace_id = c.bubble_trace_id

    --是否单勾
    --是否勾盒子
)

select 

tag,
count(*) as cnt,
sum(is_td_answer)/sum(is_call) as ar,
sum(is_td_finish)/sum(is_call) as cr

from(
    SELECT
    *,
    case when is_call_box=1 and call_prod_cnt=1 then '单勾盒子'
        when is_call_box <> 1 and call_prod_cnt=1 then '单勾非盒子'
        when is_call_box=1 and call_prod_cnt > 1 then '多勾盒子'
        when is_call_box <> 1 and call_prod_cnt > 1 then '多勾非盒子'
    end as tag

    FROM base
    where is_call =1
)
group by tag
