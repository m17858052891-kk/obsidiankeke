                    all_time

                            (
                                select
                                    hxz_driver_id driver_id,

                                        where
                                            dt = '${V_DATE}'
                                            and is_online = 1
                                    ) hxz_on_hxz on driver_list.hxz_driver_id = hxz_on_hxz.driver_id
                            ) b on a.driver_id = b.driver_id
                            and a.time1 = b.time2
                        group by
                            dt,
                            city_id,

                    ) d
                    inner join (
                        select
                            hxz_driver_id driver_id,
                            count(timeall) all_time
                        from
                            (
                                select
                                    hxz_driver_id,
                                    didi_driver_id,
                                    city_id,
                                    dt
                                from
                                    kflower_dw.dwm_kf_dri_online_hxz_and_wyc_df

                                    hxz_online_dur > 0
