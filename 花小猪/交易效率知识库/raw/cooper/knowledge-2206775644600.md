# 260206商家竞争效率司机分流实验分析

# 零、背景与结论

背景收益链路：

1. PK必败自营鸿鹄司机过滤->PK失败od不连边->播单下降->举手上升->PK获胜上涨->自营鸿鹄应答提升->自营鸿鹄完单提升->大盘CR提升

2. 长时间无应答订单&自营鸿鹄PK失败的司机DO连边加权->播单上涨->举手上涨->PK获胜上涨->自营鸿鹄应答提升->自营鸿鹄完单提升->大盘CR提升

总收益链路：分单->召回->DO匹配->播单上涨->举手上涨->PK获胜上涨->应答上涨->完单上涨->CR上涨

目前PK必败过滤工程实现为直接不播单，对人均进播数负向影响较大-10.48%，进而影响链路人均进PK数-8.92%及PK获胜数-2.51%，应答数-2.40%及完单数-2.62%，考虑下掉PK必败过滤单独开长时间无应答订单&自营鸿鹄PK失败的司机DO连边加权实验。

with did_exp as 
(
	select  distinct apollo_key as driver_id
            ,group_name as exp_group
    from    apollo.pre_flow_data_etl
    where   concat_ws('-', year, month, day) between "2026-02-07" and "2026-02-13"
    and   exp = 'dups_exp_shangjiajingzhengxiaolv_exp_20260205141858_0'
),

handup as (
    --所有举手的od对
    select
        shiftleft(
            shiftright(cast(a.orderid as bigint), 46) & 262143,
            32
        ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
        driverid driver_id,
        driver_id_3rd_part,
        round_num
    from
        kflower_ods.ods_log_kflower_g_order_strivetender a
    where
        concat_ws('-', year, month, day) between "2026-02-07" and "2026-02-13"
),
win as (
    -- 所有胜出的od对
    select 
        shiftleft(
            shiftright(cast(orderid as bigint), 46) & 262143,
            32
        ) +(cast(orderid as bigint) & 4294967295) as order_id,
        driverid driver_id,
        round_num
    from 
        kflower_marketplace.kflower_allmatch_pk_round_info
    where
        dt between "2026-02-07" and "2026-02-13"
        and driverid = win_driverid
        and channelid = 3
        and int(is_valid) = 1
),
matrix as (
    --所有过滤后连边的od对
    select
        cast(param ['orderid'] AS bigint) AS order_id,
        b.hxz_driver_id as driver_id,
        param ['uuid'] as uuid,
        ceil(
                dense_rank () over(
                    order by
                        param ['timestamp']
                ) / 2
            ) as round_num
    from
        kflower_marketplace.ods_log_kflower_stgmatrix_hh a
        left join (
            select
                other_driver_id,
                hxz_driver_id
            from
                kflower_strategy.pk_hxz_driver_id
            where
                concat_ws('-', year, month, day)="2026-02-21"
        ) b on a.param ['driverid'] = b.other_driver_id
    where
        concat_ws('-', year, month, day)  between "2026-02-07" and "2026-02-13"
        and param ['broad_source'] in ('hh_stg','hh_hxz_stg')
        and param ['order_origin'] = 3
),
broadcast as (
    --所有播单的od对
    select
        b.hxz_driver_id as driver_id,
        param ['orderid'] as order_id,
        param ['uuid'] as uuid,
        param ['driverid'] as driver_id_hh
    from
        kflower_marketplace.ods_log_kflower_orderbroadcast_hh a
        left join (
            select
                other_driver_id,
                hxz_driver_id
            from
                kflower_strategy.pk_hxz_driver_id
            where
                concat_ws('-', year, month, day)="2026-02-21"
        ) b on a.param ['driverid'] = b.other_driver_id
    where
        concat_ws('-', year, month, day) between "2026-02-07" and "2026-02-13"
        and param ['broad_source'] in ('hh_stg','hh_hxz_stg')
        -- and param ['order_origin'] = 3
),
handup_lose as (
    --所有举手且pk失败的od对
    select
        distinct
        a.order_id,
        a.driver_id,
        a.driver_id_3rd_part,
        a.round_num
    from
        handup a
        left join win b on a.order_id = b.order_id
        and a.driver_id = b.driver_id
        and a.round_num = b.round_num
    where b.driver_id is  null
),
broadcast_handup_lose as (
    select
        a.order_id,
        a.driver_id,
        a.uuid
    from
        broadcast a
        join handup_lose b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
)

select 
    exp_group,
    count(distinct a.order_id) `本轮总订单数`,
    count(distinct if( d.driver_id is null and c1.driver_id is not null, a.order_id, null)) `本轮o没分出去、未播单、本轮存在连边d举手PK失败的订单数`,
    count(distinct if( d.driver_id is null and c1.driver_id is not null, a.order_id, null))/count(distinct a.order_id) `本轮o没分出去、未播单、本轮存在连边d举手PK失败的订单占比`
from
    (select driver_id, exp_group from did_exp) e
    left join matrix a on e.driver_id = a.driver_id
    left join broadcast_handup_lose c1 on a.driver_id = c1.driver_id and a.uuid = c1.uuid
    left join broadcast d on a.order_id = d.order_id and a.uuid=d.uuid
group by 
    e.exp_group
