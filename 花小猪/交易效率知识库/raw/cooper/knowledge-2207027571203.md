# 鸿鹄端外流量调权-司机分流实验分析

# 一、背景及结论

**策略：**宏鹄司机同时连边端内订单和端外订单时，丢掉端外连边。

**收益链路：**宏鹄丢端外连边->端内DO连边数不变、端外DO连边数降->端内DO播单涨->端内举手涨->端内PK及获胜涨->端内应答涨->端内完单涨

with did_exp as 
(
	select  distinct b.hxz_driver_id as driver_id
            ,a.group_name as exp_group
    from    
    (
        select apollo_key,group_name from 
        apollo.pre_flow_data_etl
        where   concat_ws('-', year, month, day) between "2026-03-03" and "2026-03-09"
        and   exp = 'hh_pbd_od_value_adjust_v1'
    ) a
    left join (
        select
            other_driver_id,
            hxz_driver_id
        from
            kflower_strategy.pk_hxz_driver_id
        where
            concat_ws('-', year, month, day)="2026-03-08"
    ) b on a.apollo_key = b.other_driver_id
),

matrix as 
(
    --所有过滤后连边的od对
    select
        cast(param ['orderid'] AS bigint) AS order_id,
        b.hxz_driver_id as driver_id,
        param ['uuid'] as uuid,
        param ['DispatchType'] as dispatch_type,
        case 
            when param['company_id']='1400' then '自营'
            when param['company_id']<>'1400' then '宏鹄'
        end as type,
        param ['broad_source'] as broad_source,
        ceil(
            dense_rank () over(
                order by
                    param ['timestamp']
            ) / 2
        ) as round_num
    from
    (
    select * from
        kflower_marketplace.ods_log_kflower_stgmatrix_hh 
    where
        concat_ws('-', year, month, day) between "2026-03-03" and "2026-03-09"
        and param ['broad_source'] in ('hh_stg','hh_hxz_stg') --筛选hh和zy的所有od对
        and param ['order_origin'] = 3 --筛选订单来源为端内
    ) a 
    left join (
        select
            other_driver_id,
            hxz_driver_id
        from
            kflower_strategy.pk_hxz_driver_id
        where
            concat_ws('-', year, month, day)="2026-03-08"
    ) b on a.param ['driverid'] = b.other_driver_id
),

broadcast as 
(
    --所有播单的od对

        where
        concat_ws('-', year, month, day) between "2026-03-03" and "2026-03-09"

    )a

        select
            other_driver_id,

),

fin_ans_cancel as

        order_id,

),

handup as
(
    --所有举手的od对
    select
        shiftleft(
            shiftright(cast(a.orderid as bigint), 46) & 262143,
            32
        ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
        driverid as driver_id,
        driver_id_3rd_part,
        round_num
    from
        kflower_ods.ods_log_kflower_g_order_strivetender a
    where
        concat_ws('-', year, month, day) between "2026-03-03" and "2026-03-09"
),

win as
(
    -- 所有胜出的od对
    select
        shiftleft(
            shiftright(cast(orderid as bigint), 46) & 262143,
            32
        ) +(cast(orderid as bigint) & 4294967295) as order_id,
        driverid as driver_id,
        round_num as round_num
    from
        kflower_ods.ods_log_kflower_g_order_dusepk_auction a
    where
        concat_ws('-', year, month, day) between "2026-03-03" and "2026-03-09"
        and channelid = 3
        and pktype = 15
        and action = 'auction_time'
        and process_order_status_ok = 'true'
        and process_winner_ok = 'true'
),

strive_count as (
    -- 预先算出每个订单的司机竞争数
    select 
        order_id, 
        count(distinct driver_id) as driver_cnt
    from handup
    group by order_id
),

handup_win as 
(
    --所有举手且pk成功的od对
    select
        distinct
        a.order_id,
        a.driver_id,
        a.driver_id_3rd_part,
        a.round_num,
        case when c.driver_cnt = 1 then 1 else 0 end as only_strive_win
    from
        handup a
        join win b on a.order_id = b.order_id
        and a.driver_id = b.driver_id
        and a.round_num = b.round_num
        left join strive_count c on a.order_id = c.order_id
),

handup_lose as 
(
    --所有举手且pk失败的od对
    select
        distinct
        a.order_id,
        a.driver_id,
        a.driver_id_3rd_part,
        a.round_num
    from
        handup a
        left join win b on a.order_id = b.order_id
        and a.driver_id = b.driver_id
        and a.round_num = b.round_num
    where b.driver_id is null
),

handup_lose_winner_addprocase as 
(
    --所有举手且pk失败的od对
    select
        distinct
        a.order_id,
        b.driver_id,
        b.driver_id_3rd_part
    from
        handup_lose a
        left join handup_win b on a.order_id = b.order_id
        left join matrix c on c.order_id = b.order_id
        and c.driver_id = b.driver_id
),

broadcast_handup_win as 
(
    --播单表join举手且pk成功表，举手且成功的od对有值，其余为null
    select
        a.order_id,
        a.driver_id,
        a.uuid,
        b.only_strive_win
    from
        broadcast a
        join handup_win b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
),

broadcast_handup_lose as 
(
    --播单表join举手且pk失败表，举手且成功的od对有值，其余为null
    select
        a.order_id,
        a.driver_id,
        a.uuid
    from
        broadcast a
        join handup_lose b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
)
