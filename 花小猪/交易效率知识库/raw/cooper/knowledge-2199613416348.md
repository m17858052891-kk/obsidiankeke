```PlainText
                select
                    gp,
                    count(
                        distinct case
                            when is_td_call = 1 then passenger_id
                            else null
                        end
                    ) as pid_cnt,
                    sum(gmv_amt) as gmv,
                    sum(
                        subsidy
                    ) as cost,
                    sum(gmv_amt) / count(
                        distinct case
                            when is_td_call = 1 then passenger_id
                            else null
                        end
                    ) as gmv_per_pid,
                    sum(gmv_amt - actual_amt) / count(
                        distinct case
                            when is_td_call = 1 then passenger_id
                            else null
                        end
                    ) as cost_per_pid,
                    sum(subsidy) / count(
                        distinct case
                            when is_td_call = 1 then passenger_id
                            else null
                        end
                    ) as subsidy_per_pid
                from
                    (
                        SELECT
                            base.dt as dt,
                            case
                                when hash_key % 100 >= 95
                                and hash_key % 100 < 100 then 'control_group'
                                when hash_key % 100 < 90
                                then 'treatment_group'
                                else "treatment_group_2"
                            end as gp,
                            passenger_id,
                            is_td_finish,
                            is_td_answer,
                            is_td_call,
                            source_type,
                            start_city_id,
                            gmv_amt,
                            base.order_id as order_id,
                            actual_amt,
                            case
                                when willwait_deduction is null then 0
                                else willwait_deduction
                            end as subsidy
                        from
                            (
                                select
                                    dt,
                                    passenger_id,
                                    cast(order_id as bigint) as order_id,
                                    start_city_id,
                                    is_td_finish,
                                    is_td_answer,
                                    is_td_call,
                                    source_type,
                                    apollo_bucket("kflower_willing_wait_ab", passenger_id) AS hash_key
                                from
                                    kflower_dw.dwd_kf_trd_order_base_di
                                where
                                    dt between '2020-12-08' and '2020-12-10'--and is_td_call = 1
                                    and call_time is not null
                                    and call_time <> '0000-00-00 00:00:00'
                                    and call_time >= '2020-12-08 11:00:00' 
                                    and start_city_id in (5,9,13,15,26,1,11,3,17,173,21,24,33,132,81,145,8,157,87,50,283,35,133,118,284,57,282,36,19,20,6,32,86,85,34,47,38,74,12,22,153,88,89,29,46,160,158,329,62,83,45,105,90,64,143,53,84,254,154,39,41,134,92,226,135,142,25,102,138,253,49,159,306,225,224,167,79,37,170,42,121,140,280,147,255,93,56,144,156,91,97,188,101,297,117,95,94,155,285,54,174,40,256,198,161,136,281,172,59,189,114,243,139,98,100,123,148,107,193,127,60,146,72,168,80,67,61,257,287,244,162,222,294,232,261,228,151,108,235,279,262,55,137,112,259,245,251,250,113,182,298,305,196,266,177,166,233,302,227,289,110,258,264,263,124,265,319,241,231,115,300,109,307,304,303,246,199,308,197,163,301,126,141,76,131,318,296,299,249,229,220,194,171,99,234,267,187,185,150,149,350,180,190,270,191)
                            ) base
                            left join(
                                select
                                    distinct cast(order_id as bigint) as report_order_id,
                                    dt,
                                    willwait_deduction
                                from
                                    kflower_dw.dwm_kf_fin_order_report_di
                                where
                                    dt between '2020-12-08' and '2020-12-10'
                                    and willwait_deduction > 0

                                    dt between '2020-12-08' and '2020-12-10'
                                    and call_time is not null
                                    and call_time <> '0000-00-00 00:00:00'

                            and base.dt = pay.dt

                    gp
```
