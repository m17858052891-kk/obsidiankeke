### 召回到分单

核心司机hold成功率（命中hold被过滤后，在该hold时间片内是否有过价值高于阈值的匹配）

区分自营和鸿鹄，实验组与对照组核心司机作对比，【better have\*：实验组司机与大盘情况作对比】

可给出hold系数的参考，若成功率偏低下调计算时的阈值

```PlainText
select
    exp_group,
    round(
        count(
            case
                when success_cnt > 0 then 1
                else null
            end
        ) / count(1),
        2
    ) `hold成功率`
from
    (
        select
            exp_group,
            t.driver_id,
            driver_start_listen_time,
            driver_end_hold_time,
            sum(
                distinct case
                    when order_gmv >= gmv_limit
                    or driver_wait_time >= time_limit then 1
                    else 0
                end
            ) success_cnt
        from
            (
                select
                    driver_id,
                    driver_wait_time,
                    time_limit,
                    timestmp - driver_wait_time as driver_start_listen_time,
                    timestmp - driver_wait_time + time_limit as driver_end_hold_time,
                    timestmp,
                    order_id,
                    gmv_limit,
                    order_gmv
                from
                    (
                        select
                            cast(param ['area'] as bigint) city_id,
                            cast(param ['driver_wait_time'] as bigint) driver_wait_time,
                            cast(param ['driverid'] as bigint) driver_id,
                            cast(param ['gmv_limit'] as double) gmv_limit,
                            cast(param ['order_gmv'] as double) order_gmv,
                            cast(param ['orderid'] as bigint) order_id,
                            cast(param ['time_limit'] as bigint) time_limit,
                            unix_timestamp(param ['timestamp']) timestmp
                        from
                            kflower_marketplace.ods_log_kflower_gmv_hold
                        where
                            concat(year, month, day) = '20220817'
                            and param ['timestamp'] >= '2022-08-17 15:13:00'
                    ) t
            ) t
            join (
                select
                    driver_id,
                    exp_group
                from
                    kflower_strategy_dev.zy_core_driver
            ) t2 on t.driver_id = t2.driver_id
        group by
            exp_group,
            t.driver_id,
            driver_start_listen_time,
            driver_end_hold_time
    ) t
group by
    exp_group;
```

```PlainText
select
    round(
        count(
            case
                when broadcast_cnt > 0 then 1
                else null
            end

        2
    ) `播单成功率`
from

        from

                from

                        select

                            cast(param ['orderid'] as bigint) order_id
                        from
                            kflower_ods.ods_log_kflower_stgmatrix
                        where
                            concat_ws('-', year, month, day) = '2022-08-16'
                    ) t1
                    left join (
                        select
                            param ['uuid'] as round_id,
                            cast(param ['driverid'] as bigint) driver_id,
                            cast(param ['lowoid'] as bigint) order_id
                        from
                            kflower_ods.ods_log_kflower_orderbroadcast
                        where
                            concat_ws('-', year, month, day) = '2022-08-16'
                    ) t2 on t1.round_id = t2.round_id
                    and t1.driver_id = t2.driver_id
                    and t1.order_id = t2.order_id
            ) t
            join (
                select
                    driver_id
                from
                    kflower_strategy_dev.hangzhou_compliance_driver

        group by
            compliance_driver.driver_id,

    ) t;
```

核心司机both举手PK胜率（订单维度）

```PlainText
--待区分自营司机和鸿鹄司机
with winner_info as (
    select
        concat_ws('-', year, month, day) as date_,
        int(cityid) as cityid,
        int(channelid) as channelid,
        orderid,
        driverid,
        round_token,
        min(pkvalue) as win_pkvalue,
        min(unix_timestamp(`timestamp`)) as win_time,
        min(int(cpid)) as cpid,
        count(cpid) as cp_num
    from
        kflower_ods.ods_log_kflower_g_order_dusepk_auction
    where
        concat_ws('-', year, month, day) between '2022-08-16'
        and '2022-08-16'
        and hour(`timestamp`) >= 20
        and action = 'auction_time'
        and (
            pktype = 15
            or pktype = 17
        )
    group by
        concat_ws('-', year, month, day),
        cityid,
        channelid,
        orderid,
        driverid,
        round_token
    having
        cp_num = 1
),
hands_up as (
    select
        concat_ws('-', year, month, day) as date_,
        int(cityid) as cityid,
        int(channelid) as channelid,
        orderid,
        driverid,
        round_token,
        int(multi) as pkvalue,
        unix_timestamp(`timestamp`) as tend_time,
        int(cpid) as cpid,
        case
            when iscontinue = 'true' then 1
            else 0
        end as is_continue
    from
        kflower_ods.ods_log_kflower_g_order_strivetender
    where
        concat_ws('-', year, month, day) between '2022-08-16'
        and '2022-08-16'
        and hour(`timestamp`) >= 20
        and (
            pktype = 15
            or pktype = 17
        )
),
hands_up_info as (
    select
        hands_up.date_ as date_,
        hands_up.cityid as cityid,
        hands_up.channelid as channelid,
        hands_up.orderid as orderid,
        hands_up.round_token as round_token,
        hands_up.cpid as hands_up_cp,
        hands_up.driverid as driverid,
        hands_up.pkvalue as hands_up_pkvalue,
        winner_info.cpid as win_cp,
        winner_info.driverid as win_driverid,
        winner_info.win_time as win_time,
        winner_info.win_pkvalue as win_pkvalue,
        case
            when is_continue = 1
            and tend_time <= win_time then 1
            else 0
        end as is_valid
    from
        hands_up
        inner join winner_info on hands_up.date_ = winner_info.date_
        and hands_up.cityid = winner_info.cityid
        and hands_up.channelid = winner_info.channelid
        and hands_up.orderid = winner_info.orderid
        and hands_up.round_token = winner_info.round_token
),
all_hands_up_info as (
    select
        a.date_ as date_,
        a.cityid as cityid,
        a.channelid as channelid,
        a.orderid as orderid,
        a.round_token as round_token,
        a.hands_up_cp as hands_up_cp,
        a.driverid as driverid,
        a.hands_up_pkvalue as hands_up_pkvalue,
        a.win_cp as win_cp,
        a.win_driverid as win_driverid,
        a.win_time as win_time,
        a.win_pkvalue as win_pkvalue,
        a.is_valid as is_valid,
        b.valid_cp_num as valid_cp_num,
        b.all_cp_num as all_cp_num
    from
        hands_up_info a
        inner join (
            select
                date_,
                cityid,
                channelid,
                orderid,
                round_token,
                count(
                    distinct case
                        when is_valid = 1 then hands_up_cp
                        else null
                    end
                ) as valid_cp_num,
                count(
                    distinct hands_up_cp
                ) as all_cp_num
            from
                hands_up_info
            group by
                date_,
                cityid,
                channelid,
                orderid,
                round_token
        ) b on a.date_ = b.date_
        and a.cityid = b.cityid
        and a.channelid = b.channelid
        and a.orderid = b.orderid
        and a.round_token = b.round_token
),
is_pked_info as (
    select
        a.date_ as date_,
        a.cityid as cityid,
        a.channelid as channelid,
        a.orderid as orderid,
        shiftleft(
            shiftright(cast(a.orderid as bigint), 46) & 262143,
            32
        ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
        a.hands_up_cp as hands_up_cp,
        a.driverid as hands_up_driverid,
        min(a.valid_cp_num) as valid_cp_num,
        min(a.all_cp_num) as all_cp_num,
        min(a.hands_up_pkvalue) as hands_up_pkvalue,
        min(a.win_pkvalue) as win_pkvalue,
        max(a.win_driverid) as win_driverid,
        max(a.win_cp) as win_cp,
        max(a.is_valid) as is_valid,
        max(
            case
                when a.is_valid = 1
                and b.is_valid = 1
                and a.hands_up_cp = b.hands_up_cp then 1
                when a.is_valid = 1
                and b.is_valid = 1
                and a.hands_up_cp != b.hands_up_cp then 2
                else 0
            end
        ) as is_pked
    from
        all_hands_up_info a
        inner join all_hands_up_info b on a.date_ = b.date_
        and a.cityid = b.cityid
        and a.channelid = b.channelid
        and a.orderid = b.orderid
        and a.round_token = b.round_token
    group by
        a.date_,
        a.cityid,
        a.channelid,
        a.orderid,
        a.round_token,
        a.driverid,
        a.hands_up_cp
)
select
    count(
        case
            when is_win = 1
            and is_both = 1 then 1
            else null
        end
    ) / count(
        case
            when is_both = 1 then 1
            else null
        end
    ) `both举手胜率`
from
    (
        select
            date_,
            cityid,
            channelid,
            (
                case
                    when channelid = 1 then 'dd'
                    when channelid = 2 then 'pbd'
                    else 'hxz'
                end
            ) as source_platform,
            hands_up_cp as appid,
            order_id,
            hands_up_driverid as driver_id,
            case
                when is_valid = 1 then 1
                else 0
            end as is_valid,
            case
                when is_valid = 1
                and is_pked = 2 then 1
                else 0
            end as is_both,
            case
                when win_cp = hands_up_cp then 1
                else 0
            end as is_win
        from
            is_pked_info
        group by
            date_,
            cityid,
            channelid,
            (
                case
                    when channelid = 1 then 'didi'
                    when channelid = 2 then 'pbd'
                    else 'hxz'
                end
            ),
            hands_up_cp,
            order_id,
            hands_up_driverid,
            case
                when is_valid = 1 then 1
                else 0
            end,
            case
                when is_valid = 1
                and is_pked = 2 then 1
                else 0
            end,
            case
                when win_cp = hands_up_cp then 1
                else 0
            end
    ) pk
    join (
        select
            driver_id
        from
            kflower_strategy_dev.hangzhou_compliance_driver
    ) comp_driver on pk.driver_id = comp_driver.driver_id;
```

```PlainText
select
    gmv.exp_group,
    round(total_gmv / total_tsh * 60.0, 2) iph
from
    (
        select
            exp_group,
            sum(gmv) total_gmv
        from
            (
                select
                    driver_id,
                    sum(gmv_amt) as gmv
                from
                    kflower_dw.dwd_kf_trd_unity_order_di
                where
                    dt between '2022-09-01'
                    and '2022-09-07'
                    and is_td_finish = 1
                group by
                    driver_id
            ) o
            join (
                select
                    driver_id_hh as driver_id,
                    exp_group
                from
                    kflower_strategy_dev.hh_core_driver_hh_driver_id
            ) core_driver on o.driver_id = core_driver.driver_id
        group by
            exp_group
    ) gmv
    join (
        select
            exp_group,
            sum(tsh) total_tsh
        from
            (
                select
                    driver_id,
                    count(online_time) tsh
                from
                    kflower_dw.dwd_hh_dri_online_di
                where
                    dt between '2022-09-01'
                    and '2022-09-07'
                group by
                    driver_id
            ) a
            join (
                select
                    driver_id,
                    exp_group
                from
                    kflower_strategy_dev.hh_core_driver
            ) b on a.driver_id = b.driver_id
        group by
            exp_group
    ) tsh on gmv.exp_group = tsh.exp_group
order by
    exp_group;
```
