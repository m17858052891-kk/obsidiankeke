drop table if exists temp_pickup_distance_driver_behavior_order_api_id_v2
;

create table if not exists temp_pickup_distance_driver_behavior_order_api_id_v2
as
with
pas_tag as 
(
    select  distinct apollo_key as pid
            ,case   when group_name = 'control_group' then 'control_group'
                    else 'treatment_group'
             end as exp_group
    from    apollo.pre_flow_data_etl
    where   concat_ws('-', year, month, day) between '2025-06-17'
      and   '2025-06-23'
      and   exp = 'platform_pickupdistance_cons'
      and   group_name in ('control_group', 'treatment_group')
)
,finish_order as 
(
    select  a.order_id
            ,a.passenger_id
            ,a.driver_id
            ,is_td_finish
            ,a.product_id
            ,a.product_name
            ,a.product_name_unit
            ,a.product_type
            ,a.platform_name
            ,is_td_answer
            ,is_td_answer_without_reassign
            ,is_grab_before_cannel
            ,is_grab_after_pas_cancel_without_reassign
            ,is_grab_after_dri_cancel
            ,is_grab_after_dri_cancel_without_reassign
            ,c.est_arrive_dis
            ,c.est_arrive_dur
            ,a.answer_dur
            ,a.gmv_amt
            ,d.is_serial_assign
    from    (
        select  distinct dt
                ,source_id
                ,order_id
                ,driver_id
                ,is_hxz
                ,passenger_id
                ,product_id
                ,product_name
                ,product_name_unit
                ,product_type
                ,platform_name
                ,is_grab_before_cannel
                ,is_td_finish
                ,est_arrive_dur
                ,est_arrive_dis
                ,call_car_level_id
                ,combo_type
                ,level_type
                ,answer_dur
                ,gmv_amt
        from    kflower_dw.dwd_kf_trd_unity_order_di
        where   dt between '2025-06-17'
          and   '2025-06-23'
          and   is_hxz = 1
          and   start_city_id in (13, 34, 19, 6, 3, 24, 38, 157, 14, 153, 5, 9, 22, 35, 21, 29, 17, 12, 125, 2, 74, 15)
    ) a
    left join (
        select  distinct dt
                ,source_id
                ,order_id
                ,is_td_answer
                ,is_td_answer_without_reassign
                ,is_grab_after_pas_cancel_without_reassign
                ,is_grab_after_dri_cancel
                ,is_grab_after_dri_cancel_without_reassign
        from    kflower_dw.dim_kf_trd_unity_order_label_di
        where   dt between '2025-06-17'
          and   '2025-06-23'
    ) b
    on      a.dt = b.dt
      and   a.source_id = b.source_id
      and   a.order_id = b.order_id
    left join (
        select  distinct order_id
                ,product_id
                ,call_car_level_id
                ,combo_type
                ,level_type
                ,driver_start_eda as est_arrive_dis
                ,driver_start_eta as est_arrive_dur
        from    kflower_dw.dwm_kf_trd_unity_trade_funnel_order_anycar_di
        where   dt between '2025-06-17'
          and   '2025-06-23'
          and   start_city_id in (13, 34, 19, 6, 3, 24, 38, 157, 14, 153, 5, 9, 22, 35, 21, 29, 17, 12, 125, 2, 74, 15)
    ) c
    on      a.order_id = c.order_id
      and   a.product_id = c.product_id
      and   a.call_car_level_id = c.call_car_level_id
      and   coalesce(a.combo_type, 0) = coalesce(c.combo_type, 0)
      and   coalesce(a.level_type, 0) = coalesce(c.level_type, 0)
    left join (
        select  distinct dt
                ,order_id
                ,is_serial_assign
        from    kflower_dw.dwd_kf_trd_order_base_di
        where   dt between '2025-06-17'
          and   '2025-06-23'
          and   product_1level <> 4
    ) d
    on      a.order_id = d.order_id
      and   a.dt = d.dt
)
select  order_id
        ,d.passenger_id
        ,exp_group
        ,is_td_finish
        ,product_id
        ,product_name
        ,product_name_unit
        ,product_type
        ,platform_name
        ,is_td_answer
        ,is_td_answer_without_reassign
        ,is_grab_before_cannel
        ,is_grab_after_pas_cancel_without_reassign
        ,is_grab_after_dri_cancel
        ,is_grab_after_dri_cancel_without_reassign
        ,est_arrive_dis
        ,est_arrive_dur
        ,answer_dur
        ,gmv_amt
        ,is_serial_assign
from    pas_tag a
left join finish_order d
on      a.pid = d.passenger_id
;

select  *
from    temp_pickup_distance_driver_behavior_order_api_id_v2
limit   100
;

-- API商家分析
select  B_UNIT
        ,exp_group
        ,count(case when is_td_answer = 1 then 1 end)
        ,count(case when is_td_finish = 1 then 1 end)
        ,sum(case when is_td_finish = 1 then gmv_amt end)
        ,count(*)
from    (
    select  *
            ,case   when platform_name = 'API接入' then product_name_unit
                    else '非API'
             end as B_UNIT
    from    temp_pickup_distance_driver_behavior_order_api_id_v2
)
group by B_UNIT, exp_group
;
