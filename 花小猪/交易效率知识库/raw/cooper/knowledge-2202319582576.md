with score_info as (
    select distinct trace_id, real_score, ceil(real_score) as show_score
    from (
        select 
            param ['trace_id'] as trace_id,
            get_json_object(param['predict_value'], '$.eta[0]') as real_score,
            row_number() over(
                partition by param ['trace_id'], param ['passenger_id']
                order by unix_timestamp(param ['timestamp']) desc
            ) as timerank
        from kflower_ods.ods_log_ether_hxz_app_eta_model
        where concat_ws('-', year, month, day) between '${begin}' and '${end}'
            --and param["city"] in (${city_id})
    ) t0
    where timerank = 1
)
, order_info as (
    select 
        bubble_trace_id, 
        start_city_id, 
        cancel_type, 
        t0.order_id, 
        call_cnt_1d_1018, 
        answer_dur_1d_1006, 
        bf_ansr_cl_dur, 
        level_type,
        begin_time,
        begin_time1,
        end_time 
    from (
        select 
            distinct 
            order_id, 
            bubble_trace_id, 
            start_city_id,
            level_type,
            unix_timestamp(append_call_time) as begin_time
        from 
            kflower_dw.dwd_kf_trd_order_anycar_di
        where 
            dt between '${begin}' and '${end}'
            and (level_type = 200 or enter_express_box_status in (2, 3)) -- 只看特快盒子的呼叫
            --and start_city_id in (${city_id})
    )t0
    left join (
        select  
            order_id, 
            cancel_type, 
            call_cnt_1d_1018, 
            answer_dur_1d_1006, 
            bf_ansr_cl_dur,
            unix_timestamp(call_time) as begin_time1,
            case when cancel_type=1 then unix_timestamp(cancel_time) else unix_timestamp(answer_time) end as end_time

        from    kflower_dw.dwm_kf_trd_order_index_di
        where   dt between '${begin}' and '${end}'
            --and start_city_id in (${city_id})
    )t1 on t0.order_id = t1.order_id
) 

--计算赔付率
select 
    time_segment
    --, start_city_id
    , badcase_num
    , bubble_num
    , call_num
    , badcase_num / call_num as rate
from (
    select 
        time_segment
        --, start_city_id
        , count(distinct trace_id) as bubble_num
        , count(distinct case when call_cnt_1d_1018 > 0 then order_id else null end) as call_num
        , count(distinct if (call_cnt_1d_1018 > 0 and is_badcase = 1, order_id, null)) as badcase_num
    from (
        select order_id, trace_id, call_cnt_1d_1018, start_city_id, time_segment, 
               case when time_segment = 'l1' and check_time >  15 then 1
                    when time_segment = 'l2' and check_time > 30 then 1
                    when time_segment = 'l3' and check_time > 60 then 1
                    when time_segment = 'l4' and check_time > 120 then 1
                    when time_segment = 'l5' and check_time > 180 then 1
                else 0
                end as is_badcase
        from (
            select trace_id, real_score, 
            case when real_score + ${buffer}<= 15 then 'l1'
                    when real_score + ${buffer}> 15 and real_score + ${buffer}<= 25 then 'l2'
                    when real_score + ${buffer}> 25 and real_score + ${buffer}<= 50 then 'l3'
                    when real_score + ${buffer}> 50 and real_score + ${buffer}<= 90 then 'l4'
                    when real_score + ${buffer}> 90 and real_score + ${buffer}<= 120 then 'l5'
                    when real_score + ${buffer}> 120 then 'l6'
            end as time_segment
            from(
                select trace_id, cast(real_score as double) * 60 as real_score
                from score_info
            )t0       
        )t0
        left join(
            select bubble_trace_id, cancel_type, order_id, call_cnt_1d_1018, answer_dur_1d_1006, bf_ansr_cl_dur, start_city_id
                , end_time - greatest(begin_time, begin_time1) as check_time
                --, least(answer_dur_1d_1006, case when cancel_type = 1 then bf_ansr_cl_dur else answer_dur_1d_1006 end) as check_time
            from order_info
        )t1
        on t0.trace_id = t1.bubble_trace_id
    ) t0
    group by time_segment--, start_city_id
)t0;

select

    `完单gmv金额`,
    `特快ECR`,
    `竞争发单量`,
    `去重冒泡数`,
    `最大去重冒泡数`,
    1.2*1.6*`最大去重冒泡数`*0.8 as `城市阈值0`
    --1.5*2.2672*`C补率`*`完单gmv金额`/(`特快ECR`*2*0.124*0.676) as `城市阈值1`,
    --1.5*1.6614*`C补率`*`完单gmv金额`/(`特快ECR`*1.94*0.124*0.676) as `城市阈值2`
from 
(
    select 
        city_id,
        0.03 as `C补率`,
        sum(fast_lane_gmv_amt)/count(distinct dt) as `完单gmv金额`
    from 
        kflower_app.dm_hh_trd_fast_lane_scale_di

        dt between '${begin_date}' and '${end_date}'
        --and city_id in (13,15,16,25,37,40,42,50,83,90,114,132,136,160,193,245,253,283,300,304,34,162,53,235,262,19,167,41,60,8,34,85,26,58,38,19,17,6,9,12,35,1,86,81,153,24,21,22,62,32,254,5,44,134,29,74,36,23,11,138)

(

        sum(bubble_need_cnt) as bubble_need_cnt

        kflower_app.dm_hh_trd_fast_lane_bub_di
    where 
        dt between '${begin_date}' and '${end_date}'

                ,sum(gmv_amt) as gmv_amt	--当天核销券金额,超时赔发的券 被核销的金额

        union all 
        select  city_id
                ,call_start_dest_dis_range as start_dest_dis_range
                ,is_peak
                ,timeout_pay_amt
                ,0 as pid_bubble_cnt
                ,0 as lane_bubble_cnt	--特快盒子总冒泡数
                ,0 as timeout_pay_bubble_cnt	--特快盒子有超时赔的冒泡数
                ,0 as timeout_pay_total_time	--超时赔超时总时间，秒

                ,0 as timeout_pay_order_cnt	--特快盒子有超时赔的发单数

                ,0 as coupon_send_amt	--超时未应答发券成功的券金额
                ,0 as coupon_use_amt	--当天核销券金额,超时赔发的券 被核销的金额
                ,0 as gmv_amt	--当天核销券金额,超时赔发的券 被核销的金额

                ,0 as answer1084_order_cnt	--特快盒子总应答数含改派

    ) as re2

    where   re1.city_id is not null
