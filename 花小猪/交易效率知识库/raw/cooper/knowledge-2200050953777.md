```SQL
    if(a.exp_group = b.exp_group,'--',
        if(
            abs(
                (a.finish_ord_cnt_hh/a.call_ord_cnt_hh - b.finish_ord_cnt_hh/b.call_ord_cnt_hh)/sqrt(a.finish_ord_cnt_hh/a.call_ord_cnt_hh * (1-a.finish_ord_cnt_hh/a.call_ord_cnt_hh)/a.samples + b.finish_ord_cnt_hh/b.call_ord_cnt_hh * (1-b.finish_ord_cnt_hh/b.call_ord_cnt_hh)/b.samples)
            ) > 1.96,
            '显著',
            '不显著'
        )
    ) as `鸿鹄CR增幅显著性`,
    round(a.finish_ord_cnt_dd/a.call_ord_cnt_dd, 8) as `滴滴CR`,
    if(a.exp_group = b.exp_group,'--',
        round((a.finish_ord_cnt_dd/a.call_ord_cnt_dd - b.finish_ord_cnt_dd/b.call_ord_cnt_dd),8) 
        ) as `滴滴CR增幅-pp`,
    if(a.exp_group = b.exp_group,'--',
        if(
            abs(
                (a.finish_ord_cnt_dd/a.call_ord_cnt_dd - b.finish_ord_cnt_dd/b.call_ord_cnt_dd)/sqrt(a.finish_ord_cnt_dd/a.call_ord_cnt_dd * (1-a.finish_ord_cnt_dd/a.call_ord_cnt_dd)/a.samples + b.finish_ord_cnt_dd/b.call_ord_cnt_dd * (1-b.finish_ord_cnt_dd/b.call_ord_cnt_dd)/b.samples)
            ) > 1.96,
            '显著',
            '不显著'
        )
    ) as `滴滴CR增幅显著性`
from
    group_base a
left join 
    group_control b 
on a.experiment = b.experiment;
```
