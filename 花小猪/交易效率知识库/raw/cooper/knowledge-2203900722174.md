# 背景

聚合平台下，商家交易系统自主完成各自分单，即每个商家系统（自营+宏鹄+API单商家）都有可能举手出一个司机

此时对于平台而言，存在无效竞争导致资源浪费的情况

场景解释，假设只有A商家系统和B商家系统，各自有两个司机，计作da1，da2，db1，db2；共四个订单，计作o1，o2，o3，o4

在A商家系统中，最终匹配为：da1-o1 履约0.8，da2-o2  履约0.7

在B商家系统中，最终匹配为：db1-o1 履约0.75，db2-o2 履约0.85

在聚合PK系统中，最终匹配为：da1-o1 履约0.8，db2-o2 履约0.85

对平台而言：da2、db1两个司机本轮轮空，o3，o4两个订单轮空

对B商家而言，高效竞争时，应当匹配：db1-o2 履约0.75，db2-o1 履约0.85，此时两个订单都能获胜

因此在商家系统无法得知其他商家系统的信息时，对平台和商家效率都有折损。可以通过一定的策略优化这种情况，最基础的策略聚焦于这些没分出去的订单。

with handup as (
    --所有举手的od对
    select
        shiftleft(
            shiftright(cast(a.orderid as bigint), 46) & 262143,
            32
        ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
        driverid driver_id,
        driver_id_3rd_part,
        round_num
    from
        kflower_ods.ods_log_kflower_g_order_strivetender a
    where
        concat_ws('-', year, month, day) = '2025-02-13'
        -- and channelid = 3
),
win as (
    -- 所有胜出的od对
    select 
        shiftleft(
            shiftright(cast(orderid as bigint), 46) & 262143,
            32
        ) +(cast(orderid as bigint) & 4294967295) as order_id,
        driverid driver_id,
        round_num
    from 
        kflower_marketplace.kflower_allmatch_pk_round_info
    where
        dt = '2025-02-13'
        and driverid = win_driverid
        and channelid = 3
        and int(is_valid) = 1
),
matrix as (
    --所有过滤后连边的od对
    select
        cast(param ['orderid'] AS bigint) AS order_id,
        b.hxz_driver_id as driver_id,
        param ['uuid'] as uuid,
        ceil(
                dense_rank () over(
                    order by
                        param ['timestamp']
                ) / 2
            ) as round_num
    from
        kflower_marketplace.ods_log_kflower_stgmatrix_hh a
        left join (
            select
                other_driver_id,
                hxz_driver_id
            from
                kflower_strategy.pk_hxz_driver_id
            where
                concat_ws('-', year, month, day) = '2025-02-25'
        ) b on a.param ['driverid'] = b.other_driver_id
    where
        concat_ws('-', year, month, day) = '2025-02-13'
        and param ['broad_source'] in ('hh_stg','hh_hxz_stg')
        and param ['order_origin'] = 3
),
broadcast as (
    --所有播单的od对
    select
        b.hxz_driver_id as driver_id,
        param ['orderid'] as order_id,
        param ['uuid'] as uuid,
        param ['driverid'] as driver_id_hh
    from
        kflower_marketplace.ods_log_kflower_orderbroadcast_hh a
        left join (
            select
                other_driver_id,
                hxz_driver_id
            from
                kflower_strategy.pk_hxz_driver_id
            where
                concat_ws('-', year, month, day) = '2025-02-25'
        ) b on a.param ['driverid'] = b.other_driver_id
    where
        concat_ws('-', year, month, day) = '2025-02-13'
        and param ['broad_source'] in ('hh_stg','hh_hxz_stg')
        -- and param ['order_origin'] = 3
),
handup_win as (
    --所有举手且pk成功的od对
    select
        distinct
        a.order_id,
        a.driver_id,
        a.driver_id_3rd_part,
        a.round_num
    from
        handup a
        join win b on a.order_id = b.order_id
        and a.driver_id = b.driver_id
        and a.round_num = b.round_num
),
handup_lose as (
    --所有举手且pk失败的od对
    select
        distinct
        a.order_id,
        a.driver_id,
        a.driver_id_3rd_part,
        a.round_num
    from
        handup a
        left join win b on a.order_id = b.order_id
        and a.driver_id = b.driver_id
        and a.round_num = b.round_num
    where b.driver_id is  null
),
broadcast_handup_win as (
    select
        a.order_id,
        a.driver_id,
        a.uuid
    from
        broadcast a
        join handup_win b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
),
broadcast_handup_lose as (
    select
        a.order_id,
        a.driver_id,
        a.uuid
    from
        broadcast a
        join handup_lose b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
),
broadcast_handup as (
    select
        a.order_id,
        a.driver_id,
        a.uuid
    from
        broadcast a
        join handup b on a.order_id = b.order_id
        and a.driver_id_hh = b.driver_id_3rd_part
)
select 
    count(distinct a.driver_id) `本轮总司机数`,
    count(distinct if( b.driver_id is not null and b.round_num - a.round_num = 1, a.driver_id, null)) `下一轮匹配到相同o的司机数`,
    count(distinct if( b.driver_id is not null and b.round_num - a.round_num > 0, a.driver_id, null)) `未来匹配到相同o的司机数`,
    count(distinct if( c1.driver_id is not null, a.driver_id, null)) `本轮举手PK失败的司机数`,
    count(distinct if( c2.driver_id is not null, a.driver_id, null)) `本轮举手PK成功的司机数`,
    count(distinct if( e.driver_id is null, a.driver_id, null)) `本轮没分出去的司机数`,
    count(distinct if( d.driver_id is null, a.driver_id, null)) `本轮连边订单没分出去的总司机数`,
    count(distinct if( d.driver_id is null and b.driver_id is not null and b.round_num - a.round_num = 1, a.driver_id, null)) `本轮连边订单没分出去、下一轮匹配到相同o的司机数`,
    count(distinct if( d.driver_id is null and b.driver_id is not null and b.round_num - a.round_num > 0, a.driver_id, null)) `本轮连边订单没分出去、未来匹配到相同o的司机数`,
    count(distinct if( d.driver_id is null and c1.driver_id is not null, a.driver_id, null)) `本轮连边订单没分出去、本轮举手PK失败的司机数`,
    count(distinct if( d.driver_id is null and c2.driver_id is not null, a.driver_id, null)) `本轮连边订单没分出去、本轮举手PK失败的司机数`,
    count(distinct if( d.driver_id is null and e.driver_id is null, a.driver_id, null)) `本轮连边订单没分出去、本轮没分出去的司机数`,
    count(distinct a.order_id) `本轮总订单数`,
    count(distinct if( b.driver_id is not null and b.round_num - a.round_num = 1, a.order_id, null)) `下一轮匹配到相同o的订单数`,
    count(distinct if( b.driver_id is not null and b.round_num - a.round_num > 0, a.order_id, null)) `未来匹配到相同o的订单数`,
    count(distinct if( c1.driver_id is not null, a.order_id, null)) `本轮存在连边d举手PK失败的订单数`,
    count(distinct if( e.driver_id is null, a.order_id, null)) `本轮存在连边d没分出去的订单数`,
    count(distinct if( d.driver_id is null, a.order_id, null)) `本轮没分出去的总订单数`,
    count(distinct if( d.driver_id is null and b.driver_id is not null and b.round_num - a.round_num = 1, a.order_id, null)) `本轮o没分出去、下一轮匹配到相同d的订单数`,
    count(distinct if( d.driver_id is null and b.driver_id is not null and b.round_num - a.round_num > 0, a.order_id, null)) `本轮o没分出去、未来匹配到相同d的订单数`,
    count(distinct if( d.driver_id is null and c1.driver_id is not null, a.order_id, null)) `本轮o没分出去、本轮存在d举手PK失败的订单数`,
    count(distinct if( d.driver_id is null and e.driver_id is null, a.order_id, null)) `本轮o没分出去、本轮存在连边d没分出去的订单数`
from
    (
        select 
            *
        from
            matrix
    ) a
    left join matrix b  on a.order_id = b.order_id and a.driver_id = b.driver_id
    left join broadcast_handup_lose c1 on a.driver_id = c1.driver_id and a.uuid = c1.uuid
    left join broadcast_handup_win c2 on a.driver_id = c2.driver_id and a.uuid = c2.uuid
    left join broadcast d
    on a.order_id = d.order_id and a.uuid=d.uuid
    left join broadcast e
    on a.driver_id = e.driver_id and a.uuid = e.uuid
    and a.uuid = e.uuid
