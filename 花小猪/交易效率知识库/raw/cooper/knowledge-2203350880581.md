with pas_tag as (
    SELECT
        apollo_key,
        group_name as group_name,
        concat_ws('-', year, month, day) as ymd,
        dims
    FROM
        apollo.pre_flow_data_etl
    WHERE
        concat_ws('-', year, month, day) between '${start_date}'
        and '${end_date}'
        AND exp = 'kf_upgrade_dialog_render_official'
),
call as (
    select
        passenger_id,
        count(
            distinct case
                when is_compete_call = 1 then a.order_id
            end
        ) as call_cnt
    from
        (
            select
                *
            from
                kflower_dw.dwd_kf_trd_unity_order_anycar_di
            where
                dt between '${start_date}'
                and '${end_date}'
                and passenger_id <> 0
                and start_city_id in (${city_list})
                and is_hxz_call = 1
                and to_date(call_time) = dt
                and product_id <> 26
        ) as a
        left join (
            select
                source_id,
                dt,
                order_id,
                estimate_id,
                product_id,
                call_car_level_id,
                is_compete_call,
                is_objective_call,
                is_platform_objective_call
            from
                kflower_dw.dim_kf_trd_unity_order_anycar_label_di
            where
                dt between '${start_date}'
                and '${end_date}'
        ) as c on a.order_id = c.order_id
        and a.estimate_id = c.estimate_id
        and a.product_id = c.product_id
        and a.call_car_level_id = c.call_car_level_id
        and a.dt = c.dt
    group by
        passenger_id
),
finish as (
    select
        passenger_id,
        count(
            distinct case
                when is_grab_before_cannel = 1 then a.order_id
            end
        ) as grab_before_cannel_cnt --应答前取消量
,
        avg(
            case
                when is_td_answer_without_reassign > 0 then wait_dur
            end
        ) as wait_dur --愿等时长
,
        count(
            distinct case
                when is_td_answer = 1 then a.order_id
            end
        ) as ans_cnt,
        count(
            distinct case
                when is_td_answer_without_reassign = 1 then a.order_id
            end
        ) as ans_cnt_without_reassign,
        count(
            distinct case
                when is_grab_after_pas_cancel_without_reassign = 1 then a.order_id
            end
        ) as grab_after_pas_cancel_cnt_without_reassign,
        count(
            distinct case
                when is_grab_after_dri_cancel = 1 then a.order_id
            end
        ) as grab_after_dri_cancel_cnt,
        count(
            distinct case
                when is_grab_after_dri_cancel_without_reassign = 1 then a.order_id
            end
        ) as grab_after_dri_cancel_cnt_without_reassign,
        count(
            distinct case
                when is_td_finish = 1 then a.order_id
            end
        ) as fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌'
                and platform_name = '宏鹄' and is_td_finish = 1 then a.order_id
            end
        ) as hh_fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌'
                and agent_level1 like 'HH直营' and is_td_finish = 1 then a.order_id
            end
        ) as hh_zhiying_fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌'
                and agent_level1 like 'KP代运营' and is_td_finish = 1 then a.order_id
            end
        ) as hh_kp_fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌'
                and agent_level1 like 'LX代运营' and is_td_finish = 1 then a.order_id
            end
        ) as hh_lx_fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌'
                and platform_name = 'API接入' and is_td_finish = 1 then a.order_id
            end
        ) as api_fin_cnt,
        count(
            distinct case
                when product_type <> '滴滴自营' and is_td_finish = 1 then a.order_id
            end
        ) as no_dd_fin_cnt,
        count(
            distinct case
                when product_type = '花小猪自营' and is_td_finish = 1 then a.order_id
            end
        ) as zy_fin_cnt,
        count(
            distinct case
                when product_type = '三方品牌' and is_td_finish = 1 then a.order_id
            end
        ) as sf_fin_cnt,
        count(
            distinct case
                when product_type = '滴滴自营' and is_td_finish = 1 then a.order_id
            end
        ) as dd_fin_cnt,
        sum(
            case
                when is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as gmv,
        sum(
            case
                when product_type = '三方品牌'
                and platform_name = '宏鹄'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as hh_gmv,
        sum(
            case
                when product_type = '三方品牌'
                and platform_name = 'API接入'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as api_gmv,
        sum(
            case
                when product_type = '滴滴自营'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as dd_gmv,
        sum(
            case
                when product_type <> '滴滴自营'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as no_dd_gmv,
        sum(
            case
                when product_type = '花小猪自营'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as zy_gmv,
        sum(
            case
                when product_type = '三方品牌'
                and is_td_finish = 1
                and a.gmv_amt > 0 then a.gmv_amt
            end
        ) as sf_gmv,
        sum(subsidy_c) as subsidy_c,
        sum(upcabin_subsidy_amt) as upcabin_subsidy_amt
    from
        (
            select
                distinct passenger_id,
                product_id,
                order_id,
                product_type,
                platform_name,
                gmv_amt,
                start_city_id as city_id,
                source_id,
                level_type,
                is_grab_before_cannel,
                answer_time,
                call_time,
                cancel_time,case
                    when answer_time != '0000-00-00 00:00:00' then (
                        unix_timestamp(answer_time) - unix_timestamp(call_time)
                    )
                    else (
                        unix_timestamp(cancel_time) - unix_timestamp(call_time)
                    )
                end as wait_dur --愿等时长
            from
                kflower_dw.dwd_kf_trd_unity_order_di
            where
                dt between '${start_date}'
                and '${end_date}'
                and is_hxz = 1
                and start_city_id in (${city_list})
        ) a
        left join (
            select
                *
            from
                kflower_dw.dim_kf_trd_unity_order_label_di
            where
                dt between '${start_date}'
                and '${end_date}'
        ) as b on a.source_id = b.source_id
        and a.order_id = b.order_id
        left join (
            select
                order_id,
                sum(subsidy_c) as subsidy_c
            from
                kflower_dw.dwm_kf_fin_order_report_di
            where
                dt between '${start_date}'
                and '${end_date}'
                and product_category in (1, 2)
            group by
                order_id
        ) e on a.order_id = e.order_id
        left join (
            select
                order_id,
                sum(subsidy_c) as upcabin_subsidy_amt --by订单升舱C补
            from
                kflower_dw.dwm_kf_fin_order_subsidy_c_di
            where
                dt between '${start_date}'
                and '${end_date}'
                and subsidy_type = 66
            group by
                order_id
        ) f on a.order_id = f.order_id
        left join (
            select
                product_id as proid,
                agent_level1
            from
                kflower_dw.dim_hh_com_company_agent_df
            where
                dt = '${end_date}'
        ) y on a.product_id = y.proid
    group by
        passenger_id
)
select
    group_name,
    count(distinct apollo_key) as pnum,
    sum(call_cnt) as call_cnt,
    sum(ans_cnt) as ans_cnt,
    sum(fin_cnt) as fin_cnt,
    sum(fin_cnt) / sum(call_cnt) as cr,
    sum(hh_fin_cnt) as hh_fin_cnt,
    sum(hh_zhiying_fin_cnt) as hh_zhiying_fin_cnt,
    sum(hh_kp_fin_cnt) as hh_kp_fin_cnt,
    sum(hh_lx_fin_cnt) as hh_lx_fin_cnt,
    sum(api_fin_cnt) as api_fin_cnt,
    sum(no_dd_fin_cnt) as no_dd_fin_cnt,
    sum(zy_fin_cnt) as zy_fin_cnt,
    sum(sf_fin_cnt) as sf_fin_cnt,
    sum(dd_fin_cnt) as dd_fin_cnt,
    sum(gmv) as gmv,
    sum(hh_gmv) as hh_gmv,
    sum(api_gmv) as api_gmv,
    sum(dd_gmv) as dd_gmv,
    sum(no_dd_gmv) as no_dd_gmv,
    sum(zy_gmv) as zy_gmv,
    sum(sf_gmv) as sf_gmv,
    sum(subsidy_c) as subsidy_c, -- 总C补
    sum(upcabin_subsidy_amt) as upcabin_subsidy_amt, -- 升舱花费C补
    sum(gmv) / count(distinct apollo_key) as `人均gmv`,
    sum(subsidy_c) / count(distinct apollo_key) as `人均C补`,
    case when group_name <> 'control_group' then sum(upcabin_subsidy_amt) / sum(gmv) else null end as `C补率`
from
    pas_tag
    left join finish on pas_tag.apollo_key = finish.passenger_id
    left join call on pas_tag.apollo_key = call.passenger_id
group by
    group_name
