```PlainText
with
-- 1) 找出 only举手订单
single_driver_orders as (
  select order_id
  from pk_feature_pid_exp_bi_all_std
  group by order_id
  having count(distinct driver_id) = 1
),

-- 2) 过滤出这些订单的数据
orders_only_one as (
  select p.*
  from pk_feature_pid_exp_bi_all_std p
  join single_driver_orders s on p.order_id = s.order_id
)

-- 3) 统计实验组 vs 对照组的司机取消情况
select
  o.exp_group,
  count(*) as total_orders,
  sum(case when o.class = '司机取消改派' then 1 else 0 end) as driver_cancel_cnt,
  avg(case when o.class = '司机取消改派' then 1.0 else 0.0 end) as driver_cancel_rate
from orders_only_one o
-- 应答后：排除未应答的情况
where o.is_finish is not null
group by o.exp_group;
```

```PlainText
with

                 end as pk_result

from    winners w

order by w.winner_group
```
