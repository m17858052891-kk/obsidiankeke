## 离线训练

### 当前模型问题

基于pre_dispatch_acc__v4模型和数据集提取逻辑训练xgboost

**原模型数据提取存在的问题**：为了扩充数据规模（给同一个order下未发单也没有预分单结果的数据根据显式规则加上label），导致模型部分学习了人工显式规则，从而在同一个数据构造规则下，训练集和测试集的表现都很好，但是线上效果无法保证（因为线上输入的都是特征，不会有人工规则）

```SQL
select
    distinct t1.order_id as order_id,
    t1.product_id as product_id,
    t2.predict_value as predict_value,
    case when t1.is_refused = 0 then 1
    else 0 end as label
from
    (
        select
            distinct order_id,
            product_id,
            estimate_fee,
            is_refused
        from
            kflower_dw.dwd_kf_trd_pre_dispatch_decision_di
        where
            dt = '2024-09-04'
            and is_refused <> 2 -- 0为接受，1、3为拒绝
            and action = 'passenger_decision'
            -- and product_id is not null
    ) t1 -- 真实历史数据表
    inner join (
        select
            distinct a.param ['order_id'] as order_id,
            a.param ['predict_value'] as predict_value,
            a.param ['passenger_id'] as passenger_id,
            a.param ['estimate_fee_amt'] as estimate_fee_amt,
            a.param ['business_id'] as business_id,
            b.business_id as business_id_b,
            b.product_id
        from
            (
                select
                    param
                from
                    kflower_ods.ods_log_kflower_ether_hxz_pre_dispatch_accept_model_h
                where
                    concat_ws('-', year, month, day) = '2024-09-04'
                    -- and '2024-09-11'
            ) a
            left join (
                select
                    distinct business_id,
                    product_id
                from
                    kflower_dw.dim_kf_trd_pbd_product_info_df
                where
                    dt = '2024-09-04'
                    -- and '2024-09-11'
            ) b on a.param ['business_id'] = b.business_id
    ) t2 on t1.order_id = t2.order_id
    and t1.product_id = t2.product_id
```

```SQL
with history_acc_rate as (
    select
        pid as passenger_id,
        dt,
        passenger_pre_dispatch_cnt_28day,
        passenger_pre_dispatch_cnt_14day,
        passenger_pre_dispatch_accept_cnt_28day,
        passenger_pre_dispatch_accept_cnt_14day,
        passenger_pre_dispatch_refuse_cnt_28day,
        passenger_pre_dispatch_refuse_cnt_14day,
        passenger_pre_dispatch_default_cnt_14day,
        passenger_pre_dispatch_default_cnt_28day
    from
        kflower_strategy.passenger_pre_dispatch_history_accept_rate
    where
        dt between '2024-05-31'
        and '2024-08-31'
),
pas_days_cnt_data as (
    select
        passenger_id,
        dt,
        bubble_cnt_7d as pas_last_7day_bubble_cnt_kf,
        bubble_cnt_14d as pas_last_14day_bubble_cnt_kf,
        bubble_cnt_28d as pas_last_28day_bubble_cnt_kf,
        avg_start_dest_dis_14d as pas_last_14day_avg_start_dest_dis_kf,
        avg_start_dest_dis_28d as pas_last_28day_avg_start_dest_dis_kf,
        asp_amt_28d as pas_last_28day_asp_amt_kf
    from
        kflower_marketplace.dwm_trip_pas_statistics_days_total_di
    where
        dt between '2024-05-31'
        and '2024-08-31'
),
basic_feature_data as (
    select
        concat_ws('-', year, month, day) as dt,
        param ['order_id'] as order_id,
        param ['passenger_id'] as passenger_id,
        param ["traceid"] as trace_id,
        cast(param ["business_id"] as bigint) business_id,
        cast(param ['city'] as int) as city,
        cast(param ["flat"] as float) as flat,
        cast(param ["flng"] as float) as flng,
        cast(param ["tlat"] as float) as tlat,
        cast(param ["tlng"] as float) as tlng,
        cast(param ['start_dest_distance'] as float) as start_dest_distance,
        cast(param ['estimate_fee_max'] as float) as estimate_fee_max,
        cast(param ['estimate_fee_min'] as float) as estimate_fee_min,
        cast(param ['basic_total_fee_amt'] as float) as basic_total_fee_amt,
        cast(param ['estimate_fee_amt'] as float) as estimate_fee_amt,
        case
            when cast(param ['order_waiting_period'] as bigint) > 1000000 then cast(param ['ctime'] as bigint) - cast(param ['order_waiting_period'] as bigint)
            else cast(param ['order_waiting_period'] as bigint)
        end as order_waiting_period,
        cast(param ['call_hour'] as int) as call_hour,
        cast(param ["predict_scene"] as float) as predict_scene,
        cast(param ["default_select_high_fee"] as float) as default_select_high_fee,
        cast(param ["default_selected_num"] as int) as default_selected_num,
        cast(param ["wait_relay_selected_num"] as int) as wait_relay_selected_num,
        cast(param ["call_week"] as int) as call_week,
        cast(param ['pas_portrait_consume_ability_kf'] as float) as pas_portrait_consume_ability_kf,
        cast(param ['coord_start_humidity'] as float) as coord_start_humidity,
        cast(param ['coord_start_temperature'] as float) as coord_start_temperature,
        cast(param ['coord_start_intensity'] as float) as coord_start_intensity,
        cast(param ['coord_start_skycon'] as float) as coord_start_skycon
    from
        kflower_ods.ods_log_kflower_ether_hxz_pre_dispatch_accept_model_h
    where
        concat_ws('-', year, month, day) between '2024-06-01'
        and '2024-09-01'
),
feature_tb as (
    select
        t1.*,
        cast(coalesce(t2.passenger_pre_dispatch_cnt_14day, 0) as int) as passenger_pre_dispatch_cnt_14day,
        cast(coalesce(t2.passenger_pre_dispatch_cnt_28day, 0) as int) as passenger_pre_dispatch_cnt_28day,
        cast(coalesce(
            t2.passenger_pre_dispatch_accept_cnt_14day / t2.passenger_pre_dispatch_cnt_14day,
            0
        ) as float) as passenger_pre_dispatch_accept_rate_14day,
        cast(coalesce(
            t2.passenger_pre_dispatch_accept_cnt_28day / t2.passenger_pre_dispatch_cnt_28day,
            0
        ) as float) as passenger_pre_dispatch_accept_rate_28day,
        cast(coalesce(
            t2.passenger_pre_dispatch_refuse_cnt_14day / t2.passenger_pre_dispatch_cnt_14day,
            0
        ) as float) as passenger_pre_dispatch_refuse_rate_14day,
        cast(coalesce(
            t2.passenger_pre_dispatch_refuse_cnt_28day / t2.passenger_pre_dispatch_cnt_28day,
            0
        ) as float) as passenger_pre_dispatch_refuse_rate_28day,
        cast(coalesce(
            t2.passenger_pre_dispatch_default_cnt_14day / t2.passenger_pre_dispatch_cnt_14day,
            0
        ) as float) as passenger_pre_dispatch_default_rate_14day,
        cast(coalesce(
            t2.passenger_pre_dispatch_default_cnt_28day / t2.passenger_pre_dispatch_cnt_28day,
            0
        ) as float) as passenger_pre_dispatch_default_rate_28day,
        cast(t3.pas_last_7day_bubble_cnt_kf as int) as pas_last_7day_bubble_cnt_kf,
        cast(t3.pas_last_14day_bubble_cnt_kf as int) as pas_last_14day_bubble_cnt_kf,
        cast(t3.pas_last_28day_bubble_cnt_kf as int) as pas_last_28day_bubble_cnt_kf,
        cast(t3.pas_last_14day_avg_start_dest_dis_kf as float) as pas_last_14day_avg_start_dest_dis_kf,
        cast(t3.pas_last_28day_avg_start_dest_dis_kf as float) as pas_last_28day_avg_start_dest_dis_kf,
        cast(t3.pas_last_28day_asp_amt_kf as float) as pas_last_28day_asp_amt_kf
    from
        basic_feature_data t1
        left join history_acc_rate t2 on t1.passenger_id = t2.passenger_id
        and t1.dt = date_add(t2.dt, 1)
        left join pas_days_cnt_data t3 on t1.passenger_id = t3.passenger_id
        and t1.dt = date_add(t3.dt, 1)
)
select
    case
        when a.is_refused = 0 then 1 --is_refused=0为接受
        else 0
    end as label,
    b.*
from
    (
        select
            distinct order_id,
            product_id,
            is_refused -- label
        from
            kflower_dw.dwd_kf_trd_pre_dispatch_decision_di
        where
            dt between '2024-05-31'
            and '2024-08-31'
            and is_refused <> 2
            and action = 'passenger_decision'
    ) a
    inner join -- 特征表
    (
        select
            t1.*,
            t2.product_id,
            cast((t1.estimate_fee_amt - t1.estimate_fee_max) as float) as estimate_fee_max_delta,
            cast((t1.estimate_fee_amt - t1.estimate_fee_min) as float) as estimate_fee_min_delta,
            cast((t1.estimate_fee_amt - t1.estimate_fee_max) / t1.estimate_fee_max as float) as price_diff_ratio,
            cast(t1.estimate_fee_max - t1.estimate_fee_min as float) as selected_price_diff,
            cast(t1.estimate_fee_amt / t1.start_dest_distance as float) as fee_per_dis
        from
            feature_tb t1
            left join (
                select
                    distinct business_id,
                    product_id
                from
                    kflower_dw.dim_kf_trd_pbd_product_info_df
                where
                    dt = '2024-09-14'
            ) t2 on t1.business_id = t2.business_id
    ) b on a.order_id = b.order_id
    and a.product_id = b.product_id
```
