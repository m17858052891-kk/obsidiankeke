```PlainText
with flow_data as (
    select
        DISTINCT exp_group,
        apollo_key
    from
        apollo.exp_group_data
    where
        ymd between '2022-07-23'
        and '2022-07-29'
        and experiment = 'calcite_grid_order_ab_hold_v1'
        and apollo_key is not null
        and apollo_key != '0'
),
call_log1 as (
    select
        order_id,
        bubble_trace_id as estimate_trace_id,
        estimate_id,
        product_2level
    from
        kflower_dw.dwd_kf_trd_order_create_base_di
    where
        dt between '2022-07-23'
        and '2022-07-29'
        and source_type <> 2
),
call_log2 as (
    select
        order_id,
        estimate_trace_id,
        estimate_id,
        product_2level
    from
        kflower_dw.dwd_kf_trd_order_append_di
    where
        dt between '2022-07-23'
        and '2022-07-29'
        and log_type = 2
),
call_log as (
    select
        *
    from
        call_log1
    union
    select
        *
    from
        call_log2
),
ord_origin as (
    select
        order_id,
        product_1level,
        product_2level,
        product_2level_name,
        bubble_trace_id,
        is_td_call,
        is_td_answer,
        is_td_finish
    from
        kflower_dw.dwd_kf_trd_order_base_di
    where
        dt between '2022-07-23'
        and '2022-07-30'
        and product_1level not in (3, 4)
        and source_type <> 2
),
ord_gaipai as (
    select
        reassign_origin_order_id as order_id,
        product_1level,
        product_2level,
        product_2level_name,
        is_td_call,
        is_td_answer,
        is_td_finish
    from
        kflower_dw.dwd_kf_trd_order_base_di
    where
        dt between '2022-07-23'
        and '2022-07-30'
        and product_1level not in (3, 4)
        and order_id in (
            select
                order_id
            from
                (
                    select
                        order_id,
                        rank() over(
                            partition by reassign_origin_order_id
                            order by
                                unix_timestamp(call_time) desc
                        ) as `rank`
                    from
                        kflower_dw.dwd_kf_trd_order_base_di
                    where
                        dt between '2022-07-23'
                        and '2022-07-30'
                        and source_type = 2
                        and product_1level not in (3, 4)
                ) as a
            where
                `rank` = 1
        )
),
winner_info as (
    select
        concat_ws('-', year, month, day) as date_,
        int(cityid) as cityid,
        int(channelid) as channelid,
        orderid,
        driverid,
        round_token,
        min(pkvalue) as win_pkvalue,
        min(unix_timestamp(`timestamp`)) as win_time,
        min(int(cpid)) as cpid,
        count(cpid) as cp_num
    from
        kflower_ods.ods_log_kflower_g_order_dusepk_auction
    where
        concat_ws('-', year, month, day) between '2022-07-23'
        and '2022-07-29'
        and action = 'auction_time'
        and (
            pktype = 15
            or pktype = 17
        )
    group by
        concat_ws('-', year, month, day),
        cityid,
        channelid,
        orderid,
        driverid,
        round_token
    having
        cp_num = 1
),
-- 胜者信息。
hands_up as (
    select
        concat_ws('-', year, month, day) as date_,
        int(cityid) as cityid,
        int(channelid) as channelid,
        orderid,
        driverid,
        round_token,
        int(multi) as pkvalue,
        unix_timestamp(`timestamp`) as tend_time,
        int(cpid) as cpid,
        case
            when iscontinue = 'true' then 1
            else 0
        end as is_continue
    from
        kflower_ods.ods_log_kflower_g_order_strivetender
    where
        concat_ws('-', year, month, day) between '2022-07-23'
        and '2022-07-29'
        and (
            pktype = 15
            or pktype = 17
        )
),
-- (TBD)在pk中出现过的所有订单，用于计算pk胜出->应答，pk胜出->完单。
hands_up_info as (
    select
        hands_up.date_ as date_,
        hands_up.cityid as cityid,
        hands_up.channelid as channelid,
        hands_up.orderid as orderid,
        hands_up.round_token as round_token,
        hands_up.cpid as hands_up_cp,
        hands_up.driverid as driverid,
        hands_up.pkvalue as hands_up_pkvalue,
        winner_info.cpid as win_cp,
        winner_info.driverid as win_driverid,
        winner_info.win_time as win_time,
        winner_info.win_pkvalue as win_pkvalue,
        case
            when is_continue = 1
            and tend_time <= win_time then 1
            else 0
        end as is_valid
    from
        hands_up
        inner join winner_info on hands_up.date_ = winner_info.date_
        and hands_up.cityid = winner_info.cityid
        and hands_up.channelid = winner_info.channelid
        and hands_up.orderid = winner_info.orderid
        and hands_up.round_token = winner_info.round_token
),
-- 带是否有效举手标记、胜者信息的举手信息。
all_hands_up_info as (
    select
        a.date_ as date_,
        a.cityid as cityid,
        a.channelid as channelid,
        a.orderid as orderid,
        a.round_token as round_token,
        a.hands_up_cp as hands_up_cp,
        a.driverid as driverid,
        a.hands_up_pkvalue as hands_up_pkvalue,
        a.win_cp as win_cp,
        a.win_driverid as win_driverid,
        a.win_time as win_time,
        a.win_pkvalue as win_pkvalue,
        a.is_valid as is_valid,
        b.valid_cp_num as valid_cp_num,
        b.all_cp_num as all_cp_num
    from
        hands_up_info a
        inner join (
            select
                date_,
                cityid,
                channelid,
                orderid,
                round_token,
                count(
                    distinct case
                        when is_valid = 1 then hands_up_cp
                        else null
                    end
                ) as valid_cp_num,
                count(
                    distinct hands_up_cp
                ) as all_cp_num
            from
                hands_up_info
            group by
                date_,
                cityid,
                channelid,
                orderid,
                round_token
        ) b on a.date_ = b.date_
        and a.cityid = b.cityid
        and a.channelid = b.channelid
        and a.orderid = b.orderid
        and a.round_token = b.round_token
),
-- 带是否有效举手标记、胜者信息的举手信息,单轮次三方品牌数量。
is_pked_info as (
    select
        a.date_ as date_,
        a.cityid as cityid,
        a.channelid as channelid,
        a.orderid as orderid,
        shiftleft(
            shiftright(cast(a.orderid as bigint), 46) & 262143,
            32
        ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
        --高位转低位
        a.hands_up_cp as hands_up_cp,
        a.driverid as hands_up_driverid,
        min(a.valid_cp_num) as valid_cp_num,
        min(a.all_cp_num) as all_cp_num,
        min(a.hands_up_pkvalue) as hands_up_pkvalue,
        min(a.win_pkvalue) as win_pkvalue,
        max(a.win_driverid) as win_driverid,
        max(a.win_cp) as win_cp,
        max(a.is_valid) as is_valid,
        max(
            case
                when a.is_valid = 1
                and b.is_valid = 1
                and a.hands_up_cp = b.hands_up_cp then 1
                when a.is_valid = 1
                and b.is_valid = 1
                and a.hands_up_cp != b.hands_up_cp then 2
                else 0
            end
        ) as is_pked -- 0 无效举手 1 only举手 2 both举手
    from
        all_hands_up_info a
        inner join all_hands_up_info b on a.date_ = b.date_
        and a.cityid = b.cityid
        and a.channelid = b.channelid
        and a.orderid = b.orderid
        and a.round_token = b.round_token
    group by
        a.date_,
        a.cityid,
        a.channelid,
        a.orderid,
        a.round_token,
        a.driverid,
        a.hands_up_cp
),
-- 举手信息中增加是否发生真实pk。 在一个订单的同一个轮次中，若一个cp举手两次，第一次有效，第二次无效，则计为一次有效举手。
hxz as (
    select
        order_id,
        driver_id,
        is_td_answer as is_td_answer_flag,
        is_td_finish as is_td_finish_flag
    from
        kflower_dw.dwd_kf_trd_order_base_di
    where
        dt between '2022-07-23'
        and '2022-07-29'
        and is_td_call = 1
        and product_1level <> 4
),
pbd as (
    select
        a.order_id as order_id,
        driver_id,
        is_td_answer_flag,
        is_td_finish_flag
    from
        (
            select
                dt,
                call_city,
                order_id,
                driver_id,
                is_td_answer_flag,
                source_type,
                is_td_reassign_flag,
                cancel_type,
                is_td_finish_flag,
                case
                    when cancel_type = 2
                    and is_td_answer_flag = 0 then lag(product_id) over(
                        partition by trace_order_id
                        order by
                            call_time
                    )
                    else product_id
                end product_id
            from
                (
                    select
                        *,
                        if(
                            source_type <> 2,
                            order_id,
                            reassign_origin_order_id
                        ) as trace_order_id -- 原单和其改派单的trace订单
                    from
                        decdm.dwd_gflt_ord_order_base_di -- 订单表
                    where
                        dt between '2022-07-23'
                        and '2022-07-29'
                        and is_td_call_flag = 1
                ) t
        ) a
        inner join (
            select
                order_id
            from
                dd_cdm.dwd_trip_trd_order_anycar_di
            where
                dt between '2022-07-23'
                and '2022-07-29'
                and channel_id in (1000001, 1010001)
            group by
                order_id
        ) b on a.order_id = b.order_id
),
didi as (
    select
        a.order_id as order_id,
        driver_id,
        is_td_answer_flag,
        is_td_finish_flag
    from
        (
            select
                dt,
                call_city,
                order_id,
                driver_id,
                is_td_answer_flag,
                source_type,
                is_td_reassign_flag,
                cancel_type,
                is_td_finish_flag,
                case
                    when cancel_type = 2
                    and is_td_answer_flag = 0 then lag(product_id) over(
                        partition by trace_order_id
                        order by
                            call_time
                    )
                    else product_id
                end product_id
            from
                (
                    select
                        *,
                        if(
                            source_type <> 2,
                            order_id,
                            reassign_origin_order_id
                        ) as trace_order_id -- 原单和其改派单的trace订单
                    from
                        decdm.dwd_gflt_ord_order_base_di -- 订单表
                    where
                        dt between '2022-07-23'
                        and '2022-07-29'
                        and is_td_call_flag = 1
                ) t
        ) a
        inner join (
            select
                order_id
            from
                dd_cdm.dwd_trip_trd_order_anycar_di
            where
                dt between '2022-07-23'
                and '2022-07-29'
                and level_3_product in (110721, 110722)
                and channel_id not in (1000001, 1010001)
            group by
                order_id
        ) b on a.order_id = b.order_id
),
pk_data as (
    select
        date_,
        cityid,
        source_platform,
        appid,
        orderid,
        only_pk_orderid,
        both_pk_orderid,
        win_orderid,
        both_win_orderid
    from
        (
            select
                date_,
                cityid,
                channelid,
                (
                    case
                        when channelid = 1 then 'dd'
                        when channelid = 2 then 'pbd'
                        else 'hxz'
                    end
                ) as source_platform,
                hands_up_cp as appid,
                case
                    when is_valid = 1 then orderid
                    else null
                end orderid,
                case
                    when is_pked = 1 then orderid
                    else null
                end only_pk_orderid,
                case
                    when is_pked = 2 then orderid
                    else null
                end both_pk_orderid,
                case
                    when win_cp = hands_up_cp then orderid
                    else null
                end win_orderid,
                case
                    when win_cp = hands_up_cp
                    and is_pked = 2 then orderid
                    else null
                end both_win_orderid
            from
                is_pked_info
                left join (
                    select
                        order_id,
                        driver_id,
                        is_td_answer_flag,
                        is_td_finish_flag,
                        1 as channel_id
                    from
                        didi
                    union all
                    select
                        order_id,
                        driver_id,
                        is_td_answer_flag,
                        is_td_finish_flag,
                        2 as channel_id
                    from
                        pbd
                    union all
                    select
                        order_id,
                        driver_id,
                        is_td_answer_flag,
                        is_td_finish_flag,
                        3 as channel_id
                    from
                        hxz
                ) order_info on is_pked_info.order_id = order_info.order_id
                and is_pked_info.win_driverid = order_info.driver_id
                and is_pked_info.channelid = order_info.channel_id
        ) t
    group by
        date_,
        cityid,
        source_platform,
        appid,
        orderid,
        only_pk_orderid,
        both_pk_orderid,
        win_orderid,
        both_win_orderid
    having
        appid = 1197
        and source_platform = 'hxz'
)
select
    exp_group,
    count(distinct call_log.order_id) as call_all,
    count(
        distinct case
            when ord_origin.is_td_finish = 1
            or ord_gaipai.is_td_finish = 1 then call_log.order_id
            else null
        end
    ) as finsih_all,
    count(
        distinct case
            when call_log.product_2level in (80, 81, 82) then call_log.order_id
            else null
        end
    ) as call_zy,
    count(
        distinct case
            when call_log.product_2level not in (80, 81, 82, 384) then call_log.order_id
            else null
        end
    ) as call_sf,
    count(
        distinct case
            when call_log.product_2level not in (384) then call_log.order_id
            else null
        end
    ) as call_zy_sf,
    count(
        distinct case
            when call_log.product_2level in (384) then call_log.order_id
            else null
        end
    ) as call_didi,
    count(
        distinct case
            when (
                ord_origin.is_td_finish = 1
                and ord_origin.product_2level in (80, 81, 82)
            )
            or (
                ord_gaipai.is_td_finish = 1
                and ord_gaipai.product_2level in (80, 81, 82)
            ) then ord_origin.order_id
            else null
        end
    ) as finsih_zy,
    count(
        distinct case
            when (
                ord_origin.is_td_finish = 1
                and ord_origin.product_2level not in (80, 81, 82, 384)
            )
            or (
                ord_gaipai.is_td_finish = 1
                and ord_gaipai.product_2level not in (80, 81, 82, 384)
            ) then ord_origin.order_id
            else null
        end
    ) as finsih_sf,
    count(
        distinct case
            when (
                ord_origin.is_td_finish = 1
                and ord_origin.product_2level not in (384)
            )
            or (
                ord_gaipai.is_td_finish = 1
                and ord_gaipai.product_2level not in (384)
            ) then ord_origin.order_id
            else null
        end
    ) as finsih_zy_sf,
    count(
        distinct case
            when (
                ord_origin.is_td_finish = 1
                and ord_origin.product_2level in (384)
            )
            or (
                ord_gaipai.is_td_finish = 1
                and ord_gaipai.product_2level in (384)
            ) then ord_origin.order_id
            else null
        end
    ) as finsih_didi,
    count(distinct orderid) valid_pk_order_cnt,
    count(distinct only_pk_orderid) only_pk_order_cnt,
    count(distinct both_pk_orderid) both_pk_order_cnt,
    count(distinct win_orderid) win_order_cnt,
    count(distinct both_win_orderid) both_win_order_cnt
from
    flow_data
    left join call_log on shiftleft(
        shiftright(cast(flow_data.apollo_key as bigint), 46) & 262143,
        32
    ) +(
        cast(flow_data.apollo_key as bigint) & 4294967295
    ) = call_log.order_id
    left join ord_origin on shiftleft(
        shiftright(cast(flow_data.apollo_key as bigint), 46) & 262143,
        32
    ) +(
        cast(flow_data.apollo_key as bigint) & 4294967295
    ) = ord_origin.order_id
    left join ord_gaipai on ord_origin.order_id = ord_gaipai.order_id
    left join pk_data on flow_data.apollo_key = pk_data.orderid
group by
    exp_group
order by
    exp_group;
```
