# 大盘实验评估sql（平台）


（目前sql中司机接驾时长仍用的是开始计费口径，待改为到达口径）

```SQL
--========================================================================
-- 平台漏斗实验评估 整合版
-- 归一订单口径全部对齐 平台_ETP.sql：
--   1) 归一聚合时过滤"换车未应答子单"
--   2) origin_order_detail 限 is_td_answer = 1（至少被应答过）
--   3) order_status 使用完整 8 状态枚举（含"应答前乘客取消"/"客服取消改派"）
-- 输出指标 = (平台漏斗评估_结合中间表) ∪ (平台_ETP) ∪ (司机接驾时长)
-- 全部为人均口径（分子 / 分流乘客数），并附带对应 var_ 字段
-- 分流节点：乘客发单被应答后 (apollo: change_car_exp_4)
--========================================================================

with
----- 实验用户分组 -----
exp_group as (
    select
        distinct concat_ws('-', year, month, day) dt,
        apollo_key as pid,
        group_name as exp_group,
        dims['city'] city_id
    from
        apollo.pre_flow_data_etl
    where
        concat_ws('-', year, month, day) between '${exp_start_dt}' and '${exp_end_dt}'
        and exp = 'recall_exp_v1_2'
),
exp_group_clean as ---剔除窜组用户
(
    select
        distinct a.dt,
        a.pid,
        a.exp_group,
        city_id
    from
        exp_group a
        left join (
            select
                pid,
                count(distinct exp_group) group_cnt
            from
                exp_group
            group by pid
        ) b on a.pid = b.pid
    where
        group_cnt = 1
),

----------------冒泡 / 呼叫（来自 平台漏斗评估_结合中间表）---------------
wyc_bubble as (
    select
        dt,
        passenger_id as pid,
        city_id,
        count(distinct distinct_bubble_id) as wyc_bubble_cnt
    from
        (
            select
                bubble_id,
                dt,
                passenger_id,
                city_id
            from
                kflower_dw.dwd_kf_trd_bubble_base_di
            where
                dt between '${start_dt}' and '${end_dt}'
                and product_1level in (1, 2)
                and is_anycar_form = 1
                and is_from_waiting_response <> 1
                and passenger_id <> 0
        ) a
        left join (
            select
                estimate_id,
                distinct_bubble_id
            from
                kflower_dw.dim_kf_trd_unity_bubble_anycar_label_di
            where
                dt between '${start_dt}' and '${end_dt}'
                and source_id = 'kf'
        ) c on a.bubble_id = c.estimate_id
    group by dt, passenger_id, city_id
),
wyc_call as (
    select
        dt,
        passenger_id as pid,
        start_city_id as city_id,
        count(distinct case when is_compete_call = 1 then concat(source_id, order_id) end) as wyc_call_cnt
    from
        (
            select
                order_id, passenger_id, start_city_id, source_id,
                estimate_id, product_id, call_car_level_id, dt
            from
                kflower_dw.dwd_kf_trd_unity_order_anycar_di
            where
                dt between '${start_dt}' and '${end_dt}'
                and passenger_id <> 0
                and is_hxz_call = 1
                and to_date(call_time) = dt
                and product_id <> 26
        ) a
        left join (
            select
                dt as dtt, order_id as od, estimate_id, product_id, call_car_level_id,
                is_compete_call, is_objective_call, is_platform_objective_call
            from
                kflower_dw.dim_kf_trd_unity_order_anycar_label_di
            where
                dt between '${start_dt}' and '${end_dt}'
        ) c
        on a.order_id = c.od and a.estimate_id = c.estimate_id
        and a.product_id = c.product_id and a.call_car_level_id = c.call_car_level_id
        and a.dt = c.dtt
    group by dt, passenger_id, start_city_id
),
-- 去重呼叫：在 wyc_call 基础上剔除"重发单子单"
-- 来自 换车评估_剔除重发单.sql：pltf_union_stg_deduplicated_call_info 中
--   z_recall_order_id 即重发子单，left join 后取 z_recall_order_id is null 即剔除
wyc_call_dedup as (
    select
        dt,
        passenger_id as pid,
        start_city_id as city_id,
        count(distinct case when is_compete_call = 1 then concat(source_id, order_id) end) as wyc_call_dedup_cnt
    from
        (
            select
                order_id, passenger_id, start_city_id, source_id,
                estimate_id, product_id, call_car_level_id, dt
            from
                kflower_dw.dwd_kf_trd_unity_order_anycar_di
            where
                dt between '${start_dt}' and '${end_dt}'
                and passenger_id <> 0
                and is_hxz_call = 1
                and to_date(call_time) = dt
                and product_id <> 26
        ) a
        left join (
            select
                dt as dtt, order_id as od, estimate_id, product_id, call_car_level_id,
                is_compete_call, is_objective_call, is_platform_objective_call
            from
                kflower_dw.dim_kf_trd_unity_order_anycar_label_di
            where
                dt between '${start_dt}' and '${end_dt}'
        ) c
        on a.order_id = c.od and a.estimate_id = c.estimate_id
        and a.product_id = c.product_id and a.call_car_level_id = c.call_car_level_id
        and a.dt = c.dtt
        --- 剔除重发单子单
        left join (
            select
                f_recall_order_id,
                z_recall_order_id,
                dt as d_dt
            from
                -- kflower_strategy.pltf_union_stg_deduplicated_call_info --只有应答后乘客取消
                kflower_strategy.deduplicated_call --应答前取消and应答后乘客取消
            where
                dt between '${start_dt}' and '${end_dt}'
        ) z
        on a.order_id = z.z_recall_order_id and a.dt = z.d_dt
    where z.z_recall_order_id is null   -- 当前 order 不是重发子单 → 保留
    group by dt, passenger_id, start_city_id
),

------------------------- unity 订单基础表 -------------------------
-- 取并集字段：保留 ETP / 中间表 / 接驾时长 三方所需全部字段
unity_order_base as (
    select
        a.*,
        is_td_answer,
        is_td_finish,
        gmv,
        order_charge_dis,
        charge_dur / 3600 as charge_dur,
        is_grab_after_dri_cancel,
        is_grab_after_pas_cancel,
        is_grab_after_srvc_cancel,
        is_td_answer_without_reassign,
        is_grab_after_dri_cancel_without_reassign,
        is_grab_after_pas_cancel_without_reassign,
        is_grab_after_srvc_cancel_without_reassign,
        ---funnel 表取预估接驾信息
        est_arrive_dis_funnel,
        est_arrive_dur_funnel,
        driver_to_last_dest_eta,
        last_dest_to_start_eta
    from
        (
            select distinct
                start_city_id city_id,
                dt,
                order_id,
                estimate_id,
                driver_id,
                passenger_id,
                source_id,
                is_td_call,
                is_pbd, is_hxz, is_thlm,
                product_id, product_type, platform_name, platform_mode_name,
                answer_time, cancel_time, call_time,
                arrive_time,           -- 接驾时长支线需要
                begin_charge_time,
                end_charge_time,       -- 接驾时长支线需要
                answer_reassign_order_id,
                call_reassign_order_id,
                channel_name,
                driver_phone_md5,
                tripcloud_open_oid,
                is_reassign,
                is_td_reassign,
                is_grab_before_cannel,
                coalesce(call_reassign_order_id, answer_reassign_order_id, order_id) as origin_order_id,
                answer_dur,
                arrive_dur,
                call_car_level_id,
                coalesce(combo_type, 0) as combo_type,
                coalesce(level_type, 0) as level_type,
                is_serial_assign
            from
                kflower_dw.dwd_kf_trd_unity_order_di
            where
                dt between '${start_dt}' and '${end_dt}'
                and is_hxz = 1
        ) a
        left join (
            select distinct
                dt dt2,
                source_id,
                order_id as order_id2,
                is_td_answer, is_td_finish,
                gmv_amt as gmv,
                order_charge_dis,
                charge_dur,
                is_grab_after_dri_cancel,
                is_grab_after_pas_cancel,
                is_grab_after_srvc_cancel,
                is_td_answer_without_reassign,
                is_grab_after_dri_cancel_without_reassign,
                is_grab_after_pas_cancel_without_reassign,
                is_grab_after_srvc_cancel_without_reassign
            from
                kflower_dw.dim_kf_trd_unity_order_label_di
            where
                dt between '${start_dt}' and '${end_dt}'
        ) b on a.source_id = b.source_id and a.order_id = b.order_id2 and a.dt = b.dt2
        left join (
            select
                order_id, product_id, call_car_level_id,
                coalesce(combo_type, 0) as combo_type,
                coalesce(level_type, 0) as level_type,
                driver_start_eda as est_arrive_dis_funnel,
                driver_start_eta as est_arrive_dur_funnel,
                driver_to_last_dest_eta,
                last_dest_to_start_eta
            from
                kflower_dw.dwm_kf_trd_unity_trade_funnel_order_anycar_di
            where
                dt between '${start_dt}' and '${end_dt}'
        ) c
        on a.order_id = c.order_id and a.product_id = c.product_id
        and a.call_car_level_id = c.call_car_level_id
        and a.combo_type = c.combo_type and a.level_type = c.level_type
),

------------------------- 换车扩展信息（中间表逻辑） -------------------------
ord_extend as (
    select
        distinct order_id,
        get_json_object(call_more_driver, '$.change_car_type') change_car_type,
        get_json_object(call_more_driver, '$.type') fz_type
    from
        (
            select
                *,
                get_json_object(extend_feature, '$.call_more_driver') as call_more_driver
            from
                kflower_dw.dwm_kf_trd_hxz_order_base_view_di
            where
                dt between '${start_dt}' and '${end_dt}'
        ) t
),
cancel_reason_base as ---乘客取消勾选原因
(
    select
        order_id, max(cancel_reason) cancel_reason
    from
        kflower_dw.dwd_kf_pas_cancel_reason_di
    where
        dt between '${start_dt}' and '${end_dt}'
    group by order_id
),

------------------------- 订单粒度聚合 -------------------------
-- 状态枚举使用 ETP 的 8 状态完整版
order_rank_max as (
    select
        a.*,
        to_date(call_time) as call_dt,
        case
            when is_td_answer = 0 and is_grab_before_cannel = 1 then '应答前乘客取消'
            when is_td_answer = 1 and is_td_finish = 1 then '完单'
            when is_td_answer = 1 and is_grab_after_dri_cancel = 1 and is_grab_after_dri_cancel_without_reassign = 1 then '司机取消无改派'
            when is_td_answer = 1 and is_grab_after_dri_cancel = 1 and is_grab_after_dri_cancel_without_reassign <> 1 then '司机取消改派'
            when is_td_answer = 1 and is_grab_after_pas_cancel = 1 and is_grab_after_pas_cancel_without_reassign = 1 then '乘客取消无改派'
            when is_td_answer = 1 and is_grab_after_pas_cancel = 1 and is_grab_after_pas_cancel_without_reassign <> 1 then '乘客取消改派'
            when is_td_answer <> 1 and is_grab_after_pas_cancel_without_reassign = 1 then '有过应答且最后一单应答前乘客取消'
            when is_td_answer = 1 and is_grab_after_srvc_cancel = 1 and is_grab_after_srvc_cancel_without_reassign = 1 then '客服取消无改派'
            when is_td_answer = 1 and is_grab_after_srvc_cancel = 1 and is_grab_after_srvc_cancel_without_reassign <> 1 then '客服取消改派'
        end as order_status,
        case when t.order_id is not null then 1 else 0 end as is_callmoredri_sub_ord
    from
        (
            select
                order_id,
                max(origin_order_id) as origin_order_id,
                max(city_id) as city_id,
                max(driver_id) as driver_id,
                max(product_id) as product_id,
                max(passenger_id) as passenger_id,
                max(platform_mode_name) as platform_mode_name,
                max(tripcloud_open_oid) as tripcloud_open_oid,
                max(is_td_call) as is_td_call,
                max(is_td_answer) as is_td_answer,
                max(is_td_finish) as is_td_finish,
                max(gmv) as gmv,
                max(is_grab_before_cannel) as is_grab_before_cannel,
                max(is_grab_after_dri_cancel) as is_grab_after_dri_cancel,
                max(is_grab_after_pas_cancel) as is_grab_after_pas_cancel,
                max(is_grab_after_srvc_cancel) as is_grab_after_srvc_cancel,
                max(is_td_answer_without_reassign) as is_td_answer_without_reassign,
                max(is_grab_after_dri_cancel_without_reassign) as is_grab_after_dri_cancel_without_reassign,
                max(is_grab_after_pas_cancel_without_reassign) as is_grab_after_pas_cancel_without_reassign,
                max(is_grab_after_srvc_cancel_without_reassign) as is_grab_after_srvc_cancel_without_reassign,
                max(call_time) as call_time,
                max(answer_time) as answer_time,
                max(arrive_time) as arrive_time,
                max(begin_charge_time) as begin_charge_time,
                max(end_charge_time) as end_charge_time,
                max(cancel_time) as cancel_time,
                max(answer_dur) as answer_dur,
                max(arrive_dur) as arrive_dur,
                max(charge_dur) as charge_dur,
                max(order_charge_dis) as order_charge_dis,
                max(est_arrive_dis_funnel) as est_arrive_dis_funnel,
                max(est_arrive_dur_funnel) as est_arrive_dur_funnel,
                max(is_serial_assign) as is_serial_assign,
                max(nvl(driver_to_last_dest_eta, 0)) as driver_to_last_dest_eta,
                max(nvl(last_dest_to_start_eta, 0)) as last_dest_to_start_eta,
                max(case when is_td_answer = 1 and is_serial_assign = 1 then 1 else 0 end) as is_answer_serial_assign
            from
                unity_order_base
            where
                substr(call_time, 1, 10) between '${start_dt}' and '${end_dt}'
            group by order_id
        ) a
        --- 标记换车子单（用于后续过滤未应答换车子单）
        left join (
            select distinct order_id
            from
                (
                    select
                        order_id, change_car_type, fz_type
                    from ord_extend
                ) x
            where change_car_type in (1, 2) and fz_type = 2
        ) t on a.order_id = t.order_id
),

-- 补充乘客取消原因
order_rank as (
    select
        orm.*,
        cr.cancel_reason
    from order_rank_max orm
    left join cancel_reason_base cr on orm.order_id = cr.order_id

------------------------- 归一原始订单（按 ETP 口径） -------------------------
-- 1) 过滤掉换车未应答子单
-- 2) 末尾限制 is_td_answer = 1（必须被应答过）
origin_order_detail as (
    select
        *,
        to_date(call_time_min) as dt,
        unix_timestamp(cancel_time_all_min) - unix_timestamp(call_time_min) as call_to_firstcancel_dur,
        unix_timestamp(answer_time_min)     - unix_timestamp(call_time_min) as call_to_firstans_dur,
        unix_timestamp(begin_charge_time_min) - unix_timestamp(call_time_min) as call_to_firstcharge_dur,
        unix_timestamp(arrive_time_min)     - unix_timestamp(call_time_min) as call_to_firstarrive_dur,
        --- 兼容 (中间表) 的等待时长口径：首次应答 → 上车 / 最后取消
        case
            when is_last_finish = 1 then unix_timestamp(begin_charge_time_max) - unix_timestamp(answer_time_min)
            when is_last_finish <> 1 then unix_timestamp(cancel_time_max)       - unix_timestamp(answer_time_min)
        end as wait_cancel_dur
    from
        (
            select
                origin_order_id,
                max(passenger_id) passenger_id,
                ---- 时间戳聚合（用 is_td_call/is_td_answer/is_td_finish 守卫）
                min(case when is_td_call = 1 then call_time end) call_time_min,
                min(case when is_td_answer = 1 then answer_time end) answer_time_min,
                max(case when is_td_answer = 1 then answer_time end) answer_time_max,
                min(case when is_td_answer = 1 and is_td_finish <> 1 then cancel_time end) cancel_time_min,
                max(case when is_td_answer = 1 and is_td_finish <> 1 then cancel_time end) cancel_time_max,
                min(case when is_td_answer = 1 and is_td_finish = 1 then begin_charge_time end) begin_charge_time_min,
                max(case when is_td_answer = 1 and is_td_finish = 1 then begin_charge_time end) begin_charge_time_max,
                min(case when is_td_answer = 1 and is_td_finish = 1 then arrive_time end) arrive_time_min,
                max(case when is_td_answer = 1 and is_td_finish = 1 then arrive_time end) arrive_time_max,
                min(case when is_td_finish <> 1 and cancel_time not in ('0000-00-00 00:00:00') then cancel_time end) cancel_time_all_min,
                max(case when is_td_finish <> 1 and cancel_time not in ('0000-00-00 00:00:00') then cancel_time end) cancel_time_all_max,
                ---- 归一聚合状态
                count(distinct order_id) as origin_ord_cnt,
                max(is_td_answer) as is_td_answer,
                max(is_grab_before_cannel) as is_grab_before_cannel,
                max(case when order_status in ('完单') then 1 else 0 end) as is_last_finish,
                max(case when reassign_rank_desc = 1 then order_status end) as order_status_reassign_rank_desc
            from
                (
                    select
                        a.*,
                        row_number() over(partition by origin_order_id order by call_time)      as reassign_rank,
                        row_number() over(partition by origin_order_id order by call_time desc) as reassign_rank_desc
                    from
                        order_rank a
                        left join (
                            select distinct order_id
                            from order_rank
                            where is_callmoredri_sub_ord = 1 and is_td_answer <> 1
                        ) b on a.order_id = b.order_id
                    where b.order_id is null  -- 过滤未应答的换车子单
                ) t
            group by origin_order_id
        ) t
    where is_td_answer = 1   -- ETP 口径：归一单至少被应答过
),

------------------------- 司机接驾时长 / 送驾时长 -------------------------
-- 来自 平台_司机接驾时长.sql：
--   可控运力（花小猪自营 / 宏鹄）+ 连环派 → 用 hh 侧表算实际接驾
--   非可控运力 + 连环派 → 用预估 driver_to_last_dest_eta 扣 ab 段
--   其他场景 → answer→上车（取消单 answer→cancel）
hh_order_base as (
    select
        a.*,
        b.is_serial_order
    from
        (
            select
                order_id, driver_id, is_td_answer, is_td_finish, is_td_cancel,
                channel_id, call_time, answer_time, cancel_time, arrive_time,
                charge_time, finish_time,
                start_lng, start_lat, answer_lng, answer_lat, dest_lng, dest_lat,
                open_oid, start_city_id
            from
                kflower_dw.dwm_hh_trd_order_merge_base_ext_view_di
            where dt between '${start_dt}' and '${end_dt}'
        ) a
        left join (
            select
                driver_id_hh, order_id_hh, order_origin,
                case
                    when dispatch_type = 1 then '抢单'
                    when dispatch_type = 2 then '半指派'
                    when dispatch_type = 3 then '全指派'
                    when dispatch_type in (4,5,6,7,8,9) then '其他'
                end dispatch_type,
                product_id_hh, is_serial_order
            from
                kflower_dw.dwd_hh_trd_duse_orderbroadcast_di
            where dt between '${start_dt}' and '${end_dt}'
            group by
                driver_id_hh, order_id_hh, order_origin,
                case
                    when dispatch_type = 1 then '抢单'
                    when dispatch_type = 2 then '半指派'
                    when dispatch_type = 3 then '全指派'
                    when dispatch_type in (4,5,6,7,8,9) then '其他'
                end,
                product_id_hh, is_serial_order
        ) b on a.driver_id = b.driver_id_hh and a.order_id = b.order_id_hh
),

zyhh_order as (
    select
        a.order_id,
        max(a.driver_id) as driver_id,
        max(a.is_td_answer) as is_td_answer,
        max(a.is_td_finish) as is_td_finish,
        max(a.channel_id) as channel_id,
        max(a.is_serial_order) as is_serial_order,
        max(a.call_time) as serial_call_time,
        max(to_date(a.call_time)) as dt,
        max(a.answer_time) as serial_answer_time,
        max(a.cancel_time) as serial_cancel_time,
        max(a.charge_time) as serial_charge_time,
        max(a.finish_time) as serial_finish_time,
        max(a.open_oid) as serial_open_oid,
        max(a.start_city_id) as serial_city_id,
        max(b.order_id) as parent_order_id,
        max(b.answer_time) as parent_answer_time,
        max(b.finish_time) as parent_finish_time
    from
        (select * from hh_order_base where is_td_answer = 1 and is_serial_order = 1) a
        left join (select * from hh_order_base where is_td_finish = 1) b
            on a.driver_id = b.driver_id
    where
        a.channel_id in (1100300001)
        and a.answer_time > b.answer_time
        and a.answer_time < b.finish_time
        and to_date(a.call_time) between '${start_dt}' and '${end_dt}'
    group by a.order_id
),

zyhh_serial_pickup_dur as (
    select
        dt, passenger_id,
        '可控运力' as type,
        '连环派' as is_serial,
        sum(case
                when is_td_finish = 1 then finish_pickup_dur
                when is_td_finish <> 1 and is_ab_cancel = 1 then 0
                when is_td_finish <> 1 and is_ab_cancel = 0 then cancel_pick_dur
            end) as pickup_dur,
        0 as charge_dur
    from
        (
            select
                a.*,
                if(is_td_finish <> 1 and serial_cancel_time <= parent_finish_time, 1, 0) as is_ab_cancel,
                unix_timestamp(serial_charge_time) - unix_timestamp(parent_finish_time) as finish_pickup_dur,
                unix_timestamp(serial_cancel_time) - unix_timestamp(parent_finish_time) as cancel_pick_dur
            from zyhh_order a
        ) a
        left join (
            select tripcloud_open_oid, max(passenger_id) as passenger_id
            from kflower_dw.dwd_kf_trd_unity_order_di
            where dt between '${start_dt}' and '${end_dt}' and is_hxz = 1
            group by tripcloud_open_oid
        ) b on a.serial_open_oid = b.tripcloud_open_oid
    group by dt, passenger_id
),

order_dur as (
    -- 可控 × 连环派（计费侧）
    select
        call_dt as dt, passenger_id, '可控运力' as type, '连环派' as is_serial,
        0 as pickup_dur,
        sum(unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time)) as charge_dur
    from order_rank
    where platform_mode_name in ('花小猪自营','宏鹄')
        and is_answer_serial_assign = 1
        and order_status in ('完单')
    group by call_dt, passenger_id

    union all

    -- 可控 × 非连环派
    select
        call_dt as dt, passenger_id, '可控运力' as type, '非连环派' as is_serial,
        sum(case
                when order_status in ('完单') then unix_timestamp(begin_charge_time) - unix_timestamp(answer_time)
                when order_status not in ('完单') then unix_timestamp(cancel_time)       - unix_timestamp(answer_time)
            end) as pickup_dur,
        sum(case when order_status in ('完单') then unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time) end) as charge_dur
    from order_rank
    where platform_mode_name in ('花小猪自营','宏鹄')
        and is_td_answer = 1
        and is_answer_serial_assign = 0
    group by call_dt, passenger_id

    union all

    -- 非可控 × 连环派（用预估 ab 段扣减）
    select
        call_dt as dt, passenger_id, '非可控运力' as type, '连环派' as is_serial,
        sum(case
                when order_status in ('完单') then unix_timestamp(begin_charge_time) - unix_timestamp(answer_time) - driver_to_last_dest_eta
                when order_status not in ('完单') and (unix_timestamp(cancel_time) - unix_timestamp(answer_time)) > driver_to_last_dest_eta
                    then (unix_timestamp(cancel_time) - unix_timestamp(answer_time) - driver_to_last_dest_eta)
                when order_status not in ('完单') and (unix_timestamp(cancel_time) - unix_timestamp(answer_time)) <= driver_to_last_dest_eta
                    then 0
            end) as pickup_dur,
        sum(case when order_status in ('完单') then unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time) end) as charge_dur
    from order_rank
    where platform_mode_name not in ('花小猪自营','宏鹄')
        and is_answer_serial_assign = 1
    group by call_dt, passenger_id

    union all

    -- 非可控 × 非连环派
    select
        call_dt as dt, passenger_id, '非可控运力' as type, '非连环派' as is_serial,
        sum(case
                when order_status in ('完单') then unix_timestamp(begin_charge_time) - unix_timestamp(answer_time)
                when order_status not in ('完单') then unix_timestamp(cancel_time)       - unix_timestamp(answer_time)
            end) as pickup_dur,
        sum(case when order_status in ('完单') then unix_timestamp(end_charge_time) - unix_timestamp(begin_charge_time) end) as charge_dur
    from order_rank
    where platform_mode_name not in ('花小猪自营','宏鹄')
        and is_td_answer = 1
        and is_answer_serial_assign = 0
    group by call_dt, passenger_id
),

all_order_dur as (
    select
        dt, passenger_id, type, is_serial,
        sum(pickup_dur) as pickup_dur,
        sum(charge_dur) as charge_dur
    from
        (
            select * from zyhh_serial_pickup_dur
            union all
            select * from order_dur
        ) a
    group by dt, passenger_id, type, is_serial
),

------------------------- 4 个 dt-pid 维度的明细子表 -------------------------
-- (b) 冒泡
b_bubble as (
    select dt, pid, sum(wyc_bubble_cnt) as wyc_bubble_cnt
    from wyc_bubble
    group by dt, pid
),
-- (c) 呼叫
c_call as (
    select dt, pid, sum(wyc_call_cnt) as wyc_call_cnt
    from wyc_call
    group by dt, pid
),
-- (c2) 去重呼叫（剔除重发单子单后）
c_call_dedup as (
    select dt, pid, sum(wyc_call_dedup_cnt) as wyc_call_dedup_cnt
    from wyc_call_dedup
    group by dt, pid
),
-- (d) 订单粒度（来自中间表的发单当天指标）
d_order as (
    select
        substr(call_time, 1, 10) as dt,
        passenger_id as pid,
        count(distinct case when is_td_answer = 1 then order_id end) as answer_cnt,
        count(distinct case when is_td_answer_without_reassign = 1 then order_id end) as answer_without_reassign_cnt,
        count(distinct case when is_td_finish = 1 then order_id end) as finish_cnt,
        count(distinct case when is_grab_before_cannel = 1 then order_id end) as hxz_grab_before_cannel_cnt,
        count(distinct case when is_grab_after_dri_cancel = 1 then order_id end) as dri_cancel_cnt,
        count(distinct case when is_grab_after_dri_cancel_without_reassign = 1 then order_id end) as dri_cancel_without_reassign_cnt,
        count(distinct case when is_grab_after_pas_cancel = 1 then order_id end) as pas_cancel_cnt,
        count(distinct case when is_grab_after_pas_cancel_without_reassign = 1 then order_id end) as pas_cancel_without_reassign_cnt,
        sum(case when is_td_answer = 1 then est_arrive_dis_funnel end) / 1000 as est_arrive_dis,
        sum(case when is_td_answer = 1 then est_arrive_dur_funnel end) / 3600 as est_arrive_dur,
        count(distinct case when cancel_reason in ('派单太远') then order_id end) as cancel_reason_lwty,
        count(distinct case when cancel_reason in ('接我路程太堵') then order_id end) as cancel_reason_lctd
    from order_rank
    group by substr(call_time, 1, 10), passenger_id
),
-- (e) 归一原始订单粒度（按 ETP 口径）
e_origin as (
    select
        dt,
        passenger_id as pid,
        count(distinct origin_order_id) as origin_order_cnt,
        count(distinct case when is_td_answer = 1 then origin_order_id end) as origin_answer_cnt,
        count(distinct case when is_last_finish = 1 then origin_order_id end) as origin_last_finish_cnt,
        -- 含改 司机取消（与 中间表 一致：含改 = 改派 OR 无改派）
        count(distinct case when order_status_reassign_rank_desc in ('司机取消无改派', '司机取消改派') then origin_order_id end) as origin_last_dri_cancel_cnt,
        -- 不含改 乘客取消
        count(distinct case when order_status_reassign_rank_desc in ('乘客取消无改派', '有过应答且最后一单应答前乘客取消') then origin_order_id end) as origin_last_pas_cancel_without_reassign_cnt,
        -- 客服取消（ETP 增量）
        count(distinct case when order_status_reassign_rank_desc in ('客服取消无改派', '客服取消改派') then origin_order_id end) as origin_last_kefu_cancel_cnt,
        -- 等待时长 - (中间表) 口径：首次应答 → 最后上车 / 取消
        sum(wait_cancel_dur) / 3600 as wait_cancel_dur,
        -- 时长（ETP 增量）：呼叫 → 首次应答 / 计费 / 到达
        sum(call_to_firstans_dur)    as call_to_firstans_dur_sum,
        sum(call_to_firstcharge_dur) as call_to_firstcharge_dur_sum,
        sum(call_to_firstarrive_dur) as call_to_firstarrive_dur_sum
    from origin_order_detail
    group by dt, passenger_id
),
-- (f) 司机接驾 / 送驾时长（按 dt-pid 汇总）
f_pickup as (
    select
        dt, passenger_id as pid,
        sum(pickup_dur) as pickup_dur_sum,
        sum(charge_dur) as charge_dur_sum
    from all_order_dur
    group by dt, passenger_id
),

-- 实验样本骨架：先收敛到唯一的 pid + dt，再连接所有指标表。
-- 保留最早进组日及之后的样本日期，且不依赖任一行为表是否有数据。
sample_pid_dt as (
    select distinct
        e.dt,
        e.pid,
        e.exp_group
    from exp_group_clean e
    inner join (
        select
            pid,
            exp_group,
            min(dt) as earliest_group_dt
        from exp_group_clean
        group by pid, exp_group
    ) f
      on e.pid = f.pid
     and e.exp_group = f.exp_group
    where e.dt >= f.earliest_group_dt
),

------------------------- 评估宽表（exp_group × dt-pid） -------------------------
-- sample_pid_dt 为最左表；所有指标表均独立按 pid + dt LEFT JOIN。
-- 不再使用 b_bubble.dt 作为其他指标的日期锚点。
raw_pack_ab as (
    select
        a.exp_group,
        count(distinct a.pid) as `分流乘客数`,

        -- ============ 漏斗 / 订单 (来自 中间表) ============
        sum(nvl(wyc_bubble_cnt, 0)) / count(distinct a.pid) as `去重冒泡数`,
        sum(nvl(wyc_call_cnt, 0))   / count(distinct a.pid) as `呼叫数`,
        sum(nvl(wyc_call_dedup_cnt, 0)) / count(distinct a.pid) as `去重呼叫数`,
        sum(nvl(answer_cnt, 0))     / count(distinct a.pid) as `含改应答数`,
        sum(nvl(answer_without_reassign_cnt, 0)) / count(distinct a.pid) as `不含改应答数`,
        sum(nvl(finish_cnt, 0))     / count(distinct a.pid) as `完单数`,
        sum(nvl(hxz_grab_before_cannel_cnt, 0)) / count(distinct a.pid) as `应答前乘客取消量`,
        sum(nvl(dri_cancel_cnt, 0)) / count(distinct a.pid) as `含改司机取消数`,
        sum(nvl(dri_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `不含改司机取消数`,
        sum(nvl(pas_cancel_cnt, 0)) / count(distinct a.pid) as `含改乘客取消数`,
        sum(nvl(pas_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `不含改乘客取消数`,
        sum(nvl(est_arrive_dis, 0)) / count(distinct a.pid) as `预估接驾里程`,
        sum(nvl(est_arrive_dur, 0)) / count(distinct a.pid) as `预估接驾时长`,
        -- sum(nvl(cancel_reason_lwty, 0)) / count(distinct a.pid) as `取消原因-离我太远`,
        -- sum(nvl(cancel_reason_lctd, 0)) / count(distinct a.pid) as `取消原因-路程太堵`,

        -- ============ 归一订单指标（ETP 口径） ============
        sum(nvl(origin_order_cnt, 0))      / count(distinct a.pid) as `原始订单数`,
        sum(nvl(origin_answer_cnt, 0))      / count(distinct a.pid) as `应答原始订单数`,
        sum(nvl(origin_last_finish_cnt, 0)) / count(distinct a.pid) as `最终完单原始订单数`,
        sum(nvl(origin_last_dri_cancel_cnt, 0)) / count(distinct a.pid) as `最终含改司机取消原始订单数`,
        sum(nvl(origin_last_pas_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `最终乘客取消原始订单数`,
        sum(nvl(origin_last_kefu_cancel_cnt, 0)) / count(distinct a.pid) as `最终客服取消原始订单数`,
        -- sum(nvl(wait_cancel_dur, 0)) / count(distinct a.pid) as `乘客原始订单总等待时长`,
        sum(nvl(call_to_firstans_dur_sum, 0))    / count(distinct a.pid) as `呼叫-首次应答时长`,
        sum(nvl(call_to_firstcharge_dur_sum, 0)) / count(distinct a.pid) as `呼叫-计费时长`,
        sum(nvl(call_to_firstarrive_dur_sum, 0)) / count(distinct a.pid) as `呼叫-到达时长`,

        -- ============ 司机接驾 / 送驾时长 ============
        sum(nvl(pickup_dur_sum, 0)) / count(distinct a.pid) as `接驾时间`,
        sum(nvl(charge_dur_sum, 0)) / count(distinct a.pid) as `送驾时间`,

        -------------------------- var (方差) --------------------------
        var_pop(coalesce(wyc_bubble_cnt, 0)) / count(distinct a.pid) as `var_去重冒泡数`,
        var_pop(coalesce(wyc_call_cnt, 0))   / count(distinct a.pid) as `var_呼叫数`,
        var_pop(coalesce(wyc_call_dedup_cnt, 0)) / count(distinct a.pid) as `var_去重呼叫数`,
        var_pop(coalesce(answer_cnt, 0))     / count(distinct a.pid) as `var_含改应答数`,
        var_pop(coalesce(answer_without_reassign_cnt, 0)) / count(distinct a.pid) as `var_不含改应答数`,
        var_pop(coalesce(finish_cnt, 0))     / count(distinct a.pid) as `var_完单数`,
        var_pop(coalesce(hxz_grab_before_cannel_cnt, 0)) / count(distinct a.pid) as `var_应答前乘客取消量`,
        var_pop(coalesce(dri_cancel_cnt, 0)) / count(distinct a.pid) as `var_含改司机取消数`,
        var_pop(coalesce(dri_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `var_不含改司机取消数`,
        var_pop(coalesce(pas_cancel_cnt, 0)) / count(distinct a.pid) as `var_含改乘客取消数`,
        var_pop(coalesce(pas_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `var_不含改乘客取消数`,
        var_pop(coalesce(est_arrive_dis, 0)) / count(distinct a.pid) as `var_预估接驾里程`,
        var_pop(coalesce(est_arrive_dur, 0)) / count(distinct a.pid) as `var_预估接驾时长`,
        -- var_pop(coalesce(cancel_reason_lwty, 0)) / count(distinct a.pid) as `var_取消原因-离我太远`,
        -- var_pop(coalesce(cancel_reason_lctd, 0)) / count(distinct a.pid) as `var_取消原因-路程太堵`,

        var_pop(coalesce(origin_order_cnt, 0))      / count(distinct a.pid) as `var_原始订单数`,
        var_pop(coalesce(origin_answer_cnt, 0))      / count(distinct a.pid) as `var_应答原始订单数`,
        var_pop(coalesce(origin_last_finish_cnt, 0)) / count(distinct a.pid) as `var_最终完单原始订单数`,
        var_pop(coalesce(origin_last_dri_cancel_cnt, 0)) / count(distinct a.pid) as `var_最终含改司机取消原始订单数`,
        var_pop(coalesce(origin_last_pas_cancel_without_reassign_cnt, 0)) / count(distinct a.pid) as `var_最终乘客取消原始订单数`,
        var_pop(coalesce(origin_last_kefu_cancel_cnt, 0)) / count(distinct a.pid) as `var_最终客服取消原始订单数`,
        -- var_pop(coalesce(wait_cancel_dur, 0))             / count(distinct a.pid) as `var_乘客原始订单总等待时长`,
        var_pop(coalesce(call_to_firstans_dur_sum, 0))    / count(distinct a.pid) as `var_呼叫-首次应答时长`,
        var_pop(coalesce(call_to_firstcharge_dur_sum, 0)) / count(distinct a.pid) as `var_呼叫-计费时长`,
        var_pop(coalesce(call_to_firstarrive_dur_sum, 0)) / count(distinct a.pid) as `var_呼叫-到达时长`,

        var_pop(coalesce(pickup_dur_sum, 0)) / count(distinct a.pid) as `var_接驾时间`,
        var_pop(coalesce(charge_dur_sum, 0)) / count(distinct a.pid) as `var_送驾时间`
        
    from sample_pid_dt a
    left join b_bubble b
      on a.pid = b.pid and a.dt = b.dt
    left join c_call c
      on a.pid = c.pid and a.dt = c.dt
    left join c_call_dedup c2
      on a.pid = c2.pid and a.dt = c2.dt
    left join d_order d
      on a.pid = d.pid and a.dt = d.dt
    left join e_origin e
      on a.pid = e.pid and a.dt = e.dt
    left join f_pickup f
      on a.pid = f.pid and a.dt = f.dt
    group by a.exp_group
)

select
    '${ab_tag}' as AB_time,
    concat(
        substr('${start_dt}', 6, 2),
        substr('${start_dt}', 9, 2),
        '-',
        substr('${end_dt}', 6, 2),
        substr('${end_dt}', 9, 2)
    ) as time,
    *
from raw_pack_ab
```
