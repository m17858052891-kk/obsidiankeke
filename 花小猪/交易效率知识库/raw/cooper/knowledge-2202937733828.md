## 背景

验证添加司机实时特征后pk模型效果的提升。

## 特征构建

司机取消率模型，考虑在现有by天特征的基础上增加一批by小时的特征：过去一小时的1h特征和今天截止当前的td特征。

新增13个特征包括：应答后司机取消订单数（含改派）、应答后乘客取消订单数、完单数、应答后司机取消订单数、计费时长、接驾时长、接驾距离，以及分是否连环派的完单数、拒单数、应答数、改派数、司机取消、乘客取消。

司机实时特征取数sql：对应by day特征的取数口径去底表取数，并按半小时聚合

```SQL
with dri_fea as(
    select  
        dt
        ,driver_id
        ,hour(answer_time) as answer_hour
        ,from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss') as half_hour

        ,sum(charge_dur_1d_1004) as charge_dur_1h	--finish_dur_7d --计费时长
        ,sum(aft_ansr_dri_cl_cnt_1d_1007) as aft_ansr_dri_cl_cnt_1h	--total_cancel_dri_cnt_7d 应答后司机取消订单数
        ,sum(finish_cnt_1d_1028) as finish_cnt_1h --finish_cnt_1d --实时单完单次数
        ,sum(arrive_dur_1d_1005) as arrive_dur_1h --arrive_dur_1d --实时单实际接驾时长
        ,sum(est_arrive_dis_1d_1002) as est_arrive_dis_1h --arrive_dis_1d --实时单接驾距离 --???
        ,sum(aft_ansr_cl_dri_num_1d_1005) as aft_ansr_cl_dri_num_1h --应答后司机取消订单数（包含改派单）
        ,sum(aft_ansr_pas_cl_cnt_1d_1007) as aft_ansr_pas_cl_cnt_1h --应答后乘客取消订单数

    from (
        select  dt
                ,order_id
                ,answer_time
                ,driver_id
                ,charge_dur_1d_1004	--计费时长(单位：s)
                ,aft_ansr_dri_cl_cnt_1d_1007	--应答后司机取消订单数
                ,finish_cnt_1d_1028	--完单订单数
                ,arrive_dur_1d_1005 --实际接驾时长
                ,est_arrive_dis_1d_1002 --预估接驾距离(单位：米)
                ,aft_ansr_cl_dri_num_1d_1005 --应答后司机取消订单数（包含改派单）
                ,aft_ansr_pas_cl_cnt_1d_1007 --应答后乘客取消订单数
        from    kflower_dw.dwm_kf_trd_order_index_di
        where   dt between '2024-08-05' and '2024-08-18'
    ) a
    group by dt, driver_id, hour(answer_time)
    , from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss')
)

--kflower_strategy.pk_hxz_dri_2：取消和拒单特征
,serial_cancel as(
    select 
        a.driver_id as driver_id
        ,a.dt as dt --add
        ,answer_hour --add
        ,half_hour
        -- 连环派场景的司机取消特征
        ,sum(if( cancel_type=3 and is_serial_assign = 1,is_td_cancel , 0)) as serial_total_cancel_dri_cnt_1h --serial_total_cancel_dri_cnt_1d
        -- 连环派场景的乘客取消特征
        ,sum(if( cancel_type=2 and is_serial_assign = 1,is_td_cancel , 0)) as serial_total_cancel_pas_cnt_1h --serial_total_cancel_pas_cnt_1d
        -- 非连环派场景的司机取消特征，司机判责加权
        ,sum(if( cancel_type=3 and is_serial_assign <> 1 ,coalesce(dri_duty,1)*is_td_cancel , 0)) as ord_total_cancel_dri_cnt_1h --ord_total_cancel_dri_cnt_1d
        -- 非连环派场景的乘客取消特征，去除扩播与连环派
        ,sum(if( cancel_type=2 and is_serial_assign <> 1 and driver_start_dis <= 3000,is_td_cancel, 0)) as ord_total_cancel_pas_cnt_1h --ord_total_cancel_pas_cnt_1d
    from 
    (
        select 
            dt,
            hour(answer_time) as answer_hour, --add
            from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss') as half_hour,
            cancel_type,
            is_serial_assign,
            is_td_cancel,
            driver_start_dis,
            order_id,
            driver_id
        from    
            kflower_dw.dwd_kf_trd_order_base_di
        where 
            dt between '2024-08-05' and '2024-08-18'
    )a left join 
    (
        select 
            driver_id,
            order_id,
            case when dri_result_name in ("错误有责") then 1.5 
                when dri_result_name in ("恶意有责") then 2 
                else 1 end as dri_duty,
            dt
        from 
            dd_cdm.dwm_trip_crs_ihap_kf_case_1d
        where 
            dt between '2024-08-05' and '2024-08-18'
            -- and city_id=29
    )b on a.dt = b.dt and a.order_id = b.order_id and a.driver_id =b.driver_id
    group by a.driver_id,a.dt,answer_hour,half_hour
)

,serial_refuse as(
    select 
        driver_id
        ,dt
        ,answer_hour
        ,half_hour
        -- 连环派场景的司机拒单特征
        ,sum(if( is_serial_assign=1,is_refuse, 0)) as serial_total_refuse_cnt_1h --serial_total_refuse_cnt_1d
        -- 非连环派场景的司机拒单特征
        ,sum(if( is_serial_assign<>1,is_refuse, 0)) as ord_total_refuse_cnt_1h --ord_total_refuse_cnt_1d
    from 
    (
        select
            distinct 
            a.dt as dt
            ,answer_hour
            ,half_hour
            ,a.order_id as order_id
            ,a.driver_id as driver_id
            ,is_serial_assign
            ,case when a.driver_id = c.driver_id or a.driver_id = d.didi_driver_id then 0 else 1 end as is_refuse
        from
            (
                select
                    shiftleft(shiftright(cast(orderid as bigint), 46)&262143, 32)+(cast(orderid as bigint)&4294967295) as order_id --高位转低位
                    -- orderid as order_id
                    ,driverid as driver_id
                    ,round_num as round_num
                    ,concat_ws('-',year,month,day) as dt
                    ,row_number() over (partition by orderid, round_num order by `timestamp`) as row_number --pk胜出时间
                from kflower_ods.ods_log_kflower_g_order_dusepk_auction
                where concat_ws('-',year,month,day) between '2024-08-05' and '2024-08-18'
                    and action = 'auction_time'
                    and process_order_status_ok = 'true'
                    and process_winner_ok = 'true'	--过滤异常情况
            ) a
            join 
            (
                select  cast(cityid as bigint) as city_id
                        ,cast(channelid as bigint) as channel_id
                        ,cast(driverid as bigint) as driver_id
                        ,shiftleft(shiftright(cast(orderid as bigint), 46)&262143, 32)+(cast(orderid as bigint)&4294967295) as order_id --高位转低位
                        -- ,orderid as order_id
                        ,round_num as round_num
                        ,concat_ws('-', year, month, day) as dt
                        ,if(cast(serial_type as bigint) != 0 , 1, 0) as is_serial_assign
                from kflower_ods.ods_log_kflower_g_order_strivetender
                where   concat_ws('-', year, month, day) between '2024-08-05' and '2024-08-18'
                    -- and cityid=29
                    and ((pktype = 15 and cpid = 0 and kfopenapiorderid = 0)
                        or  (pktype = 15 and cpid != 0)
                        or  (pktype = 17))
            ) b on a.dt = b.dt and a.order_id = b.order_id and a.driver_id = b.driver_id and a.round_num = b.round_num
            inner join 
            (
                select
                    product_id,
                    driver_id,
                    order_id,
                    dt,
                    bubble_trace_id,
                    is_td_answer,
                    is_broadcast,
                    is_td_finish,
                    is_td_reassign,
                    start_city_id as city_id,
                    cancel_type,
                    Hour(answer_time) as answer_hour -- add
                    ,from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss') as half_hour
                from
                    kflower_dw.dwd_kf_trd_order_base_di
                where dt between '2024-08-05' and '2024-08-18'
                    and is_td_call = 1
                    and product_1level not in (3, 4)
                    and nvl(is_short_book_flag, 0) = 0  
                    and (is_td_finish = 1 or is_td_cancel = 1 or is_td_reassign = 1)
                    and driver_id is not null 
            ) c on a.order_id = c.order_id and a.dt = c.dt 
            left join 
            (
                select *
                from kflower_strategy.pk_hxz_driver_id
                where concat_ws('-',year,month,day) between '2024-08-05' and '2024-08-18'
            ) d on b.driver_id = d.hxz_driver_id
        where
            row_number = 1
    )t
    group by driver_id,t.dt,answer_hour,half_hour
)

--kflower_strategy.pk_hxz_dri_3：改派，应答，完单特征
,dri_reassign as (
    select 
        driver_id
        ,dt --add
        ,hour(answer_time) as answer_hour --add
        ,from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss') as half_hour
        -- 连环派场景的司机改派特征
        ,sum(if( is_serial_assign = 1,is_td_reassign , 0)) as serial_reassign_dri_cnt_1h --serial_reassign_dri_cnt_1d
        -- 非连环派场景的司机改派特征
        ,sum(if( is_serial_assign <> 1,is_td_reassign , 0)) as ord_reassign_dri_cnt_1h --ord_reassign_dri_cnt_1d
        -- 连环派场景的司机应答特征
        ,sum(if( is_serial_assign = 1,is_td_answer , 0)) as serial_answer_dri_cnt_1h --serial_answer_dri_cnt_1d
        -- 非连环派场景的司机应答特征
        ,sum(if( is_serial_assign <> 1,is_td_answer , 0)) as ord_answer_dri_cnt_1h --ord_answer_dri_cnt_1d
        -- 连环派场景的司机完单特征
        ,sum(if( is_serial_assign = 1,is_td_finish , 0)) as serial_finish_dri_cnt_1h --serial_finish_dri_cnt_1d
        -- 非连环派场景的司机完单特征
        ,sum(if( is_serial_assign <> 1,is_td_finish , 0)) as ord_finish_dri_cnt_1h --ord_finish_dri_cnt_1d
    from  
        kflower_dw.dwd_kf_trd_order_base_di
    where 
        dt between '2024-08-05' and '2024-08-18'
        -- and start_city_id = 29
        -- and hour(answer_time)=10
        and product_1level <> 4
    group by 
        driver_id, dt, hour(answer_time),from_unixtime(FLOOR(UNIX_TIMESTAMP(answer_time)/(30*60))*30*60, 'HH:mm:ss')
)

select 

    dri_fea.dt as dt
    ,dri_fea.driver_id as driver_id 
    ,dri_fea.answer_hour as answer_hour 
    ,dri_fea.half_hour as half_hour

    --kflower_strategy.pk_hxz_dri
    ,charge_dur_1h	--finish_dur_7d --计费时长
    ,aft_ansr_dri_cl_cnt_1h	--total_cancel_dri_cnt_7d 应答后司机取消订单数
    ,finish_cnt_1h --finish_cnt_1d --实时单完单次数
    ,arrive_dur_1h --arrive_dur_1d --实时单实际接驾时长
    ,est_arrive_dis_1h --arrive_dis_1d --实时单接驾距离 --???
    ,aft_ansr_cl_dri_num_1h --应答后司机取消订单数（包含改派单）
    ,aft_ansr_pas_cl_cnt_1h --应答后乘客取消订单数
    -- ,refuse_cnt_1h

    --kflower_strategy.pk_hxz_dri_2：取消和拒单特征
    ,serial_total_cancel_dri_cnt_1h -- 连环派场景的司机取消订单数
    ,serial_total_cancel_pas_cnt_1h -- 连环派场景的乘客取消订单数
    ,ord_total_cancel_dri_cnt_1h -- 非连环派场景的司机取消特征，司机判责加权
    ,ord_total_cancel_pas_cnt_1h -- 非连环派场景的乘客取消特征，去除扩播与连环派

    ,serial_total_refuse_cnt_1h  -- 连环派场景的司机拒单数
    ,ord_total_refuse_cnt_1h -- 非连环派场景的司机拒单数

    --kflower_strategy.pk_hxz_dri_3：改派，应答，完单特征
    ,serial_reassign_dri_cnt_1h -- 连环派场景的司机改派数--？
    ,ord_reassign_dri_cnt_1h -- 非连环派场景的司机改派数--？
    ,serial_answer_dri_cnt_1h -- 连环派场景的司机应答数
    ,ord_answer_dri_cnt_1h -- 非连环派场景的司机应答数
    ,serial_finish_dri_cnt_1h -- 连环派场景的司机完单数
    ,ord_finish_dri_cnt_1h -- 非连环派场景的司机完单数

from dri_fea
left join serial_cancel on dri_fea.driver_id=serial_cancel.driver_id and dri_fea.dt=serial_cancel.dt and dri_fea.answer_hour = serial_cancel.answer_hour and dri_fea.half_hour = serial_cancel.half_hour
left join serial_refuse on dri_fea.driver_id=serial_refuse.driver_id and dri_fea.dt=serial_refuse.dt and dri_fea.answer_hour = serial_refuse.answer_hour and dri_fea.half_hour = serial_refuse.half_hour
left join dri_reassign on dri_fea.driver_id=dri_reassign.driver_id and dri_fea.dt=dri_reassign.dt and dri_fea.answer_hour = dri_reassign.answer_hour and dri_fea.half_hour = dri_reassign.half_hour

;
```

测试集（3天数据）：取2024-08-24到2024-08-26的应答订单数据（取自pk_fea_hxz_mmoe），取司机特征表中2024-08-23到2024-08-25的by day特征，构造2024-08-24到2024-08-26的实时特征。

## 模型效果比较

#### （0）原始模型（不使用实时特征）：

auc 0.7590

#### （1）使用13个自建实时特征
