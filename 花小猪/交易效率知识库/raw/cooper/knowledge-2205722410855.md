# 二、模型在离线一致性

### 验数逻辑

根据1024数据进行在离线一致性校验

用落模型日志的oid left join pk模型日志表中的oid，比较特征diff

```SQL
with base as(select 
-- b.time_stamp as time_stamp,
cp_id,
city,
call_hour,
serial_type,
is_only_hxz,
is_from_didi,
is_honghu_driver,

order_waiting_period,
lumian_dis,
eta,
left_lumian_distance,
left_lumian_eta,
dropoff_order_distance,
dropoff_order_eta,
start_dest_distance,
est_start_dest_dur,
pre_total_fee,

cast(param['cp_id'] as int) as o_cp_id,
cast(param['city'] as int) as o_city,
cast(param['call_hour'] as int) as o_call_hour,
cast(param['serial_type'] as int) as o_serial_type,
cast(param['is_only_hxz'] as int) as o_is_only_hxz,
cast(param['is_from_didi'] as int) as o_is_from_didi,
cast(param['is_honghu_driver'] as int) as o_is_honghu_driver,

cast(param['order_waiting_period'] as float) as o_order_waiting_period,
cast(param['lumian_dis'] as float) as o_lumian_dis,
cast(param['eta'] as float) as o_eta,
cast(param['left_lumian_distance'] as float) as o_left_lumian_distance,
cast(param['left_lumian_eta'] as float) as o_left_lumian_eta,
cast(param['dropoff_order_distance'] as float) as o_dropoff_order_distance,
cast(param['dropoff_order_eta'] as float) as o_dropoff_order_eta,
cast(param['start_dest_distance'] as float) as o_start_dest_distance,
cast(param['est_start_dest_dur'] as float) as o_est_start_dest_dur,
cast(param['pre_total_fee'] as float) as o_pre_total_fee

from
(select 
concat_ws('-',year,month,day) as dt,
param,
param['traceid'] as trace_id,
-- param['order_id'] as order_id
shiftleft(shiftright(cast(param['order_id'] as bigint), 46)&262143, 32)+(cast(param['order_id'] as bigint)&4294967295) as order_id	--高位转低位
from kflower_strategy.ods_log_kflower_ether_blindbox_h
where concat_ws('-',year,month,day) between '2025-10-23' and '2025-10-23'
-- and hour > 17
)h
join 
(
    select 
        trace_id,
        -- oid
        shiftleft(shiftright(cast(oid as bigint), 46)&262143, 32)+(cast(oid as bigint)&4294967295) as oid	--高位转低位

    from kflower_marketplace.ods_log_kflower_athena_guide_answer_after_dialogue_rec_h
    where concat_ws('-',year,month,day) between '2025-10-23' and '2025-10-23'
    -- and param['first_access'] = true
)c on h.order_id = c.oid
left join(
    select distinct 
        *,
        if (class in ('乘客取消'), 1, 0) as is_p_cancel
    from
        kflower_strategy.pk_hxz_modeling_data
    where
        dt between '2025-10-23' and '2025-10-23'
        and is_finish is not null
        and class in ('完单', '乘客取消')
)a on h.order_id = a.order_id

)




select 

cp_id - o_cp_id as delta_cp_id,
city - o_city as delta_city,

order_waiting_period - o_order_waiting_period as delta_order_waiting_period,
lumian_dis - o_lumian_dis as delta_lumian_dis,
eta - o_eta as delta_eta,
left_lumian_distance - o_left_lumian_distance as delta_left_lumian_distance,
left_lumian_eta - o_left_lumian_eta as delta_left_lumian_eta,
dropoff_order_distance - o_dropoff_order_distance as delta_dropoff_order_distance,
dropoff_order_eta - o_dropoff_order_eta as delta_dropoff_order_eta,


base.*


from base

limit 10000
```
