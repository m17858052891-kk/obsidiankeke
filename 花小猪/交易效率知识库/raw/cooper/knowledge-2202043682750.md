# 背景：

通过数据分析发现部分司机通过刷单来骗取B补。

e.g 时长保底活动， 在此活动中发现司机会通过发送超短单（发单->完单<2分钟）来获取活动的奖励。

- 活动时段内全部订单都满足现状作弊特征的司机数 **21司机**（非去重）

- 活动时段内至少有一单满足现状作弊特征的司机数 **132司机**（非去重）

# 预估收益： 

- 降低 B补亏损 (总B补包括平台，品牌B补) **196873人民币/月（0.15%）** (3月数据) 总B补（129808270人民币）

- 预计提升活动的ROI和效率

- 预计解放运力

# 方案

为了避免司机作弊，我们可在分单系统中过滤掉这些DO配对。

黑名单的规则

- 30天内 ： 司机和乘客完成两次超短单（发单->完单<2分钟）会被判定为同伙，加入黑名单

# 估算：

## 收益

**3月份 B补亏损 ： 196873人民币**

预估方法：  
B补亏损 = 完成2次超短订单司机B补 - 相似完单数的司机（除去司机作弊订单） 

```PlainText
with base as (
        select
            driver_id,
            cancel_type,
            is_td_finish,
            order_id,
            hour(answer_time) as hour_,
            d_total_fee_amt,
            p_total_fee_amt,
            start_city_id as cityid,
            call_time,
            arrive_time,
            passenger_id,
            actual_cost_capital,
            coupon_cost,
            liushui as driver_income,
            (
                3600 * hour(finish_time) + 60 * minute(finish_time) + second(finish_time)
            ) - (
                3600 * hour(call_time) + 60 * minute(call_time) + second(call_time)
            ) + 24 * 3600 * datediff(finish_time, call_time) as dur
        from
            kflower_dw.dwd_hh_trd_order_base_di a
            left join (
                select
                    order_id_hh,
                    actual_cost_capital,
                    coupon_cost
                from
                    kflower_dw.dwm_hh_fin_bcr_passenger_receipts_di
                where
                    dt between "2024-03-01"
                    and "2024-03-31"
            ) b on a.order_id = b.order_id_hh
            left join (
                select
                    order_id as order_id_hh,
                    driver_id as driver_id_hh,
                    sum(
                        case
                            when trade_type in (1, 6, 162) then pay_fee
                            else 0
                        end
                    ) - sum(
                        case
                            when trade_type in (7) then pay_fee
                            else 0
                        end
                    ) as liushui
                from
                    g_caiwu.dwd_honghu_tr_driver_payable_receipt_di
                where
                    dt between '2024-03-01'
                    and '2024-03-31'
                    and order_area in (3, 24, 86)
                    and trade_type in (1, 6, 162, 7)
                group by
                    order_id,
                    driver_id
            ) c on a.driver_id = c.driver_id_hh and a.order_id = c.order_id_hh
        where
            dt between "2024-03-01"
            and "2024-03-31"
            and is_td_finish = 1
            and product_1level <> 4 
            and is_keqi_company = 0 
    ),
    subsidy_b_zy as (
        select
            dt,
            driver_id,
            order_id,
            trade_type_string,
            coalesce(subsidy_b, 0) as subsidy_b_zy_platform,
            action_type
        from
            kflower_dw.dwm_kf_fin_order_subsidy_b_di
        where
            dt between '2024-03-02'
            and '2024-04-01'
            and product_category not in (4)
    ),
    subsidy_b_hh as (
        select
            dt,
            driver_id_honghu as driver_id,
            trade_type,
            order_id_honghu,
            coalesce(subsidy_b, 0) as subsidy_b,
            coalesce(subsidy_b_hxz, 0) + coalesce(subsidy_b_channel, 0) as subsidy_b_hh_platform,
            action_type
        from
            kflower_dw.dwm_hh_fin_sb_order_subsidy_b_di
        where
            dt between '2024-03-02'
            and '2024-04-01'
    ),
    subsidy_b_tab as (
        select
            case
                when a.dt is not null then a.dt
                else b.dt
            end dt,
            case
                when a.driver_id is not null then a.driver_id
                else b.driver_id
            end driver_id,
            coalesce(subsidy_b,0) as subsidy_b,
            coalesce(subsidy_b_hh_platform,0) as subsidy_b_hh_platform,
            coalesce(subsidy_b_zy_platform,0) as subsidy_b_zy_platform,
            a.action_type,
            b.action_type
        from
            subsidy_b_zy a full
            outer join subsidy_b_hh b on a.driver_id = b.driver_id
    ),
    subsity_tab as(
        select
            driver_id,
            sum(subsidy_b) as tot_subsidy_b,
            sum(subsidy_b_hh_platform) as tot_subsidy_b_hh_platform,
            sum(subsidy_b_zy_platform) as tot_subsidy_b_zy_platform
        from
            subsidy_b_tab
        group by
            driver_id
    ),
    collision as (
        select
            passenger_id,
            driver_id,
            count(distinct order_id) as meet_cnt,
            case when count(distinct order_id) >= 2 then 1
            else 0 end 
            is_problem_couple,
            count(distinct if (dur <=120, order_id,null)) as really_fast_order_together_cnt
    
    
        from
            base
        group by
            driver_id,
            passenger_id
    )
    ,
    ref  as 
    (
    select *
    from (
    select driver_id, sum(if(really_fast_order_together_cnt>=2, meet_cnt, 0)) as lost_cnt, sum(meet_cnt) as finish_cnt 
    
    from collision 
    group by driver_id
    )  a where  lost_cnt =0
    ),
    
    ref_tab 
    as (select cityid, a.driver_id, count(distinct order_id) as finish_cnt from base a  inner join 
    (select distinct driver_id from ref ) b
    on a.driver_id = b.driver_id 
    group by a.driver_id, cityid
    ),
    
    
    collision_city as (select
    cityid,
            passenger_id,
            driver_id,
            count(distinct order_id) as meet_cnt,
            case when count(distinct order_id) >= 2 then 1
            else 0 end 
            is_problem_couple,
            count(distinct if (dur <=120, order_id,null)) as really_fast_order_together_cnt
    
    
        from
            base
        group by
            driver_id,
            passenger_id,
            cityid
            ),
    
    bad_driver as (
         select *
        from (
        select cityid ,driver_id as bad_driver, sum(if(really_fast_order_together_cnt>=2, meet_cnt, 0)) as lost_cnt,
        sum(meet_cnt) -  sum(if(really_fast_order_together_cnt>=2, meet_cnt, 0)) as real_finish_cnt,
        1 as is_problem_driver
        
        from collision_city 
        group by driver_id, cityid
        )  a 
        where lost_cnt >=1
    
    )

select sum (bad_driver_subsity- ref_subsity)

from (
    select bad_driver, avg(coalesce(ref_subsity,0)) as ref_subsity , avg(bad_driver_subsity) as bad_driver_subsity  from ref_tab inner join  bad_driver on  ref_tab.cityid = bad_driver.cityid and 
    ref_tab.finish_cnt = bad_driver.real_finish_cnt
    left join (select driver_id, tot_subsidy_b   + tot_subsidy_b_zy_platform   
     as ref_subsity from subsity_tab) ref_subsity_tab 
    
    on ref_subsity_tab.driver_id = ref_tab.driver_id 
    left join (select driver_id ,tot_subsidy_b  + tot_subsidy_b_zy_platform   
     as bad_driver_subsity from subsity_tab) bad_driver_subsity_tab 
     on 
    
    bad_driver_subsity_tab.driver_id = bad_driver.bad_driver
    
    group by bad_driver
) tab
```
