            is_td_answer_flag
            order by

                    source_type <> 2,

--全部发单
pbd as (

        )

    --分单时参与订单数  source_type is null 未应答的订单，无driver_id

                source_type is null

            ) then a.order_id
            when a.app_id not in (-1, 1017)
            and source_type = 2 then c.reassign_origin_order_id
            else null
        end
    ) as participate_order_cnt_cp,

        distinct case
            when a.app_id = -1
            and (
                source_type is null
                or source_type <> 2
            ) then a.order_id

                source_type is null
                or source_type <> 2
            ) then b.order_id

            when b.app_id = 1017
            and (
                source_type is null
                or source_type <> 2
            ) then b.order_id
            when b.app_id = 1017
            and source_type = 2 then c.reassign_origin_order_id
            else null
        end

from
    (

            city_id,
            driver_id,
            product_id,
            app_id,
            is_filtered_by_dis, --是否被距离过滤

                    server_time,
                    city_id,
                    driver_id,
                    product_id,
                    app_id,
                    is_filtered_by_dis, --是否被距离过滤

                    case

                        or (
                            substr(cast(order_id as string), -2, 1) in ('0', '1', '2', '3', '4')
                            and substr(cast(order_id as string), -1, 1) in ('9')

            and pk_type = 17

    and unix_timestamp(b.win_time) between unix_timestamp(a.server_time)
    and (
        unix_timestamp(a.server_time) + 13
    )
    left join (
        select
            a.order_id,
            a.driver_id,
            source_type,
            reassign_origin_order_id
        from
            ord_all as a
            join pbd on a.order_id = pbd.order_id
    ) c on a.order_id = c.order_id
    and a.driver_id = c.driver_id
group by
    dt,
    --a.city_id,
    --distance_bucket,
    exp_tag;
