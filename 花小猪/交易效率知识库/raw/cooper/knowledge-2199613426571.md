**无车赔非省钱卡用户实验效果** 展开源码

```PlainText
select
    gp,
    sum(answer_cnt) / sum(order_cnt) as answer_ratio,
    sum(finish_cnt) / sum(order_cnt) as finish_ratio,
    sum(order_cnt) as order_cnt,
    count(passenger_id) as pid_cnt,
    sum(gmv) as gmv,
    sum(subsidy) as subsidy,
    sum(gmv) / count(passenger_id) as gmv_per_pid,
    sum(cost) / count(passenger_id) as cost_per_pid,
    sum(subsidy) / count(passenger_id) as subsidy_per_pid
from(
        select
            base.passenger_id as passenger_id,
            max(gp) as gp,
            count(
                case
                    when is_td_call = 1
                    and source_type <> 2 then base.order_id
                    else null
                end
            ) as order_cnt,
            count(
                case
                    when is_td_call = 1
                    and is_td_answer = 1 then base.order_id
                    else null
                end
            ) as answer_cnt,
            count(
                case
                    when is_td_call = 1
                    and is_td_finish = 1 then base.order_id
                    else null
                end
            ) as finish_cnt,
            count(
                distinct case
                    when is_td_call = 1 then base.passenger_id
                    else null
                end
            ) as pid_cnt,
            sum(gmv_amt) as gmv,
            sum(gmv_amt - actual_amt) as cost
        from
            (
                select
                    distinct order_id,
                    passenger_id,
                    dt,
                    case
                        when apollo_bucket('athena_reply_guarantee_control', passenger_id) % 100 >= 95 then 'treatment'
                        else 'control'
                    end as gp,
                    answer_dur,
                    cancel_dur,
                    cancel_type,
                    is_td_answer,
                    is_td_finish,
                    is_td_call,
                    source_type
                from
                    kflower_dw.dwd_kf_trd_order_base_di
                where
                    dt between '2020-12-25'
                    and '2021-01-12'
                    and is_td_call = 1
                    and call_time is not null
                    and call_time <> '0000-00-00 00:00:00'
            ) base
            left join (
                select
                    distinct userid,
                    cast(userid as bigint) & 281474976710655 as card_pid
                from
                    kflower_dw.dwd_kf_mkt_card_user_kcard_df
                where
                    dt between '2020-12-25'
                    and '2021-01-12'
                    and (
                        to_date(expire_time) >= '2021-01-12' --过期时间>=昨天
                        or to_date(enable_time) <= '2021-01-12'
                    )
            ) card on base.passenger_id = card.card_pid
            left join(
                select
                    distinct order_id,
                    dt as pay_dt,
                    gmv_amt,
                    actual_amt
                from
                    kflower_dw.dwd_kf_trd_order_pay_suc_di
                where
                    dt between '2020-12-25'
                    and '2021-01-12'
                    and call_time is not null
                    and call_time <> '0000-00-00 00:00:00'
            ) pay on base.order_id = pay.order_id
            and base.dt = pay.pay_dt
        where
            userid is null
        group by
            passenger_id
    ) a
    left join (
        select
            passenger_id as pid_b,
            count(order_id) as account_cnt,
            sum(actual_amount) as subsidy
        from
            (
                select
                    order_id,
                    dt,
                    cast(driver_id as bigint) & 281474976710655 as passenger_id,
                    actual_amount
                from
                    kflower_dw.dwd_kf_fin_sb_order_trans_account_di
                where
                    dt between '2020-12-25'
                    and '2021-01-12'
                    and trade_type = 516
                    and actual_amount > 0
            ) tmp
        group by
            passenger_id
    ) b on a.passenger_id = b.pid_b
group by
    gp
```
