# 1. 策略介绍

目标：对分单策略中应答后取消率较高的订单进行去尾处理，优化分单效率

策略思路：

- 构建应答后取消模型，预测do对被应答后取消的概率

- 设置不同的阈值来决定do对被划分为正类或负类

- 过滤掉被预测为正类的do对

阈值设置：遵义0.6，临沂0.7

模型的重要特征：参考[2. 重要特征](http://wiki.intra.xiaojukeji.com/pages/viewpage.action?pageId=364061578#id-%E5%BA%94%E7%AD%94%E5%90%8E%E5%8F%B8%E4%B9%98%E5%8F%96%E6%B6%88%E6%A8%A1%E5%9E%8B-2.%E9%87%8D%E8%A6%81%E7%89%B9%E5%BE%81)

select
    gp,
    count(*),
    avg(busy_dur) as busy_dur_per_pid,
    avg(online_dur) as online_dur_per_pid,
    avg(finish_cnt) as finish_per_pid,
    avg(brt_cnt) as brt_per_pid,
    avg(dri_cancel_cnt/order_cnt) as dri_cancel_per_pid
from
    (
        select
            distinct tmp.driver_id as driver_id,
            cancel_ratio,
            case
                when cancel_ratio >= 0.9 then 0.9
                else int(cancel_ratio * 10) / 10
            end as gp
        from(
                select
                    distinct driver_id,
                    dri_cancel_cnt_30d / finish_cnt_30d as cancel_ratio,
                    --表中只有完单量，没有应答量，故用完单量做分母进行估计
                    dri_cancel_cnt_30d,
                    finish_cnt_30d
                from
                    kflower_marketplace.dwm_trip_dri_statistics_days_total_di
                where
                    dt = '2021-08-01'
                    and finish_cnt_30d >= 5
            ) tmp
            inner join(
                select
                    driver_id,
                    sum(online_dur) as online_dur
                from
                    kflower_dw.dwm_kf_dri_online_relate_wyc_df
                where
                    dt between '2021-07-01'
                    and '2021-08-01' --一个月出车司机
                group by
                    driver_id
            ) dri on tmp.driver_id = dri.driver_id
        where
            online_dur <> 0
    ) base
    left join (
        select
            driver_id,
            count(
                distinct case
                    when is_td_finish = 1 then order_id
                    else null
                end
            ) finish_cnt,
            count(
                distinct case
                    when order_status = 12 then order_id
                    else null
                end
            ) dri_cancel_cnt,
            count(
                distinct case
                    when source_type<>2 then order_id
                    else null
                end
            ) order_cnt
        from
            kflower_dw.dwd_kf_trd_order_base_di
        where
            dt between '2021-08-04'
            and '2021-08-08'
            and is_td_call = 1
            and call_time is not null
            and call_time <> '0000-00-00 00:00:00'
        group by
            driver_id
    ) finish on finish.driver_id = base.driver_id
    left join(
        select
            count(distinct cast(param ['lowoid'] as bigint)) as brt_cnt,
            cast(param ['driverid'] as bigint) as driver_id
        from
            kflower_ods.ods_log_kflower_orderbroadcast
        where
            concat_ws('-', year, month, day) between '2021-08-04'
            and '2021-08-08'
            and cast(param ['driverid'] as bigint) > 0
            and cast(param ['lowoid'] as bigint) > 0
            and cast(param ['DispatchType'] as bigint) in (2)
        group by
            cast(param ['driverid'] as bigint)
    ) brt on brt.driver_id = base.driver_id
    left join (
        select
            driver_id,
            sum(busy_dur) as busy_dur,
            sum(online_dur) as online_dur,
            sum(serve_dur) as serve_dur
        from
            kflower_dw.dwm_kf_dri_online_relate_wyc_df
        where
            dt between '2021-08-04'
            and '2021-08-08'--between  '2021-07-28' and '2021-08-01'
        group by
            driver_id
    ) dri on base.driver_id = dri.driver_id
group by
    gp
order by
    gp
