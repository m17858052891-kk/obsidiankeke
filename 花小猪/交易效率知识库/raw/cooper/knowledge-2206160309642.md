```PlainText
select  jichu.city_id
        ,city_name
        ,dispatch_name
        ,anycar_name
        ,order_origin
        ,is_kuobo
        ,is_grab
        ,is_serial_order
        ,is_realdestination
        ,count(distinct jichu.order_id) as ord_cnt
        ,count(distinct if(refuse.order_id is null, null, jichu.order_id)) as refuse_cnt
        ,count(distinct if(refuse.order_id is null, null, jichu.order_id)) as active_reject_cnt
        ,count(distinct if(refuse.order_id is null, null, jichu.order_id)) as default_reject_cnt
        ,count(distinct if(orders.is_td_answer = 1, jichu.order_id, null)) as answer_cnt
        ,count(distinct if(orders.is_td_finish = 1, jichu.order_id, null)) as finish_cnt
        ,count(distinct if(cancel_type = 2, jichu.order_id, null)) as pas_can_cnt
        ,count(distinct if(cancel_type = 3, jichu.order_id, null)) as dri_can_cnt
        ,count(distinct if(is_td_reassign = 1, jichu.order_id, null)) as reassign_cnt
        ,count(distinct if(strive.order_id is null, null, jichu.order_id)) as active_strive_cnt
        ,count(distinct if(strive.order_id is null, null, jichu.order_id)) as default_strive_cnt
        ,count(distinct handup.order_id) as handup_cnt
        ,count(distinct if(handup.order_id is null, null, winner.order_id)) as winner_cnt
        ,count(distinct if(handup.order_id is not null and refuse.order_id is not null, winner.order_id, null)) as winner_reject_cnt
        ,sum(driver_start_dis) as driver_start_dis
from    (
    select  concat_ws('-', year, month, day) as dt
            ,param ['area'] as city_id
            ,param ['orderid'] as order_id
            ,param ['driverid'] as driver_id
            ,shiftleft(shiftright(cast(param ['hh_order_id']  as bigint), 46)&262143, 32)+(cast(param ['hh_order_id']  as bigint)&4294967295) as hh_order_id
            -- ,case   when param['order_origin'] = 1 then '滴滴'
            --         when param['order_origin'] = 2 then 'PBD'
            --         when param['order_origin'] = 3 then '花小猪'
            --  end as order_origin
            ,case   when param['DispatchType'] = 1 then '抢单'
                    when param['DispatchType'] = 2 then '半指派'
                    when param['DispatchType'] = 3 then '全指派'
             end as dispatch_name
            ,case   when param['level_type'] in (102, 103) then '一口价'
                    when param['level_type'] = 101 then '特快'
                    when param['level_type'] = 104 then '特价'
                    when param['level_type'] = 105 then '特价盒子'
                    when param['level_type'] = 200 then '特快盒子'
             end as anycar_name
            ,param ['anycar_type'] as anycar_type
            ,param ['DispatchType'] as dispatch_type
            ,if(param ['is_serial_order']=1, 1, 0) as is_serial_order
            ,if(param ['realdestination_switch']=1, 1, 0) as is_realdestination
            ,if(param ['auto_grab_flag']=1, 1, 0) as is_grab
            ,if(cast(param ['lumian_dis'] as double) > 3000, 1, 0) as is_kuobo
            ,row_number () over(partition by concat(param ['orderid'], param ['driverid'])                         order by                             param ['timestamp'] desc) as my_index
    from    kflower_marketplace.ods_log_kflower_orderbroadcast_hh
    where   concat_ws('-', year, month, day) = '${BIZ_DATE_LINE}'
      and   param['broad_source'] in ('hh_hxz_stg', 'hh_hxz_speed')
      and   param ['DispatchType'] in (1, 2, 3)
      and   param ['is_online'] = 1
) jichu
inner join (
    select  cast(param ['orderid'] as bigint) as order_id
            ,cast(param ['driverid'] as bigint) as driver_id
            ,case   when param['order_terminal'] = 1 then '花小猪'
                    when param['order_terminal'] = 2 then '滴滴'
                    when param['order_terminal'] = 3 then '百度地图'
                    when param['order_terminal'] = 4 then '腾讯地图'
                    when param['order_terminal'] = 5 then '同程出行'
                    when param['order_terminal'] = 6 then '腾讯出行'
                    when param['order_terminal'] = 7 then '携程'
                    when param['order_terminal'] = 10 then '哈啰'
                    when param['order_terminal'] = 11 then '美团'
             end as order_origin
            ,row_number () over(partition by concat(param ['orderid'], param ['driverid'])                         order by                             param ['timestamp'] desc) as feature_index
    from    kflower_marketplace.ods_log_kflower_stgmatrix_hh
    where   concat_ws('-', year, month, day) = '${BIZ_DATE_LINE}'
      and   param['broad_source'] in ('hh_hxz_stg', 'hh_hxz_speed')
) feature
on      jichu.order_id = feature.order_id
  and   jichu.driver_id = feature.driver_id
left join (
    select  driver_id
            ,order_id
            ,min(is_td_answer) as is_td_answer
            ,min(is_td_finish) as is_td_finish
            ,min(is_td_reassign) as is_td_reassign
            ,min(cancel_type) as cancel_type
            ,min(driver_start_dis) as driver_start_dis	--司机应单时，与发单地的路面距离（米）
    from    kflower_dw.dwd_hh_trd_order_base_di
    where   dt = '${BIZ_DATE_LINE}'
      and   company_id = 1400
      and   is_td_answer = 1
    group by order_id, driver_id
) orders
on      jichu.hh_order_id = orders.order_id
  and   jichu.driver_id = orders.driver_id
left join (
    select  order_id
            ,driver_id
    from    tripcloud_data.ods_log_honghu_g_hh_assign_callback
    where   concat_ws('-', year, month, day) = '${BIZ_DATE_LINE}'
      and   status = 5
) as refuse
on      jichu.hh_order_id = refuse.order_id
  and   jichu.driver_id = refuse.driver_id
left join (
    select  order_id
            ,driver_id
    from    kflower_dw.dwi_hh_dri_strive_order_di
    where   dt = '${BIZ_DATE_LINE}'
) strive
on      jichu.hh_order_id = strive.order_id
  and   jichu.driver_id = strive.driver_id
left join (
    select  distinct lowoid as order_id
            ,driver_id_3rd_part as driver_id
            ,cpid as cp_id
    from    kflower_ods.ods_log_kflower_g_order_strivetender
    where   concat_ws('-', year, month, day) = '${BIZ_DATE_LINE}'
      and   (
                    channelid = 3
                or  channelid = 2
                or  channelid = 1
            )
      and   iscontinue = 'true'
      and   filter_rule_num = 'R0'
) as handup
on      jichu.order_id = handup.order_id
  and   jichu.driver_id = handup.driver_id
left join (
    select  distinct shiftleft(shiftright(cast(orderid as bigint), 46)&262143, 32)+(cast(orderid  as bigint)&4294967295) as order_id
            ,driver_id_3rd_part as driver_id
            ,cpid as cp_id
    from    kflower_ods.ods_log_kflower_g_order_dusepk_auction
    where   concat_ws('-', year, month, day) = '${BIZ_DATE_LINE}'
      and   cast(cityid as bigint) < 1000	--过滤测试情况
      and   process_order_status_ok = 'true'
      and   process_winner_ok = 'true'	--过滤异常情况
      and   action = 'auction_time'
) as winner
on      jichu.order_id = winner.order_id
  and   jichu.driver_id = winner.driver_id
  and   handup.cp_id = winner.cp_id
left join (
    select  city_id
            ,city_name
    from    whole_dw.dim_city
    where   dt = '${BIZ_DATE_LINE}'
) c
on      jichu.city_id = c.city_id
where   my_index = 1
  and   feature_index = 1
group by jichu.city_id, city_name, dispatch_name, anycar_name, order_origin, is_kuobo, is_grab, is_serial_order, is_realdestination
```
