select b.driver_id
, guilt_ratio
, guilt_order_cnt
, order_cnt
, dri_cl_ratio
from
(
    select driver_id
    , count(distinct case when dri_result_name in ('恶意有责', '错误有责') then order_id else null end) / count(distinct order_id) guilt_ratio
    , count(distinct case when dri_result_name in ('恶意有责', '错误有责') then order_id else null end) guilt_order_cnt

) a

(
    select driver_id
    , count(distinct case when cancel_type = 3 or is_td_reassign = 1 then order_id else null end) / count(distinct order_id) dri_cl_ratio
    from kflower_dw.dwd_kf_trd_order_base_di
    where dt between '2024-06-17' and '2024-06-30'
    and is_td_call = 1
    and is_td_answer = 1

    group by driver_id

on a.driver_id = b.driver_id;

# 四、司机判责在pk模型中的应用方法

当前的pk可解释模型中，司机判责结果和司机取消数进行特征组合。可考虑优化的特征工程方法有

- 加工司机判责类特征：统计司机在过往时间窗口内的被判责次数，和取消数并列作为pk模型的备选特征

- 优化判责与取消的特征组合：延用当前的组合方法，同时优化司机判责对司机取消数特征的加重程度

- 用判责对pk模型打分加倾斜：从预测打分层面入手，对过往被判责较多的司机直接赋负向pk打分倾斜
