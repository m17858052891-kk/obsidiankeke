# 订单表的接驾距离排查


排查订单表中的接驾距离是否有问题

订单表：kflower_dw.dwd_kf_trd_unity_order_di 字段： est_arrive_dis（距离） answer_time （时间），用于实验评估sql

与举手表比较： kflower_ods.ods_log_kflower_g_order_strivetender 字段：driver_start_distance（距离） timestamp（时间）

以太原为例，2026-06-10

```PlainText


with
strive as ( --举手表  --最后一轮PK 所有举手司机
    select 
        * 
    from (
        select 
            cast(orderid as bigint) orderid --高位
            ,shiftleft(shiftright(cast(orderid as bigint), 46) & 262143, 32) +(cast(orderid as bigint) & 4294967295) as order_id--高位转低位
            ,cast(driverid as bigint) driver_id
            ,cast(passenger_id as bigint) passenger_id
            ,cast(cityid as bigint) city_id
            ,cast(driver_start_distance as double) driver_start_distance--预估接驾距离，连环派为ab+bc
            ,cast(ether_value as bigint) ether_value --0-1的模型预测值*10000
            ,cast(pkvalue as bigint) pkvalue --最终使用的pk分
            ,ext
            ,case   when channelid = 2 then driver_id_3rd_part
                    when channelid = 3 then driverid
             end as driver_id_new	--channelid 发单渠道 1:滴滴主端 2:PBD主端 3. hxz端
            ,case   when cpid in ("0", "1017", "1373", "1374") then 'zy'	--自营
                    when is_honghu_driver = 'true' then 'hh'	--宏鹄
             end as zy_or_hh
            ,cast(round_num as bigint) round_num--当前订单所处的轮次数，从1开始递增。
            ,cast(cpid as bigint) cp_id 
            ,cast(product_id as bigint) product_id
            ,timestamp
            ,row_number() over(partition by orderid, round_num, driverid order by `timestamp` desc) as timerank
            ,dense_rank() over(partition by orderid order by cast(round_num as bigint) desc) round_rank --按照轮次倒序
            ,concat_ws('-',year,month,day) dt
        from 
            kflower_ods.ods_log_kflower_g_order_strivetender
        where 
            concat_ws('-',year,month,day) = '2026-06-10'
            and (
                    (
                            pktype = 15
                        --从胜者组中选出最终胜者的策略
                        and cpid = 0
                        and kfopenapiorderid = 0
                    --hna侧订单id
                    )
                or  (
                            pktype = 15
                        and cpid != 0
                    )
                or  (
                        pktype = 17
                    )
            )
        and   iscontinue = 'true'	--订单状态是否可抢
        and   filter_rule_num = 'R0'	--举手未被过滤
        and   useethervalue = 'true'	--是否使用模型预测值作为PK参考
        and   channelid not in (1, 2)
        and cityid = 26	
        -- and cityid in (42,157,50,38, 13,33,8,16,19,22,25,26,29,32,34,58,74,81,85,125,138,153,156,160,254)
    )t 
    where round_rank = 1 and timerank = 1
 
),
auction as (--最后一轮PK 胜出司机
    select 
        * 
    from (
    select 
        order_id,
        round_num,
        city_id,
        driver_id,
        dt,
        row_number() over(partition by order_id order by round_num desc) rn_desc
    from 
        kflower_dw.dwd_kf_trd_dusepk_auction_di
    where 
        channel_id = 3
        and dt in ('2026-06-10', '2026-06-24', '2026-07-01')
        and city_id = 26
    )t 
    where rn_desc = 1
),
order_dist as ( --订单表，应答时刻接驾距离
    select 
            distinct
            dt,
            source_id,
            order_id,
            driver_id,
            is_serial_assign,
            passenger_id,
            product_id,
            product_type,
            platform_name,
            -- gmv_amt,
            is_grab_before_cannel,
            is_td_finish,
            est_arrive_dur,
            est_arrive_dis,
            answer_time
        from
            kflower_dw.dwd_kf_trd_unity_order_di
        where
            dt = '2026-06-10'
            and is_hxz = 1
            and start_city_id in (26)
)

select 
    is_serial_assign,
    avg(b.driver_start_distance - c.est_arrive_dis)/avg(c.est_arrive_dis) as dis_gap_relative,
    avg(unix_timestamp(b.timestamp) - unix_timestamp(answer_time))  time_gap

from auction a left join  strive b --最后一轮PK 胜出司机
    on a.order_id = b.order_id and a.round_num = b.round_num and a.driver_id = b.driver_id --胜出司机
inner join order_dist c 
    on a.order_id = c.order_id and a.driver_id = c.driver_id
group by 
    is_serial_assign
```
