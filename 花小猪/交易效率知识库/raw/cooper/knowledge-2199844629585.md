    ) `应答前取消订单，平均hold时长`
from
    (
        select
            city_id,
            city_name,
            call.passenger_id,
            call.order_id,
            call.product_id,
            call.product_name,
            call.call_car_level_id,
            call.call_car_level_name,
            product_2level as answer_product_2level,
            product_2level_name as answer_product_2level_name,
            is_bub_call,
            is_wait_call,
            answer_cnt_1d_1013,
            answer_cnt_1d_1084,
            answer_dur_1d_1006,
            bef_ansr_cl_cnt_1d_1009,
            bf_ansr_cl_dur,
            cal_mode,
            hold_time,
            call.product_category,
            original_hold_time as original_hold_dd_time,
            hold_time as actual_hold_dd_time
        from
            call
            left join answer on call.passenger_id = answer.passenger_id
            and call.order_id = answer.order_id
            left join hold on hold.passenger_id = call.passenger_id
            and hold.order_id = call.order_id
            and hold.product_category = call.product_category
        where
            original_hold_time is not null
            and answer_cnt_1d_1013 + bef_ansr_cl_cnt_1d_1009 > 0
    ) t
group by    
    cal_mode;
