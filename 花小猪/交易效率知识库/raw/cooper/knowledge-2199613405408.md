**从播单到应答的重要漏斗指标** 展开源码

```PlainText
select
    yymmdd,
    count(distinct serial_low_id) as serial_order_cnt, --播单需要含改派, 因为有些改派的订单的原单没有参与连环派单
    count(
        distinct if(
            source_type <> 2 
            and cancel_type = 1 
            and is_td_cancel = 1, 
            jichu.order_id, 
            null
            )
        ) as pas_before_can_cnt, -- 应答前乘客取消
    count(
        distinct if(
            source_type <> 2 
            and is_serial_assign = 1
            and is_td_answer = 1,
            jichu.order_id,
            null
        )
    ) as serial_answer_cnt, -- 应答数
    count(
        distinct if(
            is_serial_assign = 1
            and is_td_answer = 1
            and is_td_cancel = 1
            and cancel_type = 2, -- order_status = 6 or 7
            jichu.order_id,
            null
        )
    ) as pas_can_cnt, --应答后乘客取消
    count(
        distinct if(
            is_serial_assign = 1
            and is_td_answer = 1
            and is_td_cancel = 1
            and cancel_type = 2, -- order_status = 6 or 7
            jichu.order_id,
            null
        )
    ) / count(
        distinct if(
            source_type <> 2 
            and is_serial_assign = 1
            and is_td_answer = 1,
            jichu.order_id,
            null
        )
    ) as pas_can_rate, -- 应答后乘客取消率
    count(
        distinct if (
            is_serial_assign = 1
            and is_td_answer = 1
            and (source_type <> 2 and cancel_type = 3),
            jichu.order_id,
            null
        )
    ) as dri_can_cnt, --应答后司机取消
    count(
        distinct if (
            is_serial_assign = 1
            and is_td_answer = 1
            and (source_type <> 2 and cancel_type = 3),
            jichu.order_id,
            null
        )
    ) / count(
        distinct if(
            source_type <> 2 
            and is_serial_assign = 1
            and is_td_answer = 1,
            jichu.order_id,
            null
        )
    ) as dri_can_rate, --应答后司机取消率
    count(
        distinct if (
            is_serial_assign = 1
            and is_td_answer = 1
            and (source_type <> 2 and is_td_reassign = 1),
            jichu.order_id,
            null
        )
    ) as reassign_cnt, -- 改派单
    count(
        distinct if (
            is_serial_assign = 1
            and is_td_answer = 1
            and (source_type <> 2 and is_td_reassign = 1),
            jichu.order_id,
            null
        )
    ) / count(
        distinct if(
            source_type <> 2 
            and is_serial_assign = 1
            and is_td_answer = 1,
            jichu.order_id,
            null
        )
    ) as reassign_rate, --改派率
    count(
        distinct if (
            is_serial_assign = 1
            and is_td_answer = 1
            and is_td_finish = 1,
            jichu.order_id,
            null
        )
    ) as serial_finish_cnt
from
    (
        select
            distinct param ['orderid'] as serial_order_id,
            shiftleft(
                shiftright(cast(param ['orderid'] as bigint), 46) & 262143,
                32
            ) +(cast(param ['orderid'] as bigint) & 4294967295) as serial_low_id,
            param ['driverid'] as serial_driver_id, -- 一个订单播单可能多次对应多个司机
            concat_ws('-', year, month, day) as yymmdd
        from
            kflower_ods.ods_log_kflower_orderbroadcast --日志底表
        where
            concat_ws('-', year, month, day) between '${begin}' and '${end}'
            and param ['is_serial_order'] = 1 -- 连环派的播单字段
            --and param ['timestamp'] >= '2022-02-14 12:00:00'
            and param ['area'] not in (8,26,32,6,11,24,21,58,13,25,38,85,12,50,3,34,15,33,160,1)
    ) bodan
    join ( -- left join 和 join在这里一致
        select
            order_id,
            -- 在order_base表里什么时候加distinct？需要弄清楚
            driver_id,
            is_td_answer,
            is_td_finish,
            is_serial_assign,
            source_type,
            order_status,
            is_td_cancel,
            cancel_type
        from
            kflower_dw.dwd_kf_trd_order_base_di
        where
            dt between '${begin}' and '${end}'
            and is_td_call = 1
            and (
                is_short_book_flag = 0
                or is_short_book_flag is null
            )
            and product_1level not in (3, 4) and order_type <> 1-- 花小猪自营去除早鸟单\滴滴网开台单
            --表单类型, 含聚合表单 and form_type <> 3
            and start_city_id not in (8,26,32,6,11,24,21,58,13,25,38,85,12,50,3,34,15,33,160,1)
            and call_time is not null
            and call_time <> '0000-00-00 00:00:00'
    ) jichu on bodan.serial_low_id = jichu.order_id
group by
    yymmdd
order by
    yymmdd
```

**拒单订单类型** 展开源码

```PlainText
select
    yymmdd,
    bodan.serial_low_id as judan_order_id,
    bodan.serial_driver_id as judan_driver_id,
    bodan.pre_total_fee as pre_total_fee, --播单显示金额更准
    --jichu.est_price_amt as est_price,
    jichu.estimate_dur as est_dur,
    matrix.first_dis as first_dis,
    matrix.second_dis as second_dis,
    matrix.first_eta as first_eta_sec,
    matrix.second_eta as second_eta_sec,
    float(matrix.first_eta) / 60.0 as first_eta,
    float(matrix.second_eta) / 60.0 as second_eta,
    float(bodan.pre_total_fee) / (float(matrix.first_eta + matrix.second_eta) / 60.0 + jichu.estimate_dur) as gph
from
    (
        select
            distinct param['orderid'] as serial_order_id,
            shiftleft(shiftright(cast(param['orderid'] as bigint), 46)&262143, 32)+(cast(param['orderid'] as bigint)&4294967295) as serial_low_id,
            param['driverid'] as serial_driver_id,
            param['pre_total_fee'] as pre_total_fee,
            concat_ws('-', year, month, day) as yymmdd
        from
            kflower_ods.ods_log_kflower_orderbroadcast --日志底表
        where
            concat_ws('-', year, month, day) between '${begin}' and '${end}'
            and param['is_serial_order'] = '1'
            and param['area'] not in (8,26,32,6,11,24,21,58,13,25,38,85,12,50,3,34,15,33,160,1)
    ) bodan
    join (
        select
            param ['orderid'] as serial_low_id,
            param ['driverid'] as serial_driver_id,
            param ['left_lumian_distance'] as first_dis,
            param ['dropoff_order_distance'] as second_dis,
            param ['left_lumian_eta'] as first_eta,
            param ['dropoff_order_eta'] as second_eta
        from
            kflower_ods.ods_log_kflower_stgmatrix --matrix日志
        where
            concat_ws('-', year, month, day) between '${begin}' and '${end}'
            and param ['is_serial_order'] = 1 -- 连环派的播单字段
            and param ['area'] not in (8,26,32,6,11,24,21,58,13,25,38,85,12,50,3,34,15,33,160,1)
            and param ['left_lumian_distance'] >= 0 and param ['dropoff_order_distance'] >= 0
            and param ['left_lumian_eta'] >= 0 and param ['dropoff_order_eta'] >= 0
    ) matrix on bodan.serial_low_id = matrix.serial_low_id and bodan.serial_driver_id = matrix.serial_driver_id
    join (
        select
            order_id,
            -- 在order_base表里什么时候加distinct？需要弄清楚
            is_td_answer,
            is_td_finish,
            is_serial_assign,
            source_type,
            order_status,
            is_td_cancel,
            cancel_type,
            est_price_amt,
            estimate_dur
        from kflower_dw.dwd_kf_trd_order_base_di
        where dt between '${begin}' and '${end}'
        and is_td_call = 1
        and (is_short_book_flag = 0 or is_short_book_flag is null) and order_type <> 1
        and product_1level not in (3, 4) -- 花小猪自营去除早鸟单\滴滴网开台单
        --and form_type <> 3 --表单类型, 非聚合表单
        and start_city_id not in (8,26,32,6,11,24,21,58,13,25,38,85,12,50,3,34,15,33,160,1)
        and call_time is not null and call_time <> '0000-00-00 00:00:00'
    ) jichu on bodan.serial_low_id = jichu.order_id
    join (
        select
            distinct cast(param ['oid'] as bigint) as order_id,
            cast(param ['driver_id'] as bigint) as driver_id,
            param ['is_serial'] as serial_judan
        from
            kflower_ods.ods_log_kflower_dispatch_order
        where
            concat_ws('-', year, month, day) between '${begin}' and '${end}'
            and param ['err_code'] = '0'
            --and param ['is_serial'] = '1'
            and param ['is_assign_order'] = '1'
            and param ['result'] = '1'
    ) judan on jichu.order_id = judan.order_id and bodan.serial_driver_id = judan.driver_id
```
