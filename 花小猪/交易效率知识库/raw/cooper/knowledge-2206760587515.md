    select
        passenger_id,
        count(distinct case when driver_id_3rd_part is not null and ac.level_type=105 then ac.order_id end) as handup_cnt_cc_tejiahezi,
        count(distinct case when driver_id_3rd_part is not null and (ac.level_type<>105 or ac.level_type is null) then ac.order_id end) as handup_cnt_cc_feitejiahezi,

        count(distinct case when is_td_finish = 1 then ac.order_id end) as api_fin_cnt_cc,
        count(distinct case when is_td_finish = 1 and ac.level_type=105 then ac.order_id end) as api_fin_cnt_cc_tejiahezi,
        count(distinct case when is_td_finish = 1 and (ac.level_type<>105 or ac.level_type is null) then ac.order_id end) as api_fin_cnt_cc_feitejiahezi,
        
        count(distinct case when is_td_answer = 1 then ac.order_id end) as ans_cnt_cc,
        count(distinct case when is_td_answer = 1 and ac.level_type=105 then ac.order_id end) as ans_cnt_cc_tejiahezi,
        count(distinct case when is_td_answer = 1 and (ac.level_type<>105 or ac.level_type is null) then ac.order_id end) as ans_cnt_cc_feitejiahezi
    from
        (
            select
                distinct
                dt,
                order_id,
                passenger_id,
                product_id,
                product_type,
                product_name,
                platform_name,
                is_pre_dispatch,
                level_type
            from
                kflower_dw.dwd_kf_trd_unity_order_anycar_di
            where
                dt between '2026-01-26' and '2026-02-08'
                and is_hxz_call = 1
                and product_type = '三方品牌' and  platform_name = 'API接入' and product_name like '%曹操%' 
        ) ac
        left join
        (
            select
                distinct
                dt,
                order_id,
                product_id,
                level_type,
                is_grab_before_cannel,
                is_td_finish,
                is_td_answer
            from
                kflower_dw.dwd_kf_trd_unity_order_di
            where
                dt between '2026-01-26' and '2026-02-08'
                and is_hxz = 1
        ) a on ac.order_id=a.order_id
            and ac.product_id=a.product_id
            and ac.level_type=a.level_type
        left join 
        (
            select
                shiftleft(
                    shiftright(cast(a.orderid as bigint), 46) & 262143,
                    32
                ) +(cast(a.orderid as bigint) & 4294967295) as order_id,
                driver_id_3rd_part,
                product_id,
                level_type
            from
                kflower_marketplace.ods_log_kflower_g_order_strivetender_v2_h a
            where
                concat_ws('-', year, month, day) between '2026-01-26' and '2026-02-08'
        ) handup
            on ac.order_id = handup.order_id
            and ac.product_id = handup.product_id
            and ac.level_type = handup.level_type
    group by
        passenger_id
),

raw as 
(
    select
        pid,
        exp_group,
        nvl(call_cnt_cc, 0) as call_cnt_cc,
        nvl(call_cnt_cc_tejiahezi, 0) as call_cnt_cc_tejiahezi,
        nvl(call_cnt_cc_feitejiahezi, 0) as call_cnt_cc_feitejiahezi,

        nvl(handup_cnt_cc_tejiahezi, 0) as handup_cnt_cc_tejiahezi,
        nvl(handup_cnt_cc_feitejiahezi, 0) as handup_cnt_cc_feitejiahezi,

        nvl(api_fin_cnt_cc_tejiahezi, 0) as api_fin_cnt_cc_tejiahezi,
        nvl(api_fin_cnt_cc_feitejiahezi, 0) as api_fin_cnt_cc_feitejiahezi,

        nvl(ans_cnt_cc_tejiahezi, 0) as ans_cnt_cc_tejiahezi,
        nvl(ans_cnt_cc_feitejiahezi, 0) as ans_cnt_cc_feitejiahezi
    from
        pas_tag a
    join
        bubble b
        on a.pid = b.passenger_id
    left join
        call c
        on a.pid = c.passenger_id
    left join
        finish d
        on a.pid = d.passenger_id
)
