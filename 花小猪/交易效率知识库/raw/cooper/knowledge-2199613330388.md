## 评估范围

由于触发文案的订单是所有播单的实时单，故只针对于这部分订单的应答前取消情况进行评估

## 评估指标

文案会在订单播单后的等待时间内出发，故评估：订单播单且未被拒单时的应答前取消率、订单播单后的应答前取消率、

```PlainText
select
    gp,
    count(
        distinct case
            when is_td_cancel = 1
            and cancel_type = 1
            and reject_cnt = 0 --没有拒单
            then base.order_id
            else null
        end
    ) / count(
        distinct case
            when source_type <> 2 then base.order_id
            else null
        end
    ),
    count(
        distinct case
            when is_td_cancel = 1
            and cancel_type = 1 then base.order_id
            else null
        end
    ) / count(
        distinct case
            when source_type <> 2 then base.order_id
            else null
        end
    ),
    count(
        distinct case
            when is_td_cancel = 1
            and cancel_type = 1
            and unix_timestamp(cancel_time) - unix_timestamp(start_brt_time) <= 14 then base.order_id
            else null
        end
    )/    count(
        distinct case
            when is_td_cancel = 1
            and cancel_type = 1 then base.order_id
            else null
        end
    ),
    sum(
        case
            when is_td_cancel = 1
            and cancel_type = 1 then unix_timestamp(cancel_time) - unix_timestamp(start_brt_time)
            else 0
        end
    ) / count(
        distinct case
            when is_td_cancel = 1
            and cancel_type = 1 then base.order_id
            else null
        end
    )
from
    (
        select
            dt,
            start_city_id,
            order_id,
            passenger_id,
            case
                when apollo_bucket('athena_dynamic_tip_experiment', passenger_id) < 25 then 'control'
                else 'treatment'
            end as gp,
            cancel_time,
            is_td_cancel,
            cancel_type,
            cancel_dur,
            source_type
        from
            kflower_dw.dwd_kf_trd_order_base_di
        where
            dt between '2021-07-23'
                    and '2021-07-24'
            and is_td_call = 1
            and call_time is not null
            and call_time <> '0000-00-00 00:00:00'
            and (
                is_short_book_flag = 0
                or is_short_book_flag is null
            ) --非短时预约
            and order_type <> 1 --非早鸟
    ) base
    inner join (
        select
            brt_order_id,
            min(brt_time) as start_brt_time,
            count(reject.driver_id) as reject_cnt
        from
            (
                select
                    distinct cast(param ['lowoid'] as bigint) as brt_order_id,
                    cast(param ['driverid'] as bigint) as driver_id,
                    param ['push_token'] as push_token,
                    param ['timestamp'] as brt_time
                from
                    kflower_ods.ods_log_kflower_orderbroadcast
                where
                    concat_ws('-', year, month, day) between '2021-07-23'
                    and '2021-07-24' --and cast(param['DispatchType'] as bigint) in (2)
            ) brt
            left join (
                select
                    distinct order_id as reject_order_id,
                    driver_id
                from
                    kflower_dw.dwd_kf_trd_refuse_order_di
                where
                    dt between '2021-07-23'
                    and '2021-07-24'
            ) reject on brt.brt_order_id = reject.reject_order_id
            and reject.driver_id = brt.driver_id
        group by
            brt_order_id
    ) brt_reject on base.order_id = brt_reject.brt_order_id
group by
    gp
order by
    gp
```

```PlainText
select
    count(base.order_id),
    count(fail.order_id),
    count(
        distinct case
            when unix_timestamp(cancel_time) - unix_timestamp(stamp) > 8 then fail.order_id
            else null
        end
    )
from
    (
        select
            dt, 
            start_city_id,
            order_id,
            cancel_time,
            passenger_id
        from
            kflower_dw.dwd_kf_trd_order_base_di
        where
            dt between '2021-07-23'
            and '2021-07-25'
            and is_td_call = 1
            and is_td_cancel = 1
            and cancel_type = 1
            and source_type <> 2
            and call_time is not null
            and call_time <> '0000-00-00 00:00:00'
            and (
                is_short_book_flag = 0
                or is_short_book_flag is null
            ) --非短时预约
            and order_type <> 1 --非早鸟
    ) base
    inner join(
        select
            order_id,
            max(pid) pid,
            max(timestamp) 
        from
            kflower_marketplace.ods_log_kflower_athena_dynamic_tip_log
        where
            concat_ws('-', year, month, day) between '2021-07-23'
            and '2021-07-25'
            and title_switch = 51
        group by order_id
    ) treat on base.order_id = treat.order_id
    and base.passenger_id = treat.pid
    left join(
        select
            distinct order_id,
            max(timestamp) as stamp,
            max(pid) as pid
        from
            kflower_marketplace.ods_log_kflower_athena_dynamic_tip_log
        where
            concat_ws('-', year, month, day) between '2021-07-23'
            and '2021-07-25'
            and title_switch = 52
        group by
            order_id
    ) fail on base.order_id = fail.order_id
    and base.passenger_id = fail.pid
```

```PlainText
    count(distinct cancel_order_id)/count(distinct card_order_id)

            attrs ['order_type'] as order_type,

    )cancelcard

        select
            distinct passengerid,
            udf_OrderDecode(attrs ['order_id']) as cancel_order_id,
            attrs ['page'] as page,
            attrs ['order_type'] as order_type,
            attrs ['popup_txt'] as popup_txt,

            kflower_dw.dwd_kf_log_pas_event_cn_di

    )cancelcardcancel on cancelcard.card_order_id = cancelcardcancel.cancel_order_id
```
