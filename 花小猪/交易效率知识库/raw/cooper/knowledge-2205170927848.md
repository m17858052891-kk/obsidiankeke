数据加工逻辑：举手时间在0-5s内

使用数据：1. 订单表

```PlainText
select  distinct a.source_id
        ,a.order_id
        ,a.call_time
        ,a.start_city_id as city_id
        ,a.answer_dur
        ,a.est_arrive_dis
        ,ceil(a.answer_dur / 5) * 5 as answer_dur_5
        ,a.dt
        ,order_base.is_serial_assign
        -- 关联获取连环派接驾距离
        ,serial_order.driver_start_eda as driver_start_eda
from    kflower_dw.dwd_kf_trd_unity_order_di a
left join (
    select  distinct dt
            ,order_id
            ,is_serial_assign
    from    kflower_dw.dwd_kf_trd_order_base_di
    where   dt between '2025-05-03'
      and   '2025-06-08'
      and   product_1level <> 4
) order_base
on      a.order_id = order_base.order_id
  and   a.dt = order_base.dt
left join (
    select  dt
            ,order_id
            ,driver_start_eda
            ,product_id
            ,call_car_level_id
            ,combo_type
            ,level_type
    from    kflower_dw.dwm_kf_trd_unity_trade_funnel_order_anycar_di
    where   dt between '2025-05-03'
      and   '2025-06-08'
) serial_order
on      a.order_id = serial_order.order_id
  and   a.dt = serial_order.dt
  and   a.product_id = serial_order.product_id
  and   a.call_car_level_id = serial_order.call_car_level_id
  and   coalesce(a.combo_type, 0) = coalesce(serial_order.combo_type, 0)
  and   coalesce(a.level_type, 0) = coalesce(serial_order.level_type, 0)
where   a.dt between '2025-05-03'
  and   '2025-06-08'
  and   a.is_hxz = 1
  and   a.product_id not in ('5135', '5150', '5075', '5149')
  and   a.is_td_answer = 1
  and   a.answer_dur > 0
  and   a.est_arrive_dis > 0
  and   order_base.is_serial_assign is not null
  -- 连环派需要有driver_start_eda数据，非连环派不需要
  and   (
                order_base.is_serial_assign = 0
            or  serial_order.driver_start_eda > 0
        )
;
```

2. 关联举手表

```PlainText
select  u.* -- 里面有O的结果是否 is_serial_assigin
        ,h.driver_id
        ,h.driver_id_3rd_part
        ,h.strive_timestamp
        ,h.ordertype
        ,h.group_key
        ,h.is_honghu_driver
        ,h.serial_type -- handup 0 非连环派 1 鸿鹄司机连环派 2 小猪司机连环派
        ,h.cpid
        ,h.cityid
        ,h.driver_start_distance
        ,h.old_driver_start_distance
from    tmp_unity_order_di u
left join (
    select  distinct
            (shiftleft(shiftright(CAST(a.orderid AS BIGINT), 46) & 262143, 32)       + (CAST(a.orderid AS BIGINT) & 4294967295)) AS order_id
            ,a.driverid AS driver_id
            ,a.driver_id_3rd_part
            ,a.strive_timestamp
            ,a.ordertype
            ,a.group_key
            ,a.is_honghu_driver
            ,a.serial_type
            ,a.cpid
            ,a.cityid
            ,a.driver_start_distance
            ,a.old_driver_start_distance
            ,concat_ws('-', a.year, a.month, a.day) AS dt
    from    kflower_ods.ods_log_kflower_g_order_strivetender a
    WHERE   concat_ws('-', a.year, a.month, a.day) BETWEEN '2025-05-03'
      and   '2025-06-08'
) h
ON      u.order_id = h.order_id
  and   u.dt = h.dt
  and   u.city_id = h.cityid
WHERE   u.dt BETWEEN '2025-05-03'
  and   '2025-06-08'
;
```

统计逻辑

```PlainText
SELECT 
    order_id,
    source_id,
    city_id,
    driver_id,
    strive_timestamp,
    call_time,
    driver_start_distance,
    cpid,
    is_honghu_driver,
    serial_type, -- 这个是举手的
    -- 商家分类
    CASE 
        WHEN cast(cpid as bigint) = 1373 THEN '自营'
        WHEN is_honghu_driver = true AND cast(cpid as bigint) <> 1373 THEN '宏鹄' 
        WHEN cast(cpid as bigint) IN (1676, 1219, 1220, 1056, 1309, 1181, 1157) THEN 'API'
        ELSE '其他'
    END as business_type,
    -- 举手时间（秒）
    (cast(strive_timestamp as bigint) / 1000.0 - unix_timestamp(call_time)) as handup_time_seconds
FROM tmp_unity_order_di_handup 
WHERE call_time IS NOT NULL 
  AND strive_timestamp IS NOT NULL
  AND driver_start_distance IS NOT NULL
  AND cast(driver_start_distance as double) > 0;
```
