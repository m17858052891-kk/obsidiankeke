```PlainText
with hxz_core_data as 
(
    select
        city_id,
        city_name,
        sum(call_cnt_1d_1018) / count(distinct dt) as call_cnt_per_day,
        sum(finish_cnt_1d_1028) / count(distinct dt) as finish_cnt_per_day,
        sum(gmv) / count(distinct dt) as gmv_per_day,
        sum(call_cnt_1d_1018) / sum(bubble_trace_cnt) as ecr,
        sum(finish_cnt_1d_1028) / sum(call_cnt_1d_1018) as cr
    from
        kflower_app.dm_kf_trd_core_index_di
    where
        dt between '2025-09-19' and '2025-09-25'
        and city_id > 0 
        and city_id < 10000
    group by
        city_id,
        city_name
),

hxz_total as 
(
    select
        sum(call_cnt_1d_1018) / count(distinct dt) as call_cnt_per_day_total,
        sum(finish_cnt_1d_1028) / count(distinct dt) as finish_cnt_per_day_total,
        sum(gmv) / count(distinct dt) as gmv_per_day_total
    from
        kflower_app.dm_kf_trd_core_index_di
    where
        dt between '2025-09-19' and '2025-09-25'
        and city_id > 0 
        and city_id < 10000
),

djh as 
(
    select city_id,
    sum(finish_cht)/count(distinct dt) as dajuhe_finish_cnt,
    sum(if (channel_name = '花小猪打车' and platform_mode = '花小猪自营',finish_cht,0))/count(distinct dt) as duannei_hxz_finish_cnt,
    sum(if (channel_name = '花小猪打车' and platform_mode = 'API',finish_cht,0))/count(distinct dt) as duannei_api_finish_cnt,
    sum(if (channel_name = '花小猪打车' and platform_mode = '宏鹄',finish_cht,0))/count(distinct dt) as duannei_hh_finish_cnt,
    sum(if (channel_name = '花小猪打车' and platform_mode = '滴滴运力',finish_cht,0))/count(distinct dt) as duannei_didi_finish_cnt
    from
    kflower_app.dm_kf_trd_unity_trade_funnel_plat_mode_di
    where
        dt between '2025-09-19' and '2025-09-25'
        and city_id > 0 
        and city_id < 10000
    group by city_id
),

frame as 
(
    select distinct 
        city_id
        ,city_group_name
        ,case
            when city_group_type = 2 then '整月分框'
            when city_group_type = 3 then '周中分框'
            when city_group_type = 4 then '周末分框'
            else '其他'
        end as fenkuang_type 
    from 
        kflower_ods.ods_binlog_t_city_group_d_whole
    where 
        concat_ws('-',year,month,day) = date_sub(current_date, 2)
        and city_group_type = 2 --只看全天分框
        and dt in ('2025-09') --根据具体月份改
),


fzb as 
(
    select 
        city_id,
        sum(serve_dur) / sum(online_dur) as fzb,
        sum(case when type = '自营' then serve_dur end) / sum(case when type = '自营' then online_dur end) as fzb_zy,
        sum(case when type = '宏鹄' then serve_dur end) / sum(case when type = '宏鹄' then online_dur end) as fzb_hh
    from 
    (
		select 
			'宏鹄' as type 
			,city_id
			,sum(case when is_online=1 then 1 else 0 end)/60 online_dur
            ,sum(case when is_busy=1 then 1 else 0 end)/60 serve_dur
		from 
			kflower_dw.dwd_hh_dri_online_patch_di   
		where 
			dt between '2025-09-19' and '2025-09-25'
			and company_id <> 1400
			and is_online = 1
        group by 
			city_id

		union all

		select 
			'自营' as type 
			,city_id
			,sum(case when is_online = 1 then 1 else 0 end)/60 online_dur
            ,sum(case when is_busy = 1 then 1 else 0 end)/60 serve_dur
		from 
			kflower_dw.dwd_kf_dri_online_patch_di   
		where 
			dt between '2025-09-19' and '2025-09-25'
			and product_id = 80
			and is_online = 1
        group by  
			city_id
    ) a 
    group by 
        city_id
),

raw as 
(
    select 
        a.*,
        finish_cnt_per_day / finish_cnt_per_day_total as `端内完单份额(占全国)`,
        gmv_per_day / gmv_per_day_total as `端内gmv份额(占全国)`,
        finish_cnt_per_day/dajuhe_finish_cnt as `端内完单占大聚合`,
        duannei_hxz_finish_cnt/finish_cnt_per_day as `城市端内自营份额`,
        duannei_hh_finish_cnt/finish_cnt_per_day as `城市端内宏鹄份额`,
        duannei_api_finish_cnt/finish_cnt_per_day as `城市端内api份额`,
        duannei_didi_finish_cnt/finish_cnt_per_day as `城市端内滴滴份额`,
        city_group_name as `供需分框`,
        fzb,
        fzb_zy,
        fzb_hh
    from 
        hxz_core_data a 
        left join frame b on a.city_id = b.city_id
        left join fzb c on a.city_id = c.city_id
        left join hxz_total d
        left join djh d on a.city_id = d.city_id
)

select * from raw
```
