**司机GMV_AA**

```PlainText
--PRESTO_SQL
--********************************************************************--
--author:samduzhan
--create time:2021-03-30 14:40:13
--desc:
--remind:请在资源引用中添加需要引用的资源,presto不能建表,无法使用自定义UDF
--********************************************************************--
select
    sum(gmv_amt)/collect_list(t2.cnt)[0] gmv,
    t1.exp_group
from
    (
        select
            driver_id as apollo_key,
            sum(gmv_amt) gmv_amt,
            start_city_id city,
            dt
        from
            kflower_dw.dwd_kf_trd_order_base_di
        where
            dt between '2021-03-12'
            and '2021-03-18'
            and is_td_finish = 1
            and start_city_id in (9, 60, 93, 94, 58, 3, 17, 19, 24, 86) --and start_city_id in (58, 3, 17, 19, 24, 86)
        group by
            driver_id,
            start_city_id,
            dt
    ) t0
    left join (
        select
            distinct apollo_key,
            ymd,
            exp_group
        from
            apollo.exp_group_data
        where
            ymd between '2021-03-23'
            and '2021-03-29'
            and experiment = 'heat_map'
    ) t1 on t0.apollo_key = t1.apollo_key
    left join
    (
        select
            count(*) cnt,
            exp_group
        from
            apollo.exp_group_data
        where
            ymd between '2021-03-23'
            and '2021-03-29'
            and experiment = 'heat_map'
        group by
            exp_group
    )t2 on t1.exp_group=t2.exp_group
where
    t1.exp_group is not null
group by
    t1.exp_group
order by
    t1.exp_group;
```

**司机TSH_AA**

```PlainText
--PRESTO_SQL
--********************************************************************--
--author:samduzhan
--create time:2021-03-30 14:40:03
--desc:
--remind:请在资源引用中添加需要引用的资源,presto不能建表,无法使用自定义UDF
--********************************************************************--
select
    sum(ta.online_dur)/collect_list(tb.cnt)[0] tsh,
    ta.exp_group
from
    (
        select
            city_id,
            t0.apollo_key,
            online_dur,
            exp_group
        from
            (
                select
                    driver_id as apollo_key,
                    sum(online_dur) online_dur,
                    dt,
                    city_id
                from
                    kflower_dw.dwm_kf_dri_online_di
                where
                    dt between '2021-03-12'
                    and '2021-03-18'
                    and city_id in (9, 60, 93, 94, 58, 3, 17, 19, 24, 86) --and city_id in (9, 60, 93, 94)
                group by
                    driver_id,
                    city_id,
                    dt
            ) t0
            left join (
                select
                    distinct apollo_key,
                    ymd,
                    exp_group
                from
                    apollo.exp_group_data
                where
                    ymd between '2021-03-23'
                    and '2021-03-29'
                    and experiment = 'heat_map'
            ) t1 on t0.apollo_key = t1.apollo_key
    ) ta
    left join
    (
        select
            count(*) cnt,
            exp_group
        from
            apollo.exp_group_data
        where
            ymd between '2021-03-23'
            and '2021-03-29'
            and experiment = 'heat_map'
        group by
            exp_group
    )tb on ta.exp_group=tb.exp_group
where
    ta.exp_group is not null
group by
    ta.exp_group
order by
    ta.exp_group;
```
