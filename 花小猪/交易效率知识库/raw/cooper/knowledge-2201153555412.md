--anycar发单表
drop table kevin_anycar_group_tmp;
create table kevin_anycar_group_tmp as
select
distinct order_id,
bubble_trace_id,
case when app_version >= '1.6.12'
and apollo.apollo_bucket(
    'athena_fold_cp_control',
    passenger_id
) < 15 then 'treatment_20min'
when app_version >= '1.6.12'
and apollo.apollo_bucket(
    'athena_fold_cp_control',
    passenger_id
) >= 30
and apollo.apollo_bucket(
    'athena_fold_cp_control',
    passenger_id
) < 45 then 'treatment_10min'
when app_version >= '1.6.12' then 'control_group'
else null end exp_group
from kflower_dw.dwd_kf_trd_order_anycar_di
where dt >= '2023-10-21' and dt <= '2023-10-27' 
and product_2level in (80,81,82);

drop table kevin_fold_group_tmp;
create table kevin_fold_group_tmp as  
select
distinct trace_id,
is_fold_cp
from kflower_dw.dwd_kf_trd_bubble_base_all_di
where dt >= '2023-10-21' and dt <= '2023-10-27' 
and product_2level in (80,81,82);

--分单日志
drop table kevin_log_group_tmp;
create table kevin_log_group_tmp as 
select distinct 
cast(param['orderid'] as bigint) order_id,
cast(param['passenger_id'] as bigint) passenger_id
from kflower_ods.ods_log_kflower_stgmatrix
where concat_ws('-',year,month,day) >= '2023-10-21'
and concat_ws('-',year,month,day) <= '2023-10-27'
and param['broad_source'] = 'duse_stg';

--准确率和召回率
select 
a.exp_group,
sum(if(b.is_fold_cp=1 and c.order_id is null, 1, 0)) as PR,
sum(if(b.is_fold_cp=1, 1, 0)) as P,
sum(if(c.order_id is null, 1, 0)) as R,
sum(if(b.is_fold_cp=1 and c.order_id is null, 1, 0)) / sum(if(b.is_fold_cp=1, 1, 0)) precsion,
sum(if(b.is_fold_cp=1 and c.order_id is null, 1, 0)) / sum(if(c.order_id is null, 1, 0)) recall
from kevin_anycar_group_tmp a left join 
kevin_fold_group_tmp b on a.bubble_trace_id = b.trace_id
left join kevin_log_group_tmp c
on a.order_id = c.order_id
group by a.exp_group;
